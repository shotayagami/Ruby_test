## テストの基本概念

### テストの種類

```
┌─────────────────────────────────────────────────────┐
│                   E2Eテスト                          │  ← 少なく
│        （ブラウザ操作、システム全体）                  │
├─────────────────────────────────────────────────────┤
│              インテグレーションテスト                  │
│          （複数コンポーネントの連携）                  │
├─────────────────────────────────────────────────────┤
│                  ユニットテスト                       │  ← 多く
│            （個々のクラス・メソッド）                  │
└─────────────────────────────────────────────────────┘
              テストピラミッド
```

| テストの種類 | 対象 | 実行速度 | 数 |
|-------------|------|---------|-----|
| ユニットテスト | 個々のメソッド・クラス | 速い | 多い |
| インテグレーションテスト | コンポーネント間の連携 | 中程度 | 中程度 |
| E2Eテスト | システム全体 | 遅い | 少ない |

### テスト駆動開発（TDD）

```
┌──────────┐     ┌──────────┐     ┌──────────┐
│  Red     │ ──→ │  Green   │ ──→ │ Refactor │
│ テスト失敗│     │ テスト成功│     │ コード改善│
└──────────┘     └──────────┘     └──────────┘
     ↑                                  │
     └──────────────────────────────────┘
```

1. **Red**: 失敗するテストを書く
2. **Green**: テストを通す最小限のコードを書く
3. **Refactor**: コードを改善する（テストは通ったまま）

### BDD（振る舞い駆動開発）

```ruby
# BDDスタイル - 振る舞いを記述
describe "ユーザー登録" do
  context "有効なデータの場合" do
    it "ユーザーが作成される" do
      # ...
    end
  end

  context "無効なデータの場合" do
    it "エラーメッセージが表示される" do
      # ...
    end
  end
end
```

## Minitest（標準ライブラリ）

MinitestはRubyの標準ライブラリに含まれるテストフレームワークです。

### セットアップ

```ruby
# test/test_helper.rb
require 'minitest/autorun'
require 'minitest/pride'  # カラフルな出力（オプション）

# テスト対象のコードを読み込み
require_relative '../lib/calculator'
```

### アサーションスタイル

```ruby
# test/calculator_test.rb
require_relative 'test_helper'

class CalculatorTest < Minitest::Test
  def setup
    @calculator = Calculator.new
  end

  def teardown
    # クリーンアップ処理
  end

  # テストメソッドは test_ で始める
  def test_add_two_numbers
    result = @calculator.add(2, 3)
    assert_equal 5, result
  end

  def test_subtract_two_numbers
    result = @calculator.subtract(10, 4)
    assert_equal 6, result
  end

  def test_divide_by_zero_raises_error
    assert_raises(ZeroDivisionError) do
      @calculator.divide(10, 0)
    end
  end
end
```

### Specスタイル

```ruby
# test/calculator_spec.rb
require_relative 'test_helper'

describe Calculator do
  before do
    @calculator = Calculator.new
  end

  describe '#add' do
    it 'adds two numbers' do
      _(@calculator.add(2, 3)).must_equal 5
    end

    it 'handles negative numbers' do
      _(@calculator.add(-1, 1)).must_equal 0
    end
  end

  describe '#divide' do
    it 'divides two numbers' do
      _(@calculator.divide(10, 2)).must_equal 5
    end

    it 'raises error when dividing by zero' do
      _ { @calculator.divide(10, 0) }.must_raise ZeroDivisionError
    end
  end
end
```

### 主なアサーション

```ruby
# 等価性
assert_equal expected, actual          # expected == actual
refute_equal unexpected, actual        # unexpected != actual

# 真偽値
assert value                           # value が truthy
refute value                           # value が falsey
assert_nil value                       # value.nil?
refute_nil value                       # !value.nil?

# 型チェック
assert_instance_of String, value       # value.instance_of?(String)
assert_kind_of Numeric, value          # value.kind_of?(Numeric)

# 含む
assert_includes collection, item       # collection.include?(item)
assert_match /pattern/, string         # string =~ /pattern/

# 例外
assert_raises(ErrorClass) { code }     # code が ErrorClass を発生
assert_silent { code }                 # code が何も出力しない

# 出力
assert_output("expected\n") { puts "expected" }

# 差分
assert_in_delta 3.14, result, 0.01     # 浮動小数点の比較

# 空
assert_empty collection                # collection.empty?
refute_empty collection                # !collection.empty?
```

### 実行

```bash
# 全テスト実行
ruby -Itest test/calculator_test.rb

# 特定のテスト
ruby -Itest test/calculator_test.rb --name test_add_two_numbers

# パターンマッチ
ruby -Itest test/calculator_test.rb --name /add/

# Rakeタスク
rake test
```

### Rakefile の設定

```ruby
# Rakefile
require 'rake/testtask'

Rake::TestTask.new(:test) do |t|
  t.libs << 'test'
  t.libs << 'lib'
  t.test_files = FileList['test/**/*_test.rb']
  t.verbose = true
end

task default: :test
```

## RSpec（BDDフレームワーク）

RSpecはRubyで最も広く使われているテストフレームワークです。

### セットアップ

```bash
# インストール
gem install rspec

# プロジェクト初期化
rspec --init
```

```ruby
# Gemfile
group :test do
  gem 'rspec', '~> 3.12'
end
```

### 基本構文

```ruby
# spec/calculator_spec.rb
require_relative '../lib/calculator'

RSpec.describe Calculator do
  # テスト対象のインスタンス
  let(:calculator) { Calculator.new }

  describe '#add' do
    it '2つの数を足し算する' do
      expect(calculator.add(2, 3)).to eq(5)
    end

    it '負の数も扱える' do
      expect(calculator.add(-1, 1)).to eq(0)
    end
  end

  describe '#divide' do
    context '正常な場合' do
      it '2つの数を割り算する' do
        expect(calculator.divide(10, 2)).to eq(5)
      end
    end

    context 'ゼロで割る場合' do
      it '例外を発生させる' do
        expect { calculator.divide(10, 0) }.to raise_error(ZeroDivisionError)
      end
    end
  end
end
```

### subject と let

```ruby
RSpec.describe User do
  # 暗黙のsubject（クラス名から自動生成）
  subject { User.new(name: 'Alice', age: 25) }

  # 名前付きsubject
  subject(:user) { User.new(name: 'Bob', age: 30) }

  # let（遅延評価）
  let(:valid_attributes) { { name: 'Charlie', email: 'charlie@example.com' } }

  # let!（即時評価）
  let!(:admin) { User.create(name: 'Admin', role: :admin) }

  it { is_expected.to be_valid }

  it 'has a name' do
    expect(user.name).to eq('Bob')
  end
end
```

### context でグループ化

```ruby
RSpec.describe Order do
  let(:order) { Order.new(items: items) }

  context '商品がある場合' do
    let(:items) { [Item.new(price: 100), Item.new(price: 200)] }

    it '合計金額を計算する' do
      expect(order.total).to eq(300)
    end

    it '空ではない' do
      expect(order).not_to be_empty
    end
  end

  context '商品がない場合' do
    let(:items) { [] }

    it '合計金額は0' do
      expect(order.total).to eq(0)
    end

    it '空である' do
      expect(order).to be_empty
    end
  end
end
```

### before / after フック

```ruby
RSpec.describe Database do
  before(:all) do
    # 全テスト前に1回だけ
    DatabaseCleaner.clean_with(:truncation)
  end

  before(:each) do
    # 各テスト前
    @connection = Database.connect
  end

  after(:each) do
    # 各テスト後
    @connection.close
  end

  after(:all) do
    # 全テスト後に1回だけ
    Database.disconnect_all
  end

  # around フック
  around(:each) do |example|
    Database.transaction do
      example.run
      raise ActiveRecord::Rollback
    end
  end
end
```

### マッチャー

```ruby
# 等価性
expect(actual).to eq(expected)         # ==
expect(actual).to eql(expected)        # eql?
expect(actual).to equal(expected)      # equal?（同一オブジェクト）
expect(actual).to be(expected)         # equal?

# 比較
expect(10).to be > 5
expect(10).to be >= 10
expect(10).to be < 20
expect(10).to be_between(5, 15).inclusive

# 真偽値
expect(true).to be true
expect(false).to be false
expect(nil).to be_nil
expect(nil).to be_falsey
expect(1).to be_truthy

# 型
expect('hello').to be_a(String)
expect([]).to be_an(Array)
expect(1).to be_a_kind_of(Numeric)

# コレクション
expect([1, 2, 3]).to include(2)
expect([1, 2, 3]).to contain_exactly(3, 1, 2)  # 順序無視
expect([1, 2, 3]).to match_array([3, 1, 2])    # 同上
expect([1, 2, 3]).to start_with(1)
expect([1, 2, 3]).to end_with(3)
expect([]).to be_empty
expect([2, 4, 6]).to all(be_even)

# 文字列
expect('hello').to include('ell')
expect('hello').to start_with('he')
expect('hello').to match(/^he/)

# 例外
expect { raise 'error' }.to raise_error
expect { raise ArgumentError, 'bad' }.to raise_error(ArgumentError, /bad/)
expect { safe_op }.not_to raise_error

# 変化
expect { x += 1 }.to change { x }.by(1)
expect { user.grow }.to change(user, :age).from(20).to(21)

# 出力
expect { puts 'hello' }.to output("hello\n").to_stdout
expect { warn 'oops' }.to output(/oops/).to_stderr

# 述語（be_xxx → xxx? を呼ぶ）
expect(user).to be_valid       # user.valid?
expect(user).to be_admin       # user.admin?
expect([]).to be_empty         # [].empty?

# have_xxx → has_xxx? を呼ぶ
expect(user).to have_permission(:read)  # user.has_permission?(:read)
```

## テストダブル（モック・スタブ）

外部依存を模倣するためのテストダブルについて解説します。

### 用語

| 用語 | 説明 |
|-----|------|
| ダブル | テスト用の偽オブジェクト全般 |
| スタブ | 特定の戻り値を返す |
| モック | 呼び出しを検証する |
| スパイ | 呼び出しを記録して後で検証 |
| フェイク | 簡易実装を持つ代替オブジェクト |

### RSpec でのモック・スタブ

```ruby
RSpec.describe PaymentService do
  describe '#process' do
    # ダブル（テスト用偽オブジェクト）
    let(:gateway) { instance_double(PaymentGateway) }
    let(:service) { PaymentService.new(gateway) }

    context '支払い成功' do
      before do
        # スタブ（戻り値を設定）
        allow(gateway).to receive(:charge).and_return(true)
      end

      it '成功を返す' do
        expect(service.process(100)).to be true
      end
    end

    context '支払い処理の検証' do
      it 'ゲートウェイを呼び出す' do
        # モック（呼び出しを期待）
        expect(gateway).to receive(:charge).with(100).and_return(true)

        service.process(100)
      end
    end
  end
end
```

### スタブの詳細

```ruby
# 基本的なスタブ
allow(object).to receive(:method).and_return(value)

# 引数に応じた戻り値
allow(object).to receive(:method).with(1).and_return('one')
allow(object).to receive(:method).with(2).and_return('two')

# 複数回の呼び出し
allow(object).to receive(:method).and_return(1, 2, 3)
# 1回目→1, 2回目→2, 3回目以降→3

# ブロックで動的な戻り値
allow(object).to receive(:method) do |arg|
  arg * 2
end

# 例外を発生
allow(object).to receive(:method).and_raise(StandardError, 'error')

# 元のメソッドを呼ぶ
allow(object).to receive(:method).and_call_original

# 何でも受け入れる
allow(object).to receive(:method).with(anything).and_return(true)
allow(object).to receive(:method).with(any_args).and_return(true)
```

### モックの詳細

```ruby
# 呼び出しを期待
expect(object).to receive(:method)

# 引数を検証
expect(object).to receive(:method).with(1, 2, 3)
expect(object).to receive(:method).with(hash_including(key: 'value'))
expect(object).to receive(:method).with(array_including(1, 2))

# 呼び出し回数
expect(object).to receive(:method).once
expect(object).to receive(:method).twice
expect(object).to receive(:method).exactly(3).times
expect(object).to receive(:method).at_least(:once)
expect(object).to receive(:method).at_most(5).times

# 順序
expect(object).to receive(:first).ordered
expect(object).to receive(:second).ordered
```

### スパイ

```ruby
RSpec.describe Logger do
  it 'logs messages' do
    # スパイ（全メソッドを記録）
    logger = spy('logger')

    service = Service.new(logger)
    service.perform

    # 後から検証
    expect(logger).to have_received(:info).with('Started')
    expect(logger).to have_received(:info).with('Completed')
  end
end
```

### 部分的ダブル

```ruby
RSpec.describe User do
  it 'sends email' do
    user = User.new(email: 'test@example.com')

    # 既存オブジェクトの一部をスタブ
    allow(user).to receive(:send_email).and_return(true)

    expect(user.welcome).to be true
    expect(user).to have_received(:send_email)
  end
end
```

### Minitest でのモック

```ruby
require 'minitest/autorun'
require 'minitest/mock'

class PaymentServiceTest < Minitest::Test

    gateway = Minitest::Mock.new
    gateway.expect(:charge, true, [100])

    service = PaymentService.new(gateway)
    result = service.process(100)

    gateway.verify
  end
end

      result = MyService.new.call
      assert_equal 'stubbed response', result
    end
  end
end
```

## CI/CD連携

### GitHub Actions

```yaml
# .github/workflows/test.yml
name: Ruby Tests

on:
  push:
    branches: [main, develop]
  pull_request:
    branches: [main]

jobs:
  test:
    runs-on: ubuntu-latest

    services:
      postgres:
        image: postgres:15
        env:
          POSTGRES_PASSWORD: postgres
        ports:
          - 5432:5432
        options: >-
          --health-cmd pg_isready
          --health-interval 10s
          --health-timeout 5s
          --health-retries 5

    steps:
      - uses: actions/checkout@v4

      - name: Set up Ruby
        uses: ruby/setup-ruby@v1
        with:
          ruby-version: '3.3'
          bundler-cache: true

      - name: Setup database
        env:
          DATABASE_URL: postgres://postgres:postgres@localhost:5432/test
        run: |
          bundle exec rake db:create db:schema:load

      - name: Run tests
        env:
          DATABASE_URL: postgres://postgres:postgres@localhost:5432/test
        run: bundle exec rspec --format documentation

      - name: Upload coverage
        uses: codecov/codecov-action@v3
        with:
          files: coverage/coverage.json
```

## テストのベストプラクティス

### 1. AAAパターン

```ruby
it 'calculates total price' do
  # Arrange（準備）
  cart = Cart.new
  cart.add_item(Item.new(price: 100))
  cart.add_item(Item.new(price: 200))

  # Act（実行）
  total = cart.total

  # Assert（検証）
  expect(total).to eq(300)
end
```

### 2. 1テスト1アサーション

```ruby
# 悪い例
it 'creates a user' do
  user = User.create(name: 'Alice', email: 'alice@example.com')
  expect(user).to be_persisted
  expect(user.name).to eq('Alice')
  expect(user.email).to eq('alice@example.com')
end

# 良い例
describe 'creating a user' do
  subject(:user) { User.create(name: 'Alice', email: 'alice@example.com') }

  it { is_expected.to be_persisted }
  it { is_expected.to have_attributes(name: 'Alice') }
  it { is_expected.to have_attributes(email: 'alice@example.com') }
end
```

### 3. 説明的なテスト名

```ruby
# 悪い例
it 'works' do
  expect(user.valid?).to be true
end

# 良い例
it 'is valid with valid attributes' do
  expect(user).to be_valid
end

it 'is invalid without an email' do
  user.email = nil
  expect(user).not_to be_valid
end
```

### 4. テストの独立性

```ruby
# 悪い例（テスト間の依存）
describe User do
  it 'creates user' do
    @user = User.create(name: 'Alice')
    expect(@user).to be_persisted
  end

  it 'updates user' do
    @user.update(name: 'Bob')  # 前のテストに依存
    expect(@user.name).to eq('Bob')
  end
end

# 良い例
describe User do
  let(:user) { create(:user, name: 'Alice') }

  it 'creates user' do
    expect(user).to be_persisted
  end

  it 'updates user' do
    user.update(name: 'Bob')
    expect(user.name).to eq('Bob')
  end
end
```

### 5. 実装ではなく振る舞いをテスト

```ruby
# 悪い例（実装の詳細に依存）
it 'stores items in @items array' do
  cart.add_item(item)
  expect(cart.instance_variable_get(:@items)).to include(item)
end

# 良い例（振る舞いをテスト）
it 'includes added item' do
  cart.add_item(item)
  expect(cart.items).to include(item)
end
```

### 6. 境界値テスト

```ruby
describe '#adult?' do
  context 'at boundary' do
    it 'returns false for age 17' do
      expect(User.new(age: 17)).not_to be_adult
    end

    it 'returns true for age 18' do
      expect(User.new(age: 18)).to be_adult
    end

    it 'returns true for age 19' do
      expect(User.new(age: 19)).to be_adult
    end
  end
end
```

### 7. 適切なテストダブルの使用

```ruby
# 外部サービスはモック
it 'sends notification' do
  notifier = instance_double(SlackNotifier)
  allow(notifier).to receive(:send)

  service = OrderService.new(notifier: notifier)
  service.complete(order)

  expect(notifier).to have_received(:send).with(/Order completed/)
end

# 自分のコードはなるべく実際のオブジェクトを使う
it 'calculates order total' do
  order = Order.new(items: [
    Item.new(price: 100),
    Item.new(price: 200)
  ])

  expect(order.total).to eq(300)
end
```
