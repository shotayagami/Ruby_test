## 1. モジュールとは

### モジュールの基本

```ruby
# モジュールの定義
module Greetable
  def greet
    "Hello!"
  end
end

# モジュールはインスタンス化できない
# Greetable.new  # NoMethodError!

# モジュールのメソッドを直接呼び出すには module_function
module MathUtils
  def self.square(n)
    n * n
  end

  # または module_function
  module_function

  def cube(n)
    n ** 3
  end
end

MathUtils.square(4)  # => 16
MathUtils.cube(3)    # => 27
```

**🐍 Python:** Rubyの「インスタンス化できない機能の入れ物」としてのモジュールは、Pythonでは「ファイル（.py）そのもの」に相当する。関数はモジュールのトップレベルに定義し、`import` して呼び出す。

```python
# math_utils.py
def square(n):
    return n * n

def cube(n):
    return n ** 3

# 別ファイルから
import math_utils
math_utils.square(4)  # => 16
math_utils.cube(3)    # => 27
```

Rubyの `module_function` や `def self.xxx` に相当するのは、単なるモジュールレベルの関数。Pythonのモジュールはそもそもオブジェクトではなくファイルなので、「インスタンス化できない」という制約自体が存在しない。

### クラスとモジュールの違い

| 特徴 | クラス | モジュール |
|-----|--------|----------|
| インスタンス化 | 可能 | 不可 |
| 継承 | 可能（単一継承） | 不可 |
| ミックスイン | される側 | する側 |
| 用途 | オブジェクトの設計図 | 機能の共有、名前空間 |

## 2. 名前空間としてのモジュール

### 基本的な名前空間

```ruby
# モジュールで名前空間を作成
module MyApp
  class User
    def initialize(name)
      @name = name
    end
  end

  class Admin < User
    def permissions
      [:read, :write, :delete]
    end
  end

  # 定数もスコープ内
  VERSION = "1.0.0"
end

# アクセス
user = MyApp::User.new("太郎")
admin = MyApp::Admin.new("管理者")
version = MyApp::VERSION
```

**🐍 Python:** 名前空間はファイル（モジュール）やディレクトリ（パッケージ）で表現し、`import` で取り込む。Rubyの `::` はPythonの `.` に相当する。

```python
# myapp/user.py
class User:
    def __init__(self, name):
        self.name = name

class Admin(User):
    def permissions(self):
        return ["read", "write", "delete"]

VERSION = "1.0.0"

# 使う側
from myapp import user
u = user.User("太郎")
version = user.VERSION
# あるいは import myapp.user として myapp.user.User("太郎")
```

Rubyの `MyApp::User`（区切りは `::`）は、Pythonでは `myapp.user.User`（区切りは `.`）と書く。Rubyは1ファイル内に複数の名前空間を自由にネストできるが、Pythonは基本的にファイル/ディレクトリの構成がそのまま名前空間（モジュール/パッケージ）になる。

### ネストした名前空間

```ruby
module Company
  module HR
    class Employee
      attr_reader :name
      def initialize(name)
        @name = name
      end
    end
  end

  module Finance
    class Invoice
      attr_reader :amount
      def initialize(amount)
        @amount = amount
      end
    end
  end
end

employee = Company::HR::Employee.new("山田")
invoice = Company::Finance::Invoice.new(10000)

# 短縮形での定義
class Company::HR::Manager < Company::HR::Employee
  def manage
    "管理中"
  end
end
```

### 名前空間とautoload

```ruby
# 遅延読み込み
module MyApp
  autoload :User, 'my_app/user'
  autoload :Admin, 'my_app/admin'
  autoload :Config, 'my_app/config'
end

# User を使うときに初めて読み込まれる
MyApp::User.new("太郎")
```

### 定数の解決

```ruby
module A
  CONST = "A"

  module B
    CONST = "B"

    class C
      CONST = "C"

      def print_constants
        puts CONST       # => "C"（最も近いスコープ）
        puts B::CONST    # => "B"
        puts A::CONST    # => "A"
        puts ::A::CONST  # => "A"（トップレベルから）
      end
    end
  end
end

A::B::C.new.print_constants

# 定数の検索順序
# 1. 現在のスコープ
# 2. 外側のスコープ（レキシカルスコープ）
# 3. 継承チェーン
```

## 3. ミックスイン

### 基本的なミックスイン

```ruby
# 共有する機能をモジュールで定義
module Loggable
  def log(message, level: :info)
    puts "[#{level.upcase}] #{self.class}: #{message}"
  end

  def debug(message)
    log(message, level: :debug)
  end

  def error(message)
    log(message, level: :error)
  end
end

# クラスにミックスイン
class User
  include Loggable

  def initialize(name)
    @name = name
    log("User created: #{@name}")
  end
end

class Order
  include Loggable

  def process
    log("Processing order")
    debug("Order details...")
  end
end

user = User.new("太郎")
# => [INFO] User: User created: 太郎
```

**🐍 Python:** Pythonにモジュールのミックスイン機構はなく、機能の共有は「多重継承」で行う。共有したい機能をクラス（慣習的に `Mixin` と名付ける）として定義し、それを継承する。

```python
class LoggableMixin:
    def log(self, message, level="info"):
        print(f"[{level.upper()}] {type(self).__name__}: {message}")

    def debug(self, message):
        self.log(message, level="debug")

    def error(self, message):
        self.log(message, level="error")

class User(LoggableMixin):
    def __init__(self, name):
        self.name = name
        self.log(f"User created: {name}")
```

思想の差が重要: Rubyは「モジュール（インスタンス化できない機能の束）を `include` して継承チェーンに差し込む」。Pythonは「クラスを多重継承し、メソッド解決順序（MRO, C3線形化）に沿って探索する」。Rubyの `include` に相当するものをPythonは通常の基底クラスとして表現し、`super()` もMROに従って次のクラスへ委譲する。

### 複数のモジュールをミックスイン

```ruby
module Comparable
  # <=> を定義すれば他の比較演算子が使える
end

module Enumerable
  # each を定義すれば map, select などが使える
end

module Serializable
  def to_json
    # JSON変換
  end

  def to_xml
    # XML変換
  end
end

module Validatable
  def valid?
    errors.empty?
  end

  def errors
    @errors ||= []
  end
end

class User
  include Comparable
  include Serializable
  include Validatable

  attr_reader :name, :age

  def initialize(name, age)
    @name = name
    @age = age
  end

  def <=>(other)
    age <=> other.age
  end
end
```

### モジュールでのインスタンス変数

```ruby
module Timestampable
  def created_at
    @created_at
  end

  def updated_at
    @updated_at
  end

  def touch
    @updated_at = Time.now
  end

  # includeされたクラスの初期化をフック
  def self.included(base)
    base.class_eval do
      alias_method :original_initialize, :initialize

      define_method(:initialize) do |*args, &block|
        @created_at = Time.now
        @updated_at = Time.now
        original_initialize(*args, &block)
      end
    end
  end
end
```

## 4. include vs extend vs prepend

### include

```ruby
module Greetable
  def greet
    "Hello!"
  end
end

class User
  include Greetable  # インスタンスメソッドとして追加
end

user = User.new
user.greet  # => "Hello!"

User.ancestors
# => [User, Greetable, Object, Kernel, BasicObject]
```

### extend

```ruby
module ClassMethods
  def find_all
    "Finding all..."
  end
end

class User
  extend ClassMethods  # クラスメソッドとして追加
end

User.find_all  # => "Finding all..."
# User.new.find_all  # NoMethodError!

# 特定のオブジェクトに extend
obj = Object.new
obj.extend(Greetable)
obj.greet  # => "Hello!"
```

### prepend

```ruby
module Logging
  def save
    puts "Saving..."
    result = super  # 元のメソッドを呼ぶ
    puts "Saved!"
    result
  end
end

class User
  prepend Logging  # 継承チェーンの先頭に追加

  def save
    puts "User#save"
    true
  end
end

user = User.new
user.save
# => Saving...
#    User#save
#    Saved!

User.ancestors
# => [Logging, User, Object, Kernel, BasicObject]
# prepend されたモジュールが User より前に来る
```

### 比較表

| メソッド | 追加先 | ancestors での位置 | 用途 |
|---------|--------|-------------------|------|
| include | インスタンス | クラスの後ろ | 機能の追加 |
| extend | クラス（特異クラス） | - | クラスメソッドの追加 |
| prepend | インスタンス | クラスの前 | メソッドのラップ |

### include と extend を同時に

```ruby
module FullFeature
  def self.included(base)
    base.extend(ClassMethods)
  end

  # インスタンスメソッド
  def instance_method
    "instance"
  end

  # クラスメソッド
  module ClassMethods
    def class_method
      "class"
    end
  end
end

class User
  include FullFeature
end

User.class_method     # => "class"
User.new.instance_method  # => "instance"
```

## 5. モジュールのコールバック

### included / extended / prepended

```ruby
module MyModule
  def self.included(base)
    puts "#{self} was included in #{base}"
    # base はincludeしたクラス/モジュール
    base.class_eval do
      # クラスに対する設定を行える
    end
  end

  def self.extended(base)
    puts "#{self} was extended by #{base}"
  end

  def self.prepended(base)
    puts "#{self} was prepended to #{base}"
  end
end

class User
  include MyModule  # => "MyModule was included in User"
end
```

### method_added / method_removed

```ruby
module MethodTracker
  def self.included(base)
    base.extend(ClassMethods)
  end

  module ClassMethods
    def method_added(method_name)
      puts "Method added: #{method_name}"
      super
    end

    def method_removed(method_name)
      puts "Method removed: #{method_name}"
      super
    end
  end
end

class User
  include MethodTracker

  def greet  # => "Method added: greet"
    "Hello"
  end

  remove_method :greet  # => "Method removed: greet"
end
```

### inherited（クラス継承時）

```ruby
class Base
  def self.inherited(subclass)
    puts "#{subclass} inherited from #{self}"
    # サブクラスの設定などを行える
  end
end

class Child < Base  # => "Child inherited from Base"
end
```

## 6. Concern パターン

### ActiveSupport::Concern

```ruby
# Rails の Concern パターン
# require 'active_support/concern'

module Trackable
  extend ActiveSupport::Concern

  # インスタンスメソッド
  def track(event)
    puts "Tracking: #{event}"
  end

  # クラスメソッドは ClassMethods モジュールに
  module ClassMethods
    def tracked_events
      @tracked_events ||= []
    end

    def track_event(name)
      tracked_events << name
    end
  end

  # included ブロックでクラスマクロを呼べる
  included do
    track_event :created
    track_event :updated
  end
end

class User
  include Trackable
end

User.tracked_events  # => [:created, :updated]
```

### Concern パターンの自前実装

```ruby
module Concern
  def self.extended(base)
    base.instance_variable_set(:@_included_block, nil)
  end

  def included(base = nil, &block)
    if block
      @_included_block = block
    else
      super
    end
  end

  def append_features(base)
    super
    base.extend(const_get(:ClassMethods)) if const_defined?(:ClassMethods)
    base.class_eval(&@_included_block) if @_included_block
  end
end

# 使用例
module Taggable
  extend Concern

  module ClassMethods
    def taggable?
      true
    end
  end

  included do
    attr_accessor :tags
  end

  def add_tag(tag)
    self.tags ||= []
    self.tags << tag
  end
end
```

## 7. 標準ライブラリのモジュール

### Comparable

```ruby
class Version
  include Comparable

  attr_reader :major, :minor, :patch

  def initialize(version_string)
    @major, @minor, @patch = version_string.split('.').map(&:to_i)
  end

  # <=> を実装すると <, <=, ==, >=, >, between? が使える
  def <=>(other)
    [major, minor, patch] <=> [other.major, other.minor, other.patch]
  end

  def to_s
    "#{major}.#{minor}.#{patch}"
  end
end

v1 = Version.new("1.2.3")
v2 = Version.new("1.2.4")
v3 = Version.new("2.0.0")

v1 < v2   # => true
v1 > v2   # => false
v2.between?(v1, v3)  # => true
[v3, v1, v2].sort    # => [v1, v2, v3]
```

**🐍 Python:** RubyのComparable（`<=>` を実装すると比較演算子が揃う）に相当するのは `functools.total_ordering`。`__eq__` と `__lt__` を定義すれば残りの `<=`, `>`, `>=` が自動生成される。

```python
from functools import total_ordering

@total_ordering
class Version:
    def __init__(self, version_string):
        self.major, self.minor, self.patch = (
            int(x) for x in version_string.split(".")
        )

    def _key(self):
        return (self.major, self.minor, self.patch)

    def __eq__(self, other):
        return self._key() == other._key()

    def __lt__(self, other):
        return self._key() < other._key()

    def __str__(self):
        return f"{self.major}.{self.minor}.{self.patch}"

v1, v2 = Version("1.2.3"), Version("1.2.4")
v1 < v2            # => True
sorted([v2, v1])   # => [v1, v2]
```

Rubyの `<=>`（1メソッドで大小・等値をまとめて返す）は、Pythonでは `__lt__`（小なり）と `__eq__`（等値）の2メソッドに分かれる。`between?` に完全対応するものはないが、`v1 <= x <= v3` のような連鎖比較で同等のことを表現できる。

### Enumerable

```ruby
class Playlist
  include Enumerable

  def initialize
    @songs = []
  end

  def add(song)
    @songs << song
  end

  # each を実装すると map, select, reduce などが使える
  def each(&block)
    @songs.each(&block)
  end
end

playlist = Playlist.new
playlist.add("Song 1")
playlist.add("Song 2")
playlist.add("Song 3")

playlist.map(&:upcase)           # => ["SONG 1", "SONG 2", "SONG 3"]
playlist.select { |s| s.include?("1") }  # => ["Song 1"]
playlist.count                   # => 3
playlist.first                   # => "Song 1"
```

**🐍 Python:** RubyのEnumerable（`each` を実装すると `map`/`select` 等が使える）に相当するのは「イテレータプロトコル」。`__iter__` を実装すれば `for`・内包表記・`map()`・`sorted()` などで反復できるようになる。

```python
class Playlist:
    def __init__(self):
        self._songs = []

    def add(self, song):
        self._songs.append(song)

    # __iter__ を実装すると for / 内包表記 / map などで反復できる
    def __iter__(self):
        return iter(self._songs)

playlist = Playlist()
playlist.add("Song 1")
playlist.add("Song 2")

[s.upper() for s in playlist]        # => ['SONG 1', 'SONG 2']
[s for s in playlist if "1" in s]    # => ['Song 1']
list(map(str.upper, playlist))       # => ['SONG 1', 'SONG 2']
```

重要な差: RubyのEnumerableは `map`/`select`/`reduce` 等のメソッドを**オブジェクト自身に生やす**のに対し、Pythonのイテラブルはそれらを**組み込み関数・内包表記・`itertools`** の側から使う。Rubyの `each` に `yield` で値を渡す実装は、Pythonでは `yield` を使ったジェネレータでも書ける（`__iter__` 自体をジェネレータにする）。

```python
def __iter__(self):
    for song in self._songs:
        yield song  # ジェネレータ: Rubyの each + yield に相当
```

### Forwardable

```ruby
require 'forwardable'

class Department
```


### Observable

```ruby
require 'observer'

class Stock
  include Observable

  attr_reader :symbol, :price

  def initialize(symbol, price)
    @symbol = symbol
    @price = price
  end

  def price=(new_price)
    if @price != new_price
      @price = new_price
      changed           # 変更をマーク
      notify_observers(self)  # オブザーバーに通知
    end
  end
end

class StockAlert
  def update(stock)
    puts "Alert: #{stock.symbol} is now #{stock.price}"
  end
end

stock = Stock.new("AAPL", 150)
alert = StockAlert.new
stock.add_observer(alert)

stock.price = 155  # => "Alert: AAPL is now 155"
stock.price = 160  # => "Alert: AAPL is now 160"
```

### デコレータパターン

```ruby
module Logging
  def call(*args)
    puts "Calling with args: #{args.inspect}"
    result = super
    puts "Result: #{result.inspect}"
    result
  end
end

module Timing
  def call(*args)
    start = Time.now
    result = super
    puts "Execution time: #{Time.now - start}s"
    result
  end
end

module Caching
  def call(*args)
    @cache ||= {}
    cache_key = args.hash

    return @cache[cache_key] if @cache.key?(cache_key)

    @cache[cache_key] = super
  end
end

class Calculator
  prepend Logging
  prepend Timing
  prepend Caching

  def call(a, b)
    sleep(0.1)  # 重い処理をシミュレート
    a + b
  end
end

calc = Calculator.new
calc.call(1, 2)  # ログ、タイミング、計算
calc.call(1, 2)  # キャッシュから取得
```

### 名前空間とファサード

```ruby
module PaymentGateway
  # 設定
  class Configuration
```

## 実践例: プラグインシステム

```ruby
module Pluggable
  def self.included(base)
    base.extend(ClassMethods)
  end

  module ClassMethods
    def plugins
      @plugins ||= []
    end

    def register_plugin(plugin_module)
      plugins << plugin_module
      include plugin_module
    end

    def plugin(name, &block)
      mod = Module.new
      mod.module_eval(&block)
      const_set("#{name.to_s.capitalize}Plugin", mod)
      register_plugin(mod)
    end
  end
end

class Application
  include Pluggable

  # プラグインを定義
  plugin :logging do
    def log(message)
      puts "[LOG] #{message}"
    end
  end

  plugin :metrics do
    def track(event)
      puts "[METRICS] #{event}"
    end
  end

  def run
    log("Application starting")
    track("app_start")
    puts "Running..."
    log("Application finished")
    track("app_finish")
  end
end

app = Application.new
app.run

# 外部プラグインの追加
module AuthPlugin
  def authenticate(user)
    puts "[AUTH] Authenticating #{user}"
    true
  end
end

Application.register_plugin(AuthPlugin)
app2 = Application.new
app2.authenticate("user123")
```
