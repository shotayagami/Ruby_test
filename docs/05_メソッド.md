## 1. メソッドの定義

### 基本的な定義

```ruby
# 基本形
def greet
  puts "Hello!"
end

# 引数あり
def greet(name)
  puts "Hello, #{name}!"
end

# 呼び出し
greet              # 引数なし
greet("Ruby")      # 引数あり
greet "Ruby"       # 括弧省略可能
```

### 1行メソッド（Ruby 3.0+）

```ruby
# エンドレスメソッド定義
def square(n) = n * n
def greet(name) = "Hello, #{name}!"

# 複雑な式
def status = @active ? "有効" : "無効"

# 条件付き
def adult?(age) = age >= 18
```

### メソッド定義の場所

```ruby
# トップレベル（Objectのプライベートメソッドになる）
def top_level_method
  "I'm everywhere"
end

# クラス内
class MyClass
  def instance_method
    "インスタンスメソッド"
  end

  def self.class_method
    "クラスメソッド"
  end
end

# モジュール内
module MyModule
  def module_method
    "モジュールメソッド"
  end
end
```

### メソッドの再定義

```ruby
def greet
  "Hello"
end

# 同名のメソッドを再定義（警告なし）
def greet
  "Hi"
end

greet  # => "Hi"

# undef でメソッド削除
undef greet
# greet  # NoMethodError!
```

## 2. 引数

### 必須引数

```ruby
def add(a, b)
  a + b
end

add(1, 2)      # => 3
# add(1)       # ArgumentError: wrong number of arguments
# add(1, 2, 3) # ArgumentError: wrong number of arguments
```

### デフォルト引数

```ruby
def greet(name = "Guest", greeting = "Hello")
  "#{greeting}, #{name}!"
end

greet                      # => "Hello, Guest!"
greet("Ruby")              # => "Hello, Ruby!"
greet("Ruby", "Hi")        # => "Hi, Ruby!"

# 式も使える
def log(message, time = Time.now)
  puts "[#{time}] #{message}"
end

# 他の引数を参照
def rectangle(width, height = width)
  width * height
end
rectangle(5)      # => 25（正方形）
rectangle(5, 3)   # => 15
```

### キーワード引数

```ruby
# 必須キーワード引数
def create_user(name:, email:)
  { name: name, email: email }
end
create_user(name: "太郎", email: "taro@example.com")

# デフォルト値付きキーワード引数
def create_user(name:, email:, role: "user")
  { name: name, email: email, role: role }
end
create_user(name: "太郎", email: "taro@example.com")
create_user(name: "太郎", email: "taro@example.com", role: "admin")

# 順序は自由
create_user(email: "taro@example.com", name: "太郎")
```

### 可変長引数（*args）

```ruby
# 残りの引数を配列として受け取る
def sum(*numbers)
  numbers.sum
end

sum(1, 2, 3)      # => 6
sum(1, 2, 3, 4, 5) # => 15
sum               # => 0（空配列）

# 他の引数と組み合わせ
def format(prefix, *items, suffix)
  "#{prefix}#{items.join(', ')}#{suffix}"
end
format("[", 1, 2, 3, "]")  # => "[1, 2, 3]"

# 配列の展開
args = [1, 2, 3]
sum(*args)        # => 6
```

### 可変長キーワード引数（**kwargs）

```ruby
# 残りのキーワード引数をハッシュとして受け取る
def configure(host:, **options)
  puts "Host: #{host}"
  puts "Options: #{options}"
end

configure(host: "localhost", port: 3000, timeout: 30)
# Host: localhost
# Options: {:port=>3000, :timeout=>30}

# ハッシュの展開
opts = { port: 3000, timeout: 30 }
configure(host: "localhost", **opts)
```

### ブロック引数（&block）

```ruby
# 明示的にブロックを受け取る
def with_logging(&block)
  puts "開始"
  result = block.call
  puts "終了"
  result
end

with_logging { puts "処理中"; 42 }

# ブロックを他のメソッドに渡す
def wrap(&block)
  puts "Before"
  yield
  puts "After"
end

# Procをブロックとして渡す
my_proc = Proc.new { puts "Hello" }
wrap(&my_proc)
```

### 引数の組み合わせ順序

```ruby
# 正しい順序
def method_name(
  required_arg,          # 1. 必須引数
  optional_arg = "default", # 2. デフォルト引数
  *rest_args,            # 3. 可変長引数
  keyword_arg:,          # 4. 必須キーワード引数
  optional_keyword: nil, # 5. デフォルトキーワード引数
  **rest_keywords,       # 6. 可変長キーワード引数
  &block                 # 7. ブロック引数
)
  # ...
end
```

### 引数のフォワーディング（Ruby 2.7+）

```ruby
# すべての引数を転送
def wrapper(...)
  original_method(...)
end

# 例
def log(message, level: :info, &block)
  puts "[#{level}] #{message}"
  block&.call
end

def with_timestamp(...)
  print "[#{Time.now}] "
  log(...)
end

with_timestamp("Hello", level: :debug) { puts "Done" }
```

### 無名キーワード引数の分離（Ruby 3.0+）

```ruby
# Ruby 3.0以降、ハッシュとキーワード引数は明確に分離
def foo(hash = {}, **kwargs)
  puts "hash: #{hash}"
  puts "kwargs: #{kwargs}"
end

# ハッシュとして渡す
foo({ a: 1 })           # hash: {:a=>1}, kwargs: {}

# キーワードとして渡す
foo(b: 2)               # hash: {}, kwargs: {:b=>2}

# 両方
foo({ a: 1 }, b: 2)     # hash: {:a=>1}, kwargs: {:b=>2}
```

## 3. 戻り値

### 暗黙の戻り値

```ruby
# 最後に評価された式が戻り値
def add(a, b)
  a + b  # これが戻り値
end

def status
  if active?
    "active"
  else
    "inactive"
  end  # if文の結果が戻り値
end
```

### 明示的なreturn

```ruby
# 早期リターン
def find_user(id)
  return nil if id.nil?
  return nil if id <= 0

  users.find { |u| u.id == id }
end

# ガード節
def process(data)
  return unless data
  return if data.empty?

  # メイン処理
  transform(data)
end
```

### 複数の値を返す

```ruby
# 配列として返す
def minmax(array)
  [array.min, array.max]
end

min, max = minmax([3, 1, 4, 1, 5])
puts min  # => 1
puts max  # => 5

# ハッシュとして返す
def analyze(text)
  {
    length: text.length,
    words: text.split.count,
    chars: text.chars.uniq.count
  }
end

result = analyze("hello world")
puts result[:words]  # => 2
```

### 戻り値の型

```ruby
# nilを返す可能性がある場合
def find(id)
  items.find { |item| item.id == id }
  # 見つからない場合はnilが返る
end

# selfを返してメソッドチェーン
class Builder
  def set_name(name)
    @name = name
    self  # selfを返す
  end

  def set_value(value)
    @value = value
    self
  end

  def build
    { name: @name, value: @value }
  end
end

result = Builder.new
  .set_name("test")
  .set_value(42)
  .build
```

## 4. ブロック

### ブロックの基本

```ruby
# do...end 形式（複数行）
[1, 2, 3].each do |n|
  puts n
end

# {...} 形式（1行）
[1, 2, 3].each { |n| puts n }

# 慣例：
# - 1行なら {}
# - 複数行なら do...end
# - 戻り値を使うなら {}
result = [1, 2, 3].map { |n| n * 2 }
```

### ブロック引数

```ruby
# 単一引数
[1, 2, 3].each { |n| puts n }

# 複数引数
{ a: 1, b: 2 }.each { |key, value| puts "#{key}: #{value}" }

# 引数の分解
[[1, 2], [3, 4]].each { |(a, b)| puts "#{a}, #{b}" }

# 番号付きブロックパラメータ（Ruby 2.7+）
[1, 2, 3].map { _1 * 2 }           # => [2, 4, 6]
{ a: 1 }.map { "#{_1}: #{_2}" }    # => ["a: 1"]

# アンダースコアで無視
{ a: 1, b: 2 }.each { |_, v| puts v }
```

### yield

```ruby
# ブロックを実行
def twice
  yield
  yield
end

twice { puts "Hello" }
# Hello
# Hello

# 引数を渡す
def transform(value)
  yield(value)
end

transform(5) { |n| n * 2 }  # => 10

# 複数の引数
def with_pair
  yield(1, 2)
end

with_pair { |a, b| a + b }  # => 3

# 戻り値を使う
def process
  result = yield(10)
  puts "Result: #{result}"
end

process { |n| n * 3 }  # Result: 30
```

### block_given?

```ruby
def greet(name)
  if block_given?
    yield(name)
  else
    "Hello, #{name}!"
  end
end

greet("Ruby")                    # => "Hello, Ruby!"
greet("Ruby") { |n| "Hi, #{n}!" } # => "Hi, Ruby!"
```

### ブロックをProcとして受け取る

```ruby
def process(&block)
  puts block.class      # => Proc
  block.call(10)
end

process { |n| n * 2 }   # => 20

# nilチェック
def maybe_process(&block)
  return "No block" unless block
  block.call
end

maybe_process              # => "No block"
maybe_process { "Hello" }  # => "Hello"
```

### ブロックのスコープ

```ruby
# ブロック外の変数を参照可能（クロージャ）
counter = 0
[1, 2, 3].each { |n| counter += n }
puts counter  # => 6

# ブロック内の変数は外からは見えない
[1, 2, 3].each { |n| x = n }
# puts x  # NameError!

# ブロックパラメータは外の同名変数を隠す
x = 10
[1, 2, 3].each { |x| puts x }  # 1, 2, 3（ブロック内のx）
puts x  # => 10（外のxは変わらない）

# セミコロンで明示的にローカル変数を宣言
total = 0
[1, 2, 3].each { |n; temp| temp = n * 2; total += temp }
```

## 5. Proc と Lambda

### Proc

```ruby
# 作成
my_proc = Proc.new { |x| x * 2 }
my_proc = proc { |x| x * 2 }  # 短縮形

# 呼び出し
my_proc.call(5)   # => 10
my_proc.(5)       # => 10
my_proc[5]        # => 10

# 引数のチェックが緩い
proc_example = Proc.new { |a, b| "a=#{a}, b=#{b}" }
proc_example.call(1)        # => "a=1, b="（不足はnil）
proc_example.call(1, 2, 3)  # => "a=1, b=2"（余分は無視）
```

### Lambda

```ruby
# 作成
my_lambda = lambda { |x| x * 2 }
my_lambda = ->(x) { x * 2 }  # アロー構文（推奨）

# 呼び出し
my_lambda.call(5)  # => 10
my_lambda.(5)      # => 10
my_lambda[5]       # => 10

# 引数のチェックが厳密
lambda_example = ->(a, b) { "a=#{a}, b=#{b}" }
# lambda_example.call(1)        # ArgumentError!
# lambda_example.call(1, 2, 3)  # ArgumentError!
lambda_example.call(1, 2)       # => "a=1, b=2"

# 複数行
my_lambda = ->(x, y) {
  sum = x + y
  sum * 2
}

# 複雑な引数
my_lambda = ->(a, b = 10, *rest, key:, **opts) {
  [a, b, rest, key, opts]
}
```

### Proc と Lambda の違い

```ruby
# 1. 引数チェック
proc_ex = Proc.new { |a, b| [a, b] }
lambda_ex = ->(a, b) { [a, b] }

proc_ex.call(1)        # => [1, nil]（OK）
# lambda_ex.call(1)    # ArgumentError!

# 2. returnの挙動
def proc_return
  my_proc = Proc.new { return "proc return" }
  my_proc.call
  "method return"  # 実行されない
end
proc_return  # => "proc return"（メソッドからreturn）

def lambda_return
  my_lambda = -> { return "lambda return" }
  result = my_lambda.call
  "method return: #{result}"  # 実行される
end
lambda_return  # => "method return: lambda return"

# 3. lambda? メソッド
proc_ex.lambda?    # => false
lambda_ex.lambda?  # => true
```

### ブロックとProcの変換

```ruby
# ブロック → Proc
def capture(&block)
  block  # Procオブジェクト
end

my_proc = capture { |x| x * 2 }
my_proc.call(5)  # => 10

# Proc → ブロック
my_proc = Proc.new { |x| puts x }
[1, 2, 3].each(&my_proc)

# シンボル → Proc
[1, 2, 3].map(&:to_s)  # => ["1", "2", "3"]

# 上記は以下と同等
[1, 2, 3].map { |x| x.to_s }

# Symbol#to_proc の仕組み
# :to_s.to_proc は ->(obj) { obj.to_s } と同等
```

### カリー化

```ruby
# curry でカリー化
add = ->(a, b, c) { a + b + c }
curried = add.curry

curried.call(1).call(2).call(3)  # => 6
curried.(1).(2).(3)              # => 6
curried[1][2][3]                 # => 6

# 部分適用
add_10 = curried.(10)
add_10.(20).(30)  # => 60

# 引数をまとめて渡すことも可能
curried.(1, 2).(3)  # => 6
curried.(1, 2, 3)   # => 6
```

### クロージャとしてのProc

```ruby
# 定義時の環境をキャプチャ
def make_counter
  count = 0
  -> { count += 1 }
end

counter = make_counter
counter.call  # => 1
counter.call  # => 2
counter.call  # => 3

# 別のカウンターは独立
counter2 = make_counter
counter2.call  # => 1

# 複数のProcで同じ変数を共有
def make_counter_pair
  count = 0
  increment = -> { count += 1 }
  decrement = -> { count -= 1 }
  get = -> { count }
  [increment, decrement, get]
end

inc, dec, get = make_counter_pair
inc.call  # => 1
inc.call  # => 2
dec.call  # => 1
get.call  # => 1
```

## 6. メソッドオブジェクト

### Method オブジェクト

```ruby
# メソッドをオブジェクトとして取得
def greet(name)
  "Hello, #{name}!"
end

method_obj = method(:greet)
method_obj.class        # => Method
method_obj.call("Ruby") # => "Hello, Ruby!"

# クラスのインスタンスメソッド
class Calculator
  def add(a, b)
    a + b
  end
end

calc = Calculator.new
add_method = calc.method(:add)
add_method.call(1, 2)  # => 3
```

### UnboundMethod

```ruby
# インスタンスにバインドされていないメソッド
class MyClass
  def hello
    "Hello from #{self.class}"
  end
end

unbound = MyClass.instance_method(:hello)
unbound.class  # => UnboundMethod

# バインドして呼び出し
obj = MyClass.new
bound = unbound.bind(obj)
bound.call  # => "Hello from MyClass"
```

### メソッドの情報取得

```ruby
def example(a, b = 10, *rest, key:, **opts, &block)
  # ...
end

m = method(:example)

m.name           # => :example
m.owner          # => Object（定義されたクラス/モジュール）
m.receiver       # => main（レシーバ）
m.arity          # => -2（必須引数の数、可変は負数）
m.parameters     # => [[:req, :a], [:opt, :b], [:rest, :rest],
                 #     [:keyreq, :key], [:keyrest, :opts], [:block, :block]]
m.source_location # => ["file.rb", 1]（ファイルと行番号）

# Ruby 2.7+
m.super_method   # 親クラスの同名メソッド
```

### メソッドをProcに変換

```ruby
def double(n)
  n * 2
end

# to_proc で変換
double_proc = method(:double).to_proc

[1, 2, 3].map(&method(:double))  # => [2, 4, 6]

# クラスメソッドでも
[1, 2, 3].map(&Math.method(:sqrt))  # => [1.0, 1.414..., 1.732...]
```

## 7. メソッドの可視性

### public / private / protected

```ruby
class MyClass
  # デフォルトは public
  def public_method
    "誰でも呼べる"
  end

  private

  # private以降のメソッドはprivate
  def private_method
    "クラス内からのみ呼べる"
  end

  def another_private
    "これもprivate"
  end

  protected

  def protected_method
    "同じクラスまたはサブクラスから呼べる"
  end

  public

  # publicに戻す
  def another_public
    "再びpublic"
  end
end
```

### 明示的な指定

```ruby
class MyClass
  def method1; end
  def method2; end
  def method3; end

  # まとめて指定
  private :method1, :method2
  protected :method3

  # 定義と同時に指定
  private def secret
    "秘密"
  end
end
```

### private と protected の違い

```ruby
class Parent
  def call_private
    private_method  # OK: 同じクラス内
  end

  def call_protected(other)
    other.protected_method  # OK: 同じクラスのインスタンス
  end

  private

  def private_method
    "private"
  end

  protected

  def protected_method
    "protected"
  end
end

class Child < Parent
  def call_parent_private
    private_method  # OK: 継承したprivateメソッド
  end

  def call_other_protected(other)
    other.protected_method  # OK: 同じ継承階層
  end
end

p = Parent.new
# p.private_method    # NoMethodError!
# p.protected_method  # NoMethodError!

p.call_private  # => "private"
p.call_protected(Parent.new)  # => "protected"
```

### Ruby 2.7+ のprivateメソッド呼び出し

```ruby
class MyClass
  def public_method
    self.private_method  # Ruby 2.7以降はOK
  end

  private

  def private_method
    "private"
  end
end
```

### クラスメソッドの可視性

```ruby
class MyClass
  # クラスメソッドのprivate
  class << self
    private

    def private_class_method
      "プライベートクラスメソッド"
    end
  end

  # または
  def self.another_private
    "プライベート"
  end
  private_class_method :another_private
end
```

## 8. メソッドの命名規則

### 慣習的な接尾辞

```ruby
# ? で終わる: 真偽値を返す（述語メソッド）
def empty?
  @items.empty?
end

def valid?
  @errors.none?
end

def include?(item)
  @items.include?(item)
end

# ! で終わる: 破壊的メソッドまたは例外を発生
class Array
  def sort    # 非破壊的（新しい配列を返す）
  end

  def sort!   # 破壊的（自身を変更）
  end
end

class User
  def save    # 失敗時は false を返す
  end

  def save!   # 失敗時は例外を発生
  end
end

# = で終わる: セッターメソッド
def name=(value)
  @name = value
end

# 呼び出し
obj.name = "太郎"  # obj.name=("太郎") と同じ
```

### 演算子風メソッド

```ruby
class Vector
  def initialize(x, y)
    @x, @y = x, y
  end

  # 配列風アクセス
  def [](index)
    index == 0 ? @x : @y
  end

  def []=(index, value)
    index == 0 ? @x = value : @y = value
  end

  # 算術演算子
  def +(other)
    Vector.new(@x + other.x, @y + other.y)
  end

  def -@  # 単項マイナス
    Vector.new(-@x, -@y)
  end

  # 比較
  def ==(other)
    @x == other.x && @y == other.y
  end

  def <=>(other)
    magnitude <=> other.magnitude
  end

  protected

  attr_reader :x, :y

  private

  def magnitude
    Math.sqrt(@x**2 + @y**2)
  end
end

v1 = Vector.new(1, 2)
v2 = Vector.new(3, 4)
v1 + v2    # => Vector(4, 6)
v1[0]      # => 1
-v1        # => Vector(-1, -2)
```

## 9. 演算子メソッド

### 定義可能な演算子

```ruby
# 算術演算子
+  -  *  /  %  **

# 比較演算子
==  !=  <  >  <=  >=  <=>

# ビット演算子
&  |  ^  ~  <<  >>

# 論理演算子（定義不可）
# &&  ||  !  not  and  or

# 配列アクセス
[]  []=

# 単項演算子
+@  -@  ~

# その他
=~  !~  ===
```

### 実装例

```ruby
class Money
  attr_reader :amount, :currency

  def initialize(amount, currency = "JPY")
    @amount = amount
    @currency = currency
  end

  def +(other)
    raise "Currency mismatch" if currency != other.currency
    Money.new(amount + other.amount, currency)
  end

  def -(other)
    raise "Currency mismatch" if currency != other.currency
    Money.new(amount - other.amount, currency)
  end

  def *(multiplier)
    Money.new(amount * multiplier, currency)
  end

  def ==(other)
    amount == other.amount && currency == other.currency
  end

  def <=>(other)
    return nil if currency != other.currency
    amount <=> other.amount
  end

  def -@
    Money.new(-amount, currency)
  end

  def to_s
    "#{amount} #{currency}"
  end
end

m1 = Money.new(1000)
m2 = Money.new(500)
puts m1 + m2   # => 1500 JPY
puts m1 - m2   # => 500 JPY
puts m1 * 2    # => 2000 JPY
puts m1 == m2  # => false
puts -m1       # => -1000 JPY
```

### Comparableモジュール

```ruby
class Person
  include Comparable

  attr_reader :name, :age

  def initialize(name, age)
    @name = name
    @age = age
  end

  # <=> を定義すると、<, <=, ==, >=, > が使える
  def <=>(other)
    age <=> other.age
  end
end

p1 = Person.new("太郎", 25)
p2 = Person.new("次郎", 30)

p1 < p2   # => true
p1 >= p2  # => false
[p2, p1].sort  # => [p1, p2]
```

### define_method

```ruby
class MyClass

  define_method(:greet) do |name|
    "Hello, #{name}!"
  end

  # ループで複数のメソッドを定義

    end
  end
end
```

### method_missing

```ruby
class Flexible
  def method_missing(name, *args, &block)
    if name.to_s.start_with?("find_by_")
      attribute = name.to_s.sub("find_by_", "")
      puts "Finding by #{attribute}: #{args.first}"
    else
      super  # 親クラスに委譲（NoMethodErrorを発生）
    end
  end

  # respond_to_missing? も定義すべき
  def respond_to_missing?(name, include_private = false)
    name.to_s.start_with?("find_by_") || super
  end
end

f = Flexible.new
f.find_by_name("太郎")   # => Finding by name: 太郎
f.find_by_email("a@b")   # => Finding by email: a@b
f.respond_to?(:find_by_id)  # => true
```

### send / public_send

```ruby
class Calculator
  def add(a, b)
    a + b
  end

  private

  def secret
    "秘密"
  end
end

calc = Calculator.new

# send: privateメソッドも呼べる
calc.send(:add, 1, 2)    # => 3
calc.send(:secret)       # => "秘密"

# public_send: publicメソッドのみ
calc.public_send(:add, 1, 2)  # => 3
# calc.public_send(:secret)   # NoMethodError!

# 動的なメソッド呼び出し
method_name = :add
calc.send(method_name, 10, 20)  # => 30
```

### alias_method

```ruby
class MyClass
  def original
    "元のメソッド"
  end

  # 別名を作成
  alias_method :aliased, :original

  # 元のメソッドを保存してオーバーライド
  alias_method :original_without_logging, :original

  def original
    puts "Logging..."
    original_without_logging
  end
end

obj = MyClass.new
obj.original  # Logging... → "元のメソッド"
obj.aliased   # => "元のメソッド"
```
