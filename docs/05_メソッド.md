## 1. メソッドの定義

### 基本的な定義

```ruby
# 基本形
def greet
  puts "Hello!"
end

# 引数あり
def greet(name)
  puts "Hello, #{name}!"
end

# 呼び出し
greet              # 引数なし
greet("Ruby")      # 引数あり
greet "Ruby"       # 括弧省略可能
```

**🐍 Python:**

```python
# def ... : （end不要、コロンとインデントでブロックを表す）
def greet():
    print("Hello!")

def greet(name):
    print(f"Hello, {name}!")

# 呼び出し
greet()            # 引数なし
greet("Ruby")      # 引数あり（括弧は省略できない）
```

Pythonは呼び出し時の括弧を省略できません（`greet` だけでは関数オブジェクトを指す）。またブロックの終わりは `end` ではなくインデントで表します。

### 1行メソッド（Ruby 3.0+）

```ruby
# エンドレスメソッド定義
def square(n) = n * n
def greet(name) = "Hello, #{name}!"

# 複雑な式
def status = @active ? "有効" : "無効"

# 条件付き
def adult?(age) = age >= 18
```

**🐍 Python:**

```python
# Pythonにエンドレスメソッド構文はないが、lambda が近い
square = lambda n: n * n
greet = lambda name: f"Hello, {name}!"

# 通常は def で書く（1行に押し込むより推奨）
def square(n): return n * n
```

Python版の `def ... = ...` に相当する専用構文はありません。単純な式なら `lambda`、それ以外は通常の `def` を使います。

### メソッド定義の場所

```ruby
# トップレベル（Objectのプライベートメソッドになる）
def top_level_method
  "I'm everywhere"
end

# クラス内
class MyClass
  def instance_method
    "インスタンスメソッド"
  end

  def self.class_method
    "クラスメソッド"
  end
end

# モジュール内
module MyModule
  def module_method
    "モジュールメソッド"
  end
end
```

**🐍 Python:**

```python
# トップレベル関数
def top_level_function():
    return "I'm everywhere"

class MyClass:
    def instance_method(self):        # 第1引数に self が必須
        return "インスタンスメソッド"

    @classmethod
    def class_method(cls):            # デコレータで明示
        return "クラスメソッド"
```

Pythonのインスタンスメソッドは第1引数 `self` を明示的に受け取ります。クラスメソッドは `def self.x` ではなく `@classmethod` デコレータで表現します。

### メソッドの再定義

```ruby
def greet
  "Hello"
end

# 同名のメソッドを再定義（警告なし）
def greet
  "Hi"
end

greet  # => "Hi"

# undef でメソッド削除
undef greet
# greet  # NoMethodError!
```

## 2. 引数

### 必須引数

```ruby
def add(a, b)
  a + b
end

add(1, 2)      # => 3
# add(1)       # ArgumentError: wrong number of arguments
# add(1, 2, 3) # ArgumentError: wrong number of arguments
```

### デフォルト引数

```ruby
def greet(name = "Guest", greeting = "Hello")
  "#{greeting}, #{name}!"
end

greet                      # => "Hello, Guest!"
greet("Ruby")              # => "Hello, Ruby!"
greet("Ruby", "Hi")        # => "Hi, Ruby!"

# 式も使える
def log(message, time = Time.now)
  puts "[#{time}] #{message}"
end

# 他の引数を参照
def rectangle(width, height = width)
  width * height
end
rectangle(5)      # => 25（正方形）
rectangle(5, 3)   # => 15
```

**🐍 Python:**

```python
def greet(name="Guest", greeting="Hello"):
    return f"{greeting}, {name}!"

greet()                  # => "Hello, Guest!"
greet("Ruby", "Hi")      # => "Hi, Ruby!"
```

デフォルト値の書き方はほぼ同じです。ただしPythonのデフォルト値は定義時に一度だけ評価されるため、`def log(message, time=datetime.now())` は起動時刻で固定されてしまいます。可変オブジェクトも同様で、慣習として `def f(x=None)` と書き本体で `x = x or ...` とします。また他の引数を参照する `height = width` のような書き方はできません。

### キーワード引数

```ruby
# 必須キーワード引数
def create_user(name:, email:)
  { name: name, email: email }
end
create_user(name: "太郎", email: "taro@example.com")

# デフォルト値付きキーワード引数
def create_user(name:, email:, role: "user")
  { name: name, email: email, role: role }
end
create_user(name: "太郎", email: "taro@example.com")
create_user(name: "太郎", email: "taro@example.com", role: "admin")

# 順序は自由
create_user(email: "taro@example.com", name: "太郎")
```

**🐍 Python:**

```python
def create_user(name, email, role="user"):
    return {"name": name, "email": email, "role": role}

# キーワードで渡せば順序は自由
create_user(email="taro@example.com", name="太郎")

# * 以降をキーワード専用引数（必須キーワード）にできる
def create_user2(*, name, email, role="user"):
    return {"name": name, "email": email, "role": role}
create_user2(name="太郎", email="taro@example.com")  # 位置指定は不可
```

Rubyの `name:`（必須キーワード引数）に相当するのはPythonの「キーワード専用引数」です。引数リストに `*,` を置くと、それ以降は必ずキーワードで渡す必要があります。通常のPython引数は位置でもキーワードでも渡せる点がRubyと異なります。

### 可変長引数（*args）

```ruby
# 残りの引数を配列として受け取る
def sum(*numbers)
  numbers.sum
end

sum(1, 2, 3)      # => 6
sum(1, 2, 3, 4, 5) # => 15
sum               # => 0（空配列）

# 他の引数と組み合わせ
def format(prefix, *items, suffix)
  "#{prefix}#{items.join(', ')}#{suffix}"
end
format("[", 1, 2, 3, "]")  # => "[1, 2, 3]"

# 配列の展開
args = [1, 2, 3]
sum(*args)        # => 6
```

**🐍 Python:**

```python
def total(*numbers):        # *args と同じ仕組み
    return sum(numbers)     # numbers は tuple

total(1, 2, 3)     # => 6
total()            # => 0

# リストの展開も * で行う
args = [1, 2, 3]
total(*args)       # => 6
```

`*numbers` はRubyとほぼ同じ働きで、余った位置引数をまとめて受け取ります（Rubyでは配列、Pythonではタプル）。呼び出し側の `*args` による展開も共通です。

### 可変長キーワード引数（**kwargs）

```ruby
# 残りのキーワード引数をハッシュとして受け取る
def configure(host:, **options)
  puts "Host: #{host}"
  puts "Options: #{options}"
end

configure(host: "localhost", port: 3000, timeout: 30)
# Host: localhost
# Options: {:port=>3000, :timeout=>30}

# ハッシュの展開
opts = { port: 3000, timeout: 30 }
configure(host: "localhost", **opts)
```

**🐍 Python:**

```python
def configure(host, **options):     # **kwargs と同じ仕組み
    print(f"Host: {host}")
    print(f"Options: {options}")    # options は dict

configure(host="localhost", port=3000, timeout=30)

# 辞書の展開も ** で行う
opts = {"port": 3000, "timeout": 30}
configure(host="localhost", **opts)
```

`**options` はRubyの `**options` とほぼ同じで、残りのキーワード引数を辞書（Ruby ではハッシュ）として受け取ります。

### ブロック引数（&block）

```ruby
# 明示的にブロックを受け取る
def with_logging(&block)
  puts "開始"
  result = block.call
  puts "終了"
  result
end

with_logging { puts "処理中"; 42 }

# ブロックを他のメソッドに渡す
def wrap(&block)
  puts "Before"
  yield
  puts "After"
end

# Procをブロックとして渡す
my_proc = Proc.new { puts "Hello" }
wrap(&my_proc)
```

**🐍 Python:**

```python
# 方法1: 関数を引数として渡す（高階関数）
def with_logging(func):
    print("開始")
    result = func()
    print("終了")
    return result

with_logging(lambda: print("処理中"))

# 方法2: 前後を囲む用途はコンテキストマネージャ（with）が自然
from contextlib import contextmanager

@contextmanager
def logging_block():
    print("開始")
    yield
    print("終了")

with logging_block():
    print("処理中")

# 方法3: 関数呼び出しそのものを囲むならデコレータ
def with_logging_deco(func):
    def wrapper(*args, **kwargs):
        print("開始")
        result = func(*args, **kwargs)
        print("終了")
        return result
    return wrapper
```

`&block`（明示的にブロックを受け取る）に相当する専用構文はPythonにありません。呼び出し可能オブジェクトを普通の引数として渡すか、`with_logging` のように「前後を囲む」用途ではコンテキストマネージャ(`with`)やデコレータで表現します。

### 引数の組み合わせ順序

```ruby
# 正しい順序
def method_name(
  required_arg,          # 1. 必須引数
  optional_arg = "default", # 2. デフォルト引数
  *rest_args,            # 3. 可変長引数
  keyword_arg:,          # 4. 必須キーワード引数
  optional_keyword: nil, # 5. デフォルトキーワード引数
  **rest_keywords,       # 6. 可変長キーワード引数
  &block                 # 7. ブロック引数
)
  # ...
end
```

**🐍 Python:**

```python
def method_name(
    required_arg,              # 1. 必須（位置）引数
    optional_arg="default",    # 2. デフォルト引数
    *rest_args,                # 3. 可変長引数
    keyword_arg,               # 4. * の後なのでキーワード専用（必須）
    optional_keyword=None,     # 5. デフォルト付きキーワード専用
    **rest_keywords,           # 6. 可変長キーワード引数
):                             # ブロック引数に相当するものはない
    ...
```

Pythonでは `*rest_args` 以降の引数が自動的にキーワード専用になります。ブロック引数(`&block`)に対応する要素はなく、必要なら関数を通常の引数として受け取ります。

### 引数のフォワーディング（Ruby 2.7+）

```ruby
# すべての引数を転送
def wrapper(...)
  original_method(...)
end

# 例
def log(message, level: :info, &block)
  puts "[#{level}] #{message}"
  block&.call
end

def with_timestamp(...)
  print "[#{Time.now}] "
  log(...)
end

with_timestamp("Hello", level: :debug) { puts "Done" }
```

**🐍 Python:**

```python
# 全引数の転送は *args, **kwargs で行う
def wrapper(*args, **kwargs):
    return original_method(*args, **kwargs)
```

Rubyの `...` に相当するのは `*args, **kwargs` の組み合わせです（ブロックは含まれない）。

### 無名キーワード引数の分離（Ruby 3.0+）

```ruby
# Ruby 3.0以降、ハッシュとキーワード引数は明確に分離
def foo(hash = {}, **kwargs)
  puts "hash: #{hash}"
  puts "kwargs: #{kwargs}"
end

# ハッシュとして渡す
foo({ a: 1 })           # hash: {:a=>1}, kwargs: {}

# キーワードとして渡す
foo(b: 2)               # hash: {}, kwargs: {:b=>2}

# 両方
foo({ a: 1 }, b: 2)     # hash: {:a=>1}, kwargs: {:b=>2}
```

## 3. 戻り値

### 暗黙の戻り値

```ruby
# 最後に評価された式が戻り値
def add(a, b)
  a + b  # これが戻り値
end

def status
  if active?
    "active"
  else
    "inactive"
  end  # if文の結果が戻り値
end
```

**🐍 Python:**

```python
def add(a, b):
    return a + b        # 必ず return が必要

def status():
    if active():
        return "active"
    return "inactive"   # return を書かないと None が返る
```

Pythonに暗黙のreturnはありません。最後の式の値を返したい場合も必ず `return` を書きます。`return` が無いと（あるいは値なしの `return` だと）自動的に `None` が返ります。

### 明示的なreturn

```ruby
# 早期リターン
def find_user(id)
  return nil if id.nil?
  return nil if id <= 0

  users.find { |u| u.id == id }
end

# ガード節
def process(data)
  return unless data
  return if data.empty?

  # メイン処理
  transform(data)
end
```

### 複数の値を返す

```ruby
# 配列として返す
def minmax(array)
  [array.min, array.max]
end

min, max = minmax([3, 1, 4, 1, 5])
puts min  # => 1
puts max  # => 5

# ハッシュとして返す
def analyze(text)
  {
    length: text.length,
    words: text.split.count,
    chars: text.chars.uniq.count
  }
end

result = analyze("hello world")
puts result[:words]  # => 2
```

**🐍 Python:**

```python
# タプルとして返し、分割代入で受け取る
def minmax(array):
    return min(array), max(array)   # (min, max) のタプル

lo, hi = minmax([3, 1, 4, 1, 5])    # lo=1, hi=5

# 辞書として返す
def analyze(text):
    return {"length": len(text), "words": len(text.split())}

result = analyze("hello world")
print(result["words"])   # => 2
```

`[array.min, array.max]` のように複数値を返す場合、Pythonでは慣用的にタプルを返し `lo, hi = ...` で受け取ります。

### 戻り値の型

```ruby
# nilを返す可能性がある場合
def find(id)
  items.find { |item| item.id == id }
  # 見つからない場合はnilが返る
end

# selfを返してメソッドチェーン
class Builder
  def set_name(name)
    @name = name
    self  # selfを返す
  end

  def set_value(value)
    @value = value
    self
  end

  def build
    { name: @name, value: @value }
  end
end

result = Builder.new
  .set_name("test")
  .set_value(42)
  .build
```

## 4. ブロック

### ブロックの基本

```ruby
# do...end 形式（複数行）
[1, 2, 3].each do |n|
  puts n
end

# {...} 形式（1行）
[1, 2, 3].each { |n| puts n }

# 慣例：
# - 1行なら {}
# - 複数行なら do...end
# - 戻り値を使うなら {}
result = [1, 2, 3].map { |n| n * 2 }
```

**🐍 Python:**

```python
# Pythonにブロック構文はない。用途ごとに手段が分かれる
for n in [1, 2, 3]:          # each 相当は普通の for ループ
    print(n)

# map 相当は内包表記（Pythonではこちらが主流）
result = [n * 2 for n in [1, 2, 3]]   # => [2, 4, 6]
result = list(map(lambda n: n * 2, [1, 2, 3]))  # 関数を渡す書き方
```

ブロックはRuby特有の機能で、Pythonに直接対応する構文はありません。代替手段は用途で使い分けます。①要素ごとの処理を渡す → 高階関数（`map`/`filter` に関数を渡す）や内包表記、②遅延的に値を生成する → ジェネレータ（`yield`）、③前後処理で囲む → コンテキストマネージャ（`with`）。以下の各節で対比します。

### ブロック引数

```ruby
# 単一引数
[1, 2, 3].each { |n| puts n }

# 複数引数
{ a: 1, b: 2 }.each { |key, value| puts "#{key}: #{value}" }

# 引数の分解
[[1, 2], [3, 4]].each { |(a, b)| puts "#{a}, #{b}" }

# 番号付きブロックパラメータ（Ruby 2.7+）
[1, 2, 3].map { _1 * 2 }           # => [2, 4, 6]
{ a: 1 }.map { "#{_1}: #{_2}" }    # => ["a: 1"]

# アンダースコアで無視
{ a: 1, b: 2 }.each { |_, v| puts v }
```

### yield

```ruby
# ブロックを実行
def twice
  yield
  yield
end

twice { puts "Hello" }
# Hello
# Hello

# 引数を渡す
def transform(value)
  yield(value)
end

transform(5) { |n| n * 2 }  # => 10

# 複数の引数
def with_pair
  yield(1, 2)
end

with_pair { |a, b| a + b }  # => 3

# 戻り値を使う
def process
  result = yield(10)
  puts "Result: #{result}"
end

process { |n| n * 3 }  # Result: 30
```

**🐍 Python:**

```python
# Rubyのyield（ブロックを呼び出す）に相当するのは、渡された関数の呼び出し
def twice(func):
    func()
    func()

twice(lambda: print("Hello"))

def transform(value, func):
    return func(value)

transform(5, lambda n: n * 2)   # => 10
```

注意: Pythonにも `yield` がありますが意味が全く異なります。Rubyの `yield` は「渡されたブロックをその場で実行」しますが、Pythonの `yield` は「ジェネレータとして値を1つずつ生成」する機能です（後述）。Rubyの `yield` に対応するのは、引数で受け取った関数を `func()` と呼び出すことです。

### block_given?

```ruby
def greet(name)
  if block_given?
    yield(name)
  else
    "Hello, #{name}!"
  end
end

greet("Ruby")                    # => "Hello, Ruby!"
greet("Ruby") { |n| "Hi, #{n}!" } # => "Hi, Ruby!"
```

### ブロックをProcとして受け取る

```ruby
def process(&block)
  puts block.class      # => Proc
  block.call(10)
end

process { |n| n * 2 }   # => 20

# nilチェック
def maybe_process(&block)
  return "No block" unless block
  block.call
end

maybe_process              # => "No block"
maybe_process { "Hello" }  # => "Hello"
```

### ブロックのスコープ

```ruby
# ブロック外の変数を参照可能（クロージャ）
counter = 0
[1, 2, 3].each { |n| counter += n }
puts counter  # => 6

# ブロック内の変数は外からは見えない
[1, 2, 3].each { |n| x = n }
# puts x  # NameError!

# ブロックパラメータは外の同名変数を隠す
x = 10
[1, 2, 3].each { |x| puts x }  # 1, 2, 3（ブロック内のx）
puts x  # => 10（外のxは変わらない）

# セミコロンで明示的にローカル変数を宣言
total = 0
[1, 2, 3].each { |n; temp| temp = n * 2; total += temp }
```

## 5. Proc と Lambda

### Proc

```ruby
# 作成
my_proc = Proc.new { |x| x * 2 }
my_proc = proc { |x| x * 2 }  # 短縮形

# 呼び出し
my_proc.call(5)   # => 10
my_proc.(5)       # => 10
my_proc[5]        # => 10

# 引数のチェックが緩い
proc_example = Proc.new { |a, b| "a=#{a}, b=#{b}" }
proc_example.call(1)        # => "a=1, b="（不足はnil）
proc_example.call(1, 2, 3)  # => "a=1, b=2"（余分は無視）
```

### Lambda

```ruby
# 作成
my_lambda = lambda { |x| x * 2 }
my_lambda = ->(x) { x * 2 }  # アロー構文（推奨）

# 呼び出し
my_lambda.call(5)  # => 10
my_lambda.(5)      # => 10
my_lambda[5]       # => 10

# 引数のチェックが厳密
lambda_example = ->(a, b) { "a=#{a}, b=#{b}" }
# lambda_example.call(1)        # ArgumentError!
# lambda_example.call(1, 2, 3)  # ArgumentError!
lambda_example.call(1, 2)       # => "a=1, b=2"

# 複数行
my_lambda = ->(x, y) {
  sum = x + y
  sum * 2
}

# 複雑な引数
my_lambda = ->(a, b = 10, *rest, key:, **opts) {
  [a, b, rest, key, opts]
}
```

**🐍 Python:**

```python
# Pythonの lambda は「式ひとつ」だけ（複数行は書けない）
double = lambda x: x * 2
double(5)          # => 10（呼び出しは通常の関数と同じ）

# 引数チェックはRubyのlambdaと同様に厳密
f = lambda a, b: (a, b)
# f(1)             # TypeError: 引数が足りない

# 複数行が必要なら def で名前付き関数を定義する
def my_func(x, y):
    s = x + y
    return s * 2
```

Pythonの `lambda` はRubyの `lambda`/`->` に近く、引数の数が厳密にチェックされます。ただしPythonの `lambda` は式を1つしか書けないため、複数行の処理は `def` で関数を定義します（Rubyのように複数行ラムダは書けません）。RubyのProc（引数チェックが緩い）に相当する仕組みはありません。呼び出しは `.call` ではなく `f(5)` の形式です。

### Proc と Lambda の違い

```ruby
# 1. 引数チェック
proc_ex = Proc.new { |a, b| [a, b] }
lambda_ex = ->(a, b) { [a, b] }

proc_ex.call(1)        # => [1, nil]（OK）
# lambda_ex.call(1)    # ArgumentError!

# 2. returnの挙動
def proc_return
  my_proc = Proc.new { return "proc return" }
  my_proc.call
  "method return"  # 実行されない
end
proc_return  # => "proc return"（メソッドからreturn）

def lambda_return
  my_lambda = -> { return "lambda return" }
  result = my_lambda.call
  "method return: #{result}"  # 実行される
end

# 3. lambda? メソッド

lambda_ex.lambda?  # => true
```

### ブロックとProcの変換

```ruby
# ブロック → Proc
def capture(&block)
  block  # Procオブジェクト
end

my_proc.call(5)  # => 10

# Proc → ブロック
my_proc = Proc.new { |x| puts x }

# シンボル → Proc

# 上記は以下と同等
[1, 2, 3].map { |x| x.to_s }

# Symbol#to_proc の仕組み
# :to_s.to_proc は ->(obj) { obj.to_s } と同等
```

**🐍 Python:**

```python
# &proc で「Procをブロック化」する構文はない。関数をそのまま渡す
```


### クロージャとしてのProc

```ruby
# 定義時の環境をキャプチャ

```

## 7. メソッドの可視性

### public / private / protected

```ruby
class MyClass
  # デフォルトは public
  def public_method
    "誰でも呼べる"
  end

  private

  # private以降のメソッドはprivate
  def private_method
    "クラス内からのみ呼べる"
  end

  def another_private
    "これもprivate"
  end

  protected

  def protected_method
    "同じクラスまたはサブクラスから呼べる"
  end

  public

  # publicに戻す
  def another_public
    "再びpublic"
  end
end
```

**🐍 Python:**

```python
class MyClass:
    def public_method(self):        # 公開（すべてのメソッドが実質public）
        return "誰でも呼べる"

    def _internal(self):            # 先頭 _ は「内部用」の慣習（強制力はない）
        return "非公開のつもり"

    def __private(self):            # 先頭 __ は名前修飾で外から呼びにくくなる
        return "name mangling"
```

Pythonには `private`/`protected` のような言語レベルのアクセス制御がありません。代わりに命名の慣習を使います。先頭アンダースコア1つ `_name` は「内部用（触らないで）」という約束ごと（強制されない）、2つ `__name` はクラス外から `_ClassName__name` に名前修飾されて偶発的アクセスを防ぎます。Rubyの `protected` に直接対応する仕組みはありません。

### 明示的な指定

```ruby
class MyClass
  def method1; end
  def method2; end
  def method3; end

  # まとめて指定
  private :method1, :method2
  protected :method3

  # 定義と同時に指定
  private def secret
    "秘密"
  end
end
```

### private と protected の違い

```ruby
class Parent
  def call_private
    private_method  # OK: 同じクラス内
  end

  def call_protected(other)
    other.protected_method  # OK: 同じクラスのインスタンス
  end

  private

  def private_method
    "private"
  end

  protected

  def protected_method
    "protected"
  end
end

class Child < Parent
  def call_parent_private
    private_method  # OK: 継承したprivateメソッド
  end

  def call_other_protected(other)
    other.protected_method  # OK: 同じ継承階層
  end
end

# p.private_method    # NoMethodError!
# p.protected_method  # NoMethodError!

p.call_private  # => "private"
p.call_protected(Parent.new)  # => "protected"
```

## 8. メソッドの命名規則

### 慣習的な接尾辞

```ruby
# ? で終わる: 真偽値を返す（述語メソッド）
def empty?
  @items.empty?
end

def valid?
  @errors.none?
end

def include?(item)
  @items.include?(item)
end

# ! で終わる: 破壊的メソッドまたは例外を発生
class Array
  def sort    # 非破壊的（新しい配列を返す）
  end

  def sort!   # 破壊的（自身を変更）
  end
end

class User
  def save    # 失敗時は false を返す
  end

  def save!   # 失敗時は例外を発生
  end
end

# = で終わる: セッターメソッド
def name=(value)
  @name = value
end

# 呼び出し
obj.name = "太郎"  # obj.name=("太郎") と同じ
```

**🐍 Python:**

```python
# ? （述語）に相当する記号はない。慣習として is_ / has_ を前置する
def is_empty(self):   return len(self.items) == 0
def has_errors(self): return len(self.errors) > 0

# ! （破壊的）に相当する命名慣習はない。破壊的かどうかは個々のAPI次第
nums = [3, 1, 2]
nums.sort()           # 破壊的（自身を並べ替え、返り値は None）
sorted(nums)          # 非破壊的（新しいリストを返す）

# name= セッターは @property / @x.setter で表現する
class User:
    @property
    def name(self):
        return self._name

    @name.setter
    def name(self, value):
        self._name = value

u = User()
u.name = "太郎"        # 見た目は属性代入だが setter が呼ばれる
```

Pythonのメソッド名に `?` や `!` は使えません（識別子に許されない）。述語は `is_empty` / `has_errors` のように語頭で表し、破壊的かどうかを示す命名慣習はありません（例: `list.sort()` は破壊的、組み込み関数 `sorted()` は非破壊）。Rubyの `name=` セッターに相当するのは `@property` と `@name.setter` の組み合わせです。

### 演算子風メソッド

```ruby
class Vector
  def initialize(x, y)
    @x, @y = x, y
  end

  # 配列風アクセス
  def [](index)
    index == 0 ? @x : @y
  end

  def []=(index, value)
    index == 0 ? @x = value : @y = value
  end

  # 算術演算子
  def +(other)
    Vector.new(@x + other.x, @y + other.y)
  end

  def -@  # 単項マイナス
    Vector.new(-@x, -@y)
  end

  # 比較
  def ==(other)
    @x == other.x && @y == other.y
  end

  def <=>(other)
    magnitude <=> other.magnitude
  end

  protected

  attr_reader :x, :y

  private

  def magnitude
    Math.sqrt(@x**2 + @y**2)
  end
end

v1 = Vector.new(1, 2)
v2 = Vector.new(3, 4)
v1 + v2    # => Vector(4, 6)
v1[0]      # => 1
-v1        # => Vector(-1, -2)
```

## 9. 演算子メソッド

### 定義可能な演算子

```ruby
# 算術演算子
+  -  *  /  %  **

# 比較演算子
==  !=  <  >  <=  >=  <=>

# ビット演算子
&  |  ^  ~  <<  >>

# 論理演算子（定義不可）
# &&  ||  !  not  and  or

# 配列アクセス
[]  []=

# 単項演算子
+@  -@  ~

# その他
=~  !~  ===
```

### 実装例

```ruby
class Money
  attr_reader :amount, :currency

  def initialize(amount, currency = "JPY")
    @amount = amount
    @currency = currency
  end

  def +(other)
    raise "Currency mismatch" if currency != other.currency
    Money.new(amount + other.amount, currency)
  end

  def -(other)
    raise "Currency mismatch" if currency != other.currency
    Money.new(amount - other.amount, currency)
  end

  def *(multiplier)
    Money.new(amount * multiplier, currency)
  end

  def ==(other)
    amount == other.amount && currency == other.currency
  end

  def <=>(other)
    return nil if currency != other.currency
    amount <=> other.amount
  end

  def -@
    Money.new(-amount, currency)
  end

  def to_s
    "#{amount} #{currency}"
  end
end

m1 = Money.new(1000)
m2 = Money.new(500)
puts m1 + m2   # => 1500 JPY
puts m1 - m2   # => 500 JPY
puts m1 * 2    # => 2000 JPY
puts m1 == m2  # => false
puts -m1       # => -1000 JPY
```

**🐍 Python:**

```python
# 演算子はダンダー（__名前__）メソッドで定義する
class Money:
    def __init__(self, amount, currency="JPY"):
        self.amount = amount
        self.currency = currency

    def __add__(self, other):        # +
        return Money(self.amount + other.amount, self.currency)

    def __sub__(self, other):        # -
        return Money(self.amount - other.amount, self.currency)

    def __mul__(self, m):            # *
        return Money(self.amount * m, self.currency)

    def __eq__(self, other):         # ==
        return self.amount == other.amount and self.currency == other.currency

    def __neg__(self):               # 単項マイナス（Rubyの -@）
        return Money(-self.amount, self.currency)

    def __str__(self):               # Rubyの to_s に相当
        return f"{self.amount} {self.currency}"

m1, m2 = Money(1000), Money(500)
print(m1 + m2)   # => 1500 JPY
print(-m1)       # => -1000 JPY
```

Rubyが演算子名そのものをメソッド名にするのに対し、Pythonは `+` → `__add__`、`==` → `__eq__`、`-@` → `__neg__`、`[]` → `__getitem__`、`[]=` → `__setitem__` のように「特殊メソッド名（ダンダー）」で定義します。`<=>`（宇宙船演算子）はPythonにありません。

### Comparableモジュール

```ruby
class Person
  include Comparable

  attr_reader :name, :age

  def initialize(name, age)
    @name = name
    @age = age
  end

  # <=> を定義すると、<, <=, ==, >=, > が使える
  def <=>(other)
    age <=> other.age
  end
end

p1 = Person.new("太郎", 25)
p2 = Person.new("次郎", 30)

p1 < p2   # => true
p1 >= p2  # => false
[p2, p1].sort  # => [p1, p2]
```

**🐍 Python:**

```python
from functools import total_ordering

@total_ordering                      # Comparable に相当
class Person:
    def __init__(self, name, age):
        self.name, self.age = name, age

    def __eq__(self, other):
        return self.age == other.age

    def __lt__(self, other):         # < を定義
        return self.age < other.age

p1, p2 = Person("太郎", 25), Person("次郎", 30)
p1 < p2                              # => True
sorted([p2, p1], key=lambda p: p.age)   # キー指定でのソートも一般的
```

Rubyの `include Comparable` + `<=>` に相当するのが、Pythonの `@functools.total_ordering` デコレータです。`__eq__` と `__lt__` を定義すれば残りの比較演算子（`<=`, `>`, `>=`）を自動生成してくれます。ソートは `<=>` に頼らず `sorted(..., key=...)` でキーを指定する書き方も広く使われます。

## 10. メソッドの動的定義

### define_method

```ruby
class MyClass
  # 動的にメソッドを定義
  define_method(:greet) do |name|
    "Hello, #{name}!"
  end

  # ループで複数のメソッドを定義
  %w[red green blue].each do |color|
    define_method("#{color}?") do
      @color == color
    end
  end
end

obj = MyClass.new
obj.greet("Ruby")  # => "Hello, Ruby!"
```

**🐍 Python:**

```python
# setattr でクラスにメソッド（関数）を後付けできる
class MyClass:
    pass

def greet(self, name):
    return f"Hello, {name}!"

setattr(MyClass, "greet", greet)

# ループで複数定義
for color in ("red", "green", "blue"):
    setattr(MyClass, f"is_{color}",
            lambda self, c=color: self.color == c)   # c=color で束縛を固定
```

Rubyの `define_method` に相当するのは `setattr(クラス, 名前, 関数)` です。ループ内で `lambda` を使う場合は、Ruby と違いループ変数が遅延束縛されるため、`c=color` のようにデフォルト引数で値を固定する必要があります。

### method_missing

```ruby
class Flexible
  def method_missing(name, *args, &block)
    if name.to_s.start_with?("find_by_")
      attribute = name.to_s.sub("find_by_", "")
      puts "Finding by #{attribute}: #{args.first}"
    else
      super  # 親クラスに委譲（NoMethodErrorを発生）
    end
  end

  # respond_to_missing? も定義すべき
  def respond_to_missing?(name, include_private = false)
    name.to_s.start_with?("find_by_") || super
  end
end

f = Flexible.new
f.find_by_name("太郎")   # => Finding by name: 太郎
f.find_by_email("a@b")   # => Finding by email: a@b
f.respond_to?(:find_by_id)  # => true
```

**🐍 Python:**

```python
# __getattr__ は「未定義の属性/メソッドにアクセスしたとき」に呼ばれる
class Flexible:
    def __getattr__(self, name):
        if name.startswith("find_by_"):
            attribute = name[len("find_by_"):]
            def finder(value):
                print(f"Finding by {attribute}: {value}")
            return finder                       # 呼び出し可能オブジェクトを返す
        raise AttributeError(name)              # Rubyの super 相当

f = Flexible()
f.find_by_name("太郎")    # => Finding by name: 太郎
```

Rubyの `method_missing` に相当するのがPythonの `__getattr__` です。ただし `__getattr__` は「属性アクセス」をフックするので、メソッド呼び出しに対応するには（引数を受け取る）関数を返す点が異なります。`respond_to_missing?` に直接対応するものはありません。

```ruby
obj.aliased   # => "元のメソッド"
```

**🐍 Python:**

```python
class MyClass:
    def original(self):
        return "元のメソッド"

    aliased = original     # 別名は単なる代入（同じ関数オブジェクトを参照）
```

Rubyの `alias_method` に相当する専用構文はありません。関数もオブジェクトなので、`aliased = original` のようにクラス本体で代入するだけで別名になります。
