## 1. 条件分岐

### if文

```ruby
# 基本形
if condition
  # 条件が真の場合
end

# if-else
if condition
  # 真の場合
else
  # 偽の場合
end

# if-elsif-else
score = 85

if score >= 90
  grade = "A"
elsif score >= 80
  grade = "B"
elsif score >= 70
  grade = "C"
elsif score >= 60
  grade = "D"
else
  grade = "F"
end

# then キーワード（1行で書く場合）
if condition then do_something end
if score >= 80 then puts "合格" else puts "不合格" end
```

### 後置if（修飾子）

```ruby
# 1行で書ける場合に使用
puts "成人です" if age >= 18
return nil if value.nil?
raise "Error" if invalid?

# 複雑な条件でも可能（ただし読みにくくなる）
process_data if valid? && authorized?
```

### if式（値を返す）

```ruby
# ifは値を返す
result = if score >= 60
           "合格"
         else
           "不合格"
         end

# 1行で書く場合
status = if active? then "有効" else "無効" end

# 変数への代入を条件にする
if (match = string.match(/pattern/))
  puts match[1]
end

# 三項演算子（短い場合）
status = age >= 18 ? "成人" : "未成年"
```

### unless文

```ruby
# ifの逆（条件が偽の場合に実行）
unless logged_in?
  redirect_to login_path
end

# 後置unless
return unless valid?
puts "警告" unless silent_mode?

# unless-else（非推奨、if-elseを使う）
unless condition
  # 偽の場合
else
  # 真の場合（わかりにくい）
end

# elsifは使えない
# unless condition
#   ...
# elsif other_condition  # SyntaxError!
#   ...
# end
```

### 三項演算子

```ruby
# condition ? true_value : false_value
status = active ? "有効" : "無効"

# ネスト（非推奨）
grade = score >= 90 ? "A" :
        score >= 80 ? "B" :
        score >= 70 ? "C" : "F"

# 上記は以下のように書くべき
grade = case
        when score >= 90 then "A"
        when score >= 80 then "B"
        when score >= 70 then "C"
        else "F"
        end
```

### case文

```ruby
# 値によるマッチング
case fruit
when "apple"
  puts "りんご"
when "banana"
  puts "バナナ"
when "orange", "mandarin"  # 複数の値
  puts "柑橘類"
else
  puts "不明"
end

# 範囲でマッチング
case score
when 90..100
  "A"
when 80...90  # 90は含まない
  "B"
when 70...80
  "C"
else
  "F"
end

# 正規表現でマッチング
case email
when /\A[\w+\-.]+@gmail\.com\z/i
  "Gmail"
when /\A[\w+\-.]+@yahoo\.com\z/i
  "Yahoo"
else
  "Other"
end

# クラスでマッチング
case value
when Integer
  "整数"
when Float
  "浮動小数点"
when String
  "文字列"
when Array
  "配列"
else
  "その他"
end

# Procでマッチング
even = ->(n) { n.even? }
positive = ->(n) { n > 0 }

case number
when even
  "偶数"
when positive
  "正の奇数"
else
  "負の奇数"
end

# 条件式なしのcase（if-elsifの代替）
case
when score >= 90
  "A"
when score >= 80
  "B"
when score >= 70
  "C"
else
  "F"
end
```

### case文の仕組み（===演算子）

```ruby
# case文は内部で === 演算子を使用
# when value は case_value === value と同等

# 各クラスの === の振る舞い
Integer === 5        # => true（型チェック）
(1..10) === 5        # => true（範囲チェック）
/\d+/ === "123"      # => true（正規表現マッチ）
String === "hello"   # => true（型チェック）
->(x) { x > 0 } === 5  # => true（Proc呼び出し）

# カスタムクラスで === を定義
class Even
  def self.===(value)
    value.is_a?(Integer) && value.even?
  end
end

case 4
when Even
  puts "偶数です"
end
```

### 論理演算子

```ruby
# && と and
# どちらも論理AND、ただし優先度が異なる
result = condition1 && condition2  # 高優先度
result = condition1 and condition2 # 低優先度

# || と or
result = value1 || value2          # 高優先度
result = value1 or value2          # 低優先度

# not と !
result = !condition                # 高優先度
result = not condition             # 低優先度

# 優先度の違いに注意
x = false || true   # x = true
x = false or true   # x = false（代入が先に評価）
# 実際は (x = false) or true と解釈される

# 短絡評価（ショートサーキット）
false && expensive_operation  # 右辺は評価されない
true || expensive_operation   # 右辺は評価されない

# デフォルト値のイディオム
name = params[:name] || "Guest"
name ||= "Guest"  # 同上（nilまたはfalseの場合に代入）

# 安全なメソッド呼び出し
user && user.name           # userがnilなら nil
user&.name                  # 同上（Ruby 2.3+、推奨）
```

## 2. 繰り返し

### while文

```ruby
# 条件が真の間、繰り返す
i = 0
while i < 5
  puts i
  i += 1
end

# 1行で書く場合
i = 0
i += 1 while i < 5

# 後置while（最低1回は実行）
i = 0
begin
  puts i
  i += 1
end while i < 5
```

### until文

```ruby
# 条件が偽の間、繰り返す（whileの逆）
i = 0
until i >= 5
  puts i
  i += 1
end

# 後置until
i = 0
i += 1 until i >= 5

# begin-until
i = 0
begin
  puts i
  i += 1
end until i >= 5
```

### for文

```ruby
# Rubyではあまり使わない（eachを推奨）
for i in 1..5
  puts i
end

for item in [1, 2, 3]
  puts item
end

for key, value in { a: 1, b: 2 }
  puts "#{key}: #{value}"
end

# 注意: forはスコープを作らない
for i in 1..3
  x = i
end
puts x  # => 3（forの外でも参照可能）

# eachはブロックスコープ
[1, 2, 3].each do |i|
  y = i
end
# puts y  # NameError! yは見えない
```

### loop

```ruby
# 無限ループ
loop do
  puts "永遠に続く"
  break if should_stop?
end

# Kernel#loopはStopIteration例外で終了
iter = [1, 2, 3].each
loop do
  puts iter.next
end
# => 1, 2, 3（StopIterationで自動終了）
```

### timesメソッド

```ruby
# 指定回数繰り返す
5.times do
  puts "Hello"
end

# インデックス付き
5.times do |i|
  puts "#{i}回目"  # 0, 1, 2, 3, 4
end

# 1行で
5.times { |i| puts i }
```

### upto / downto

```ruby
# 増加
1.upto(5) do |i|
  puts i  # 1, 2, 3, 4, 5
end

# 減少
5.downto(1) do |i|
  puts i  # 5, 4, 3, 2, 1
end

# 1行で
1.upto(5) { |i| puts i }
```

### step

```ruby
# 指定したステップで繰り返す
1.step(10, 2) do |i|
  puts i  # 1, 3, 5, 7, 9
end

# 減少
10.step(1, -2) { |i| puts i }  # 10, 8, 6, 4, 2

# Float
0.step(1, 0.1) { |f| puts f }  # 0.0, 0.1, 0.2, ...
```

## 3. ループ制御

### break

```ruby
# ループを終了
i = 0
while i < 10
  break if i == 5
  puts i
  i += 1
end
# => 0, 1, 2, 3, 4

# 値を返す
result = [1, 2, 3, 4, 5].each do |n|
  break n * 10 if n > 3
end
puts result  # => 40

# ネストしたループ
[1, 2, 3].each do |i|
  [4, 5, 6].each do |j|
    break if j == 5  # 内側のループのみ終了
    puts "#{i}, #{j}"
  end
end
```

### next

```ruby
# 次のイテレーションへスキップ
(1..5).each do |i|
  next if i.even?
  puts i
end
# => 1, 3, 5

# 値を返す（ブロックの戻り値）
result = [1, 2, 3, 4, 5].map do |n|
  next 0 if n.even?
  n * 2
end
puts result  # => [2, 0, 6, 0, 10]
```

### redo

```ruby
# 同じイテレーションを再実行（条件変更なし）
count = 0
(1..3).each do |i|
  count += 1
  if count == 2 && i == 1
    redo  # i=1のまま再実行
  end
  puts "i=#{i}, count=#{count}"
end
# => i=1, count=1
#    i=1, count=3（redoで再実行）
#    i=2, count=4
#    i=3, count=5

# 実用例: 入力の再試行
loop do
  print "名前を入力: "
  name = gets.chomp
  redo if name.empty?
  puts "Hello, #{name}!"
  break
end
```

### retry（例外処理用）

```ruby
# 例外処理のコンテキストで使用
attempts = 0
begin
  attempts += 1
  connect_to_server
rescue ConnectionError
  retry if attempts < 3
  raise
end
```

### throw / catch（大域脱出）

```ruby
# ネストしたループからの脱出
catch(:done) do
  [1, 2, 3].each do |i|
    [4, 5, 6].each do |j|
      if i * j > 10
        throw :done  # catchまでジャンプ
      end
      puts "#{i} * #{j} = #{i * j}"
    end
  end
end

# 値を返す
result = catch(:found) do
  [1, 2, 3].each do |i|
    [4, 5, 6].each do |j|
      throw :found, [i, j] if i * j > 10
    end
  end
  nil  # 見つからなかった場合
end
puts result  # => [2, 6]
```

## 4. イテレータ

### each

```ruby
# 配列
[1, 2, 3].each do |item|
  puts item
end

# ハッシュ
{ a: 1, b: 2 }.each do |key, value|
  puts "#{key}: #{value}"
end

# 範囲
(1..5).each { |n| puts n }

# 文字列
"hello".each_char { |c| puts c }
"hello\nworld".each_line { |line| puts line }
```

### each_with_index / each_with_object

```ruby
# インデックス付き
%w[a b c].each_with_index do |item, index|
  puts "#{index}: #{item}"
end

# オブジェクト付き（アキュムレータ）
result = [1, 2, 3].each_with_object({}) do |num, hash|
  hash[num] = num * 2
end
# => {1=>2, 2=>4, 3=>6}

# 配列を累積
result = [1, 2, 3].each_with_object([]) do |num, arr|
  arr << num * 2
end
# => [2, 4, 6]
```

### map / collect

```ruby
# 各要素を変換した新しい配列を返す
[1, 2, 3].map { |n| n * 2 }     # => [2, 4, 6]
[1, 2, 3].collect { |n| n * 2 } # => [2, 4, 6]（別名）

# メソッド参照
%w[hello world].map(&:upcase)   # => ["HELLO", "WORLD"]

# 破壊的
arr = [1, 2, 3]
arr.map! { |n| n * 2 }
puts arr  # => [2, 4, 6]
```

### select / reject / filter

```ruby
# 条件に合う要素を抽出
[1, 2, 3, 4, 5].select { |n| n.even? }  # => [2, 4]
[1, 2, 3, 4, 5].filter { |n| n.even? }  # => [2, 4]（別名）
[1, 2, 3, 4, 5].find_all { |n| n.even? } # => [2, 4]（別名）

# 条件に合わない要素を抽出
[1, 2, 3, 4, 5].reject { |n| n.even? }  # => [1, 3, 5]

# 破壊的
arr = [1, 2, 3, 4, 5]
arr.select! { |n| n.even? }
puts arr  # => [2, 4]

arr = [1, 2, 3, 4, 5]
arr.reject! { |n| n.even? }
puts arr  # => [1, 3, 5]
```

### find / detect

```ruby
# 条件に合う最初の要素を返す
[1, 2, 3, 4, 5].find { |n| n > 3 }    # => 4
[1, 2, 3, 4, 5].detect { |n| n > 3 }  # => 4（別名）

# 見つからない場合
[1, 2, 3].find { |n| n > 10 }         # => nil

# デフォルト値
[1, 2, 3].find(-> { "not found" }) { |n| n > 10 }
# => "not found"
```

### reduce / inject

```ruby
# 畳み込み
[1, 2, 3, 4, 5].reduce(0) { |sum, n| sum + n }  # => 15
[1, 2, 3, 4, 5].inject(0) { |sum, n| sum + n }  # => 15（別名）

# 初期値を省略（最初の要素が初期値）
[1, 2, 3, 4, 5].reduce { |sum, n| sum + n }    # => 15

# シンボルで指定
[1, 2, 3, 4, 5].reduce(:+)   # => 15
[1, 2, 3, 4, 5].reduce(:*)   # => 120

# 複雑な例
words = %w[apple banana cherry]
result = words.reduce({}) do |hash, word|
  hash[word] = word.length
  hash
end
# => {"apple"=>5, "banana"=>6, "cherry"=>6}
```

### any? / all? / none? / one?

```ruby
arr = [1, 2, 3, 4, 5]

# いずれかが条件を満たす
arr.any? { |n| n > 3 }   # => true

# すべてが条件を満たす
arr.all? { |n| n > 0 }   # => true

# どれも条件を満たさない
arr.none? { |n| n < 0 }  # => true

# ちょうど1つが条件を満たす
arr.one? { |n| n > 4 }   # => true

# 引数なし
[nil, false, nil].any?   # => false
[1, nil, 2].all?         # => false
[nil, false].none?       # => true

# パターン引数（Ruby 2.5+）
[1, 2, 3].any?(Integer)  # => true
%w[a b c].all?(/\A\w\z/) # => true
```

### count / size / length

```ruby
arr = [1, 2, 3, 4, 5, 2, 2]

arr.size      # => 7
arr.length    # => 7（同じ）
arr.count     # => 7（同じ）

# 条件付きcount
arr.count(2)              # => 3（2の個数）
arr.count { |n| n.even? } # => 4（偶数の個数）
```

### first / last / take / drop

```ruby
arr = [1, 2, 3, 4, 5]

arr.first           # => 1
arr.first(2)        # => [1, 2]
arr.last            # => 5
arr.last(2)         # => [4, 5]

arr.take(3)         # => [1, 2, 3]
arr.drop(2)         # => [3, 4, 5]

arr.take_while { |n| n < 4 }  # => [1, 2, 3]
arr.drop_while { |n| n < 3 }  # => [3, 4, 5]
```

### sort / sort_by

```ruby
# 基本ソート
[3, 1, 4, 1, 5].sort          # => [1, 1, 3, 4, 5]
[3, 1, 4, 1, 5].sort.reverse  # => [5, 4, 3, 1, 1]

# ブロックでカスタム
[3, 1, 4].sort { |a, b| b <=> a }  # => [4, 3, 1]（降順）

# sort_by（変換してソート）
words = %w[banana apple cherry]
words.sort_by { |w| w.length }     # => ["apple", "banana", "cherry"]
words.sort_by(&:length)            # 同上

# 複数条件
users.sort_by { |u| [u.age, u.name] }

# 破壊的
arr.sort!
arr.sort_by! { |x| -x }
```

### group_by / partition

```ruby
# グループ化
numbers = [1, 2, 3, 4, 5, 6]
numbers.group_by { |n| n % 3 }
# => {1=>[1, 4], 2=>[2, 5], 0=>[3, 6]}

words = %w[apple banana avocado cherry]
words.group_by { |w| w[0] }
# => {"a"=>["apple", "avocado"], "b"=>["banana"], "c"=>["cherry"]}

# 2分割
numbers.partition { |n| n.even? }
# => [[2, 4, 6], [1, 3, 5]]
```

### flat_map / flatten

```ruby
# mapしてflatten
[[1, 2], [3, 4]].flat_map { |arr| arr.map { |n| n * 2 } }
# => [2, 4, 6, 8]

# 同等の処理
[[1, 2], [3, 4]].map { |arr| arr.map { |n| n * 2 } }.flatten
# => [2, 4, 6, 8]

# flatten
[[1, [2, 3]], [4, [5, 6]]].flatten     # => [1, 2, 3, 4, 5, 6]
[[1, [2, 3]], [4, [5, 6]]].flatten(1)  # => [1, 2, 3, 4, [5, 6]]
```

### zip / product

```ruby
# zip: 対応する要素をペアにする
[1, 2, 3].zip([4, 5, 6])
# => [[1, 4], [2, 5], [3, 6]]

[1, 2, 3].zip([4, 5], [6, 7, 8, 9])
# => [[1, 4, 6], [2, 5, 7], [3, nil, 8]]

# product: 直積
[1, 2].product([3, 4])
# => [[1, 3], [1, 4], [2, 3], [2, 4]]

[1, 2].product([3, 4], [5, 6])
# => [[1, 3, 5], [1, 3, 6], [1, 4, 5], [1, 4, 6],
#     [2, 3, 5], [2, 3, 6], [2, 4, 5], [2, 4, 6]]
```

### チェーン

```ruby
# メソッドチェーンで複雑な処理
users
  .select { |u| u.active? }
  .reject { |u| u.admin? }
  .sort_by { |u| u.name }
  .map { |u| u.email }
  .first(10)

# Enumerator::Lazy（遅延評価）
(1..Float::INFINITY)
  .lazy
  .select { |n| n.odd? }
  .map { |n| n * 2 }
  .take(5)
  .to_a
# => [2, 6, 10, 14, 18]
```

## 5. パターンマッチング

### 基本（Ruby 3.0+）

```ruby
# case/in でパターンマッチ
case [1, 2, 3]
in [a, b, c]
  puts "a=#{a}, b=#{b}, c=#{c}"
end

# ハッシュのマッチング
case { name: "太郎", age: 25 }
in { name:, age: }
  puts "#{name}は#{age}歳"
end

# 型のマッチング
case value
in Integer => n
  puts "整数: #{n}"
in String => s
  puts "文字列: #{s}"
end
```

### 配列パターン

```ruby
case arr
in []
  "空配列"
in [x]
  "1要素: #{x}"
in [x, y]
  "2要素: #{x}, #{y}"
in [x, *rest]
  "先頭: #{x}, 残り: #{rest}"
in [*head, x]
  "前: #{head}, 末尾: #{x}"
in [first, *, last]
  "先頭: #{first}, 末尾: #{last}"
end
```

### ハッシュパターン

```ruby
case user
in { name: "太郎" }
  "太郎さん"
in { name:, age: 18.. }
  "成人の#{name}さん"
in { name:, age: }
  "未成年の#{name}さん（#{age}歳）"
in { name:, **rest }
  "名前: #{name}, その他: #{rest}"
end

# キーの存在確認
case hash
in { required_key: }
  "キーが存在"
else
  "キーが存在しない"
end
```

### ガード条件

```ruby
case value
in n if n.positive?
  "正の数: #{n}"
in n if n.negative?
  "負の数: #{n}"
in n
  "ゼロ: #{n}"
end
```

### Alternative パターン

```ruby
case value
in 0 | 1 | 2
  "0, 1, または 2"
in Integer
  "その他の整数"
end
```

### 変数のピン演算子

```ruby
expected = 42

case value
in ^expected  # expectedの値とマッチ（変数への代入ではない）
  "期待値と一致"
in other
  "#{other}は期待値と異なる"
end
```

### 1行パターンマッチ

```ruby
# => でマッチと代入
{ name: "太郎", age: 25 } => { name:, age: }
puts name  # => "太郎"

# in で真偽値を返す
{ name: "太郎" } in { name: "太郎" }  # => true
{ name: "太郎" } in { name: "次郎" }  # => false
```

### 実践的な例

```ruby
# APIレスポンスの処理
def process_response(response)
  case response
  in { status: 200, body: { data: } }
    puts "成功: #{data}"
  in { status: 404 }
    puts "Not Found"
  in { status: 500, body: { error: message } }
    puts "サーバーエラー: #{message}"
  in { status: code }
    puts "予期しないステータス: #{code}"
  end
end

# JSONのパース結果を処理
def parse_user(json)
  case json
  in { user: { name: String => name, age: Integer => age, address: { city: } } }
    User.new(name: name, age: age, city: city)
  in { user: { name:, age: } }
    User.new(name: name, age: age)
  in { error: message }
    raise ParseError, message
  end
end
```

## 6. 例外的な制御フロー

### return

```ruby
def find_user(id)
  return nil if id.nil?  # 早期リターン

  users.find { |u| u.id == id }
end

# ブロック内のreturn
def process
  [1, 2, 3].each do |n|
    return n if n > 1  # メソッドからreturn
  end
end
process  # => 2

# Proc内のreturn（注意が必要）
def test
  proc = Proc.new { return 42 }  # メソッドからreturn
  proc.call
  puts "ここは実行されない"
end
test  # => 42

# lambdaのreturn（ブロックからのみ）
def test2
  lam = -> { return 42 }  # lambdaからのみreturn
  result = lam.call
  puts "ここは実行される: #{result}"
end
test2  # => "ここは実行される: 42"
```

### 多重代入での制御

# nilガード
user = find_user(id) or return nil

if (result = expensive_calculation) > threshold
  process(result)
end

### BEGIN / END

```ruby
# プログラム開始前に実行
BEGIN {
  puts "プログラム開始"
}

# プログラム終了時に実行
END {
  puts "プログラム終了"
}

puts "メイン処理"

# 出力:
# プログラム開始
# メイン処理
# プログラム終了
```

### at_exit

```ruby
# プログラム終了時のクリーンアップ
at_exit do
  puts "終了処理1"
end

at_exit do
  puts "終了処理2"
end

# 後に登録したものから実行（LIFO）
# 出力:
# 終了処理2
# 終了処理1
```
