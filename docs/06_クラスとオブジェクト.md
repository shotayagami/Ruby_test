## 1. オブジェクト指向の基礎

### Rubyのオブジェクトモデル

```ruby
# すべてがオブジェクト
42.class           # => Integer
"hello".class      # => String
[1, 2, 3].class    # => Array
nil.class          # => NilClass
true.class         # => TrueClass

# クラスもオブジェクト
String.class       # => Class
Class.class        # => Class

# クラスの継承階層
Integer.ancestors
# => [Integer, Numeric, Comparable, Object, Kernel, BasicObject]

String.ancestors
# => [String, Comparable, Object, Kernel, BasicObject]
```

### オブジェクトの基本操作

```ruby
obj = Object.new

# オブジェクトID（一意の識別子）
obj.object_id      # => 70123456789

# クラスの確認
obj.class          # => Object
obj.is_a?(Object)  # => true
obj.instance_of?(Object)  # => true

# メソッドの確認
obj.methods        # 利用可能なメソッド一覧
obj.respond_to?(:to_s)  # => true

# オブジェクトの文字列表現
obj.to_s           # => "#<Object:0x00007f...>"
obj.inspect        # => "#<Object:0x00007f...>"
```

## 2. クラスの定義

### 基本的なクラス

```ruby
class User
  def initialize(name)
    @name = name
  end

  def greet
    "Hello, #{@name}!"
  end
end

user = User.new("太郎")
user.greet  # => "Hello, 太郎!"
```

### クラス名の規則

```ruby
# PascalCase（必須）
class MyClass; end
class HTTPClient; end
class XMLParser; end

# 名前空間を使う場合
class MyApp::User; end

# または
module MyApp
  class User
  end
end
```

### クラスの構造

```ruby
class Person
  # 定数
  DEFAULT_AGE = 0

  # クラス変数
  @@count = 0

  # クラスメソッド
  def self.count
    @@count
  end

  # 初期化メソッド
  def initialize(name, age = DEFAULT_AGE)
    @name = name
    @age = age
    @@count += 1
  end

  # インスタンスメソッド
  def introduce
    "私は#{@name}、#{@age}歳です"
  end

  # プライベートメソッド
  private

  def secret
    "秘密の情報"
  end
end
```

## 3. インスタンス変数とアクセサ

### インスタンス変数

```ruby
class User
  def initialize(name, email)
    @name = name      # インスタンス変数
    @email = email
  end

  # ゲッター（読み取り）
  def name
    @name
  end

  # セッター（書き込み）
  def name=(value)
    @name = value
  end

  # ゲッターとセッター
  def email
    @email
  end

  def email=(value)
    @email = value
  end
end

user = User.new("太郎", "taro@example.com")
user.name          # => "太郎"
user.name = "次郎"
user.name          # => "次郎"
```

### アクセサメソッド

```ruby
class User
  # ゲッターのみ
  attr_reader :id

  # セッターのみ
  attr_writer :password

  # ゲッターとセッター
  attr_accessor :name, :email

  def initialize(id, name, email)
    @id = id
    @name = name
    @email = email
  end
end

user = User.new(1, "太郎", "taro@example.com")
user.id              # => 1
# user.id = 2        # NoMethodError!（読み取り専用）
user.name            # => "太郎"
user.name = "次郎"   # OK
user.password = "secret"  # OK
# user.password      # NoMethodError!（書き込み専用）
```

### カスタムアクセサ

```ruby
class Temperature
  def initialize(celsius)
    @celsius = celsius
  end

  # ゲッター
  attr_reader :celsius

  # カスタムセッター（バリデーション付き）
  def celsius=(value)
    raise ArgumentError, "温度が低すぎます" if value < -273.15
    @celsius = value
  end

  # 計算プロパティ
  def fahrenheit
    @celsius * 9.0 / 5.0 + 32
  end

  def fahrenheit=(value)
    @celsius = (value - 32) * 5.0 / 9.0
  end
end

temp = Temperature.new(25)
temp.fahrenheit      # => 77.0
temp.fahrenheit = 100
temp.celsius         # => 37.777...
```

### インスタンス変数の操作

```ruby
class Demo
  def initialize
    @a = 1
    @b = 2
  end
end

obj = Demo.new

# インスタンス変数の一覧
obj.instance_variables  # => [:@a, :@b]

# 直接アクセス
obj.instance_variable_get(:@a)      # => 1
obj.instance_variable_set(:@a, 10)  # @a = 10
obj.instance_variable_defined?(:@a) # => true
```

## 4. クラスメソッドとクラス変数

### クラスメソッド

```ruby
class Calculator
  # 方法1: self.メソッド名
  def self.add(a, b)
    a + b
  end

  # 方法2: class << self
  class << self
    def subtract(a, b)
      a - b
    end

    def multiply(a, b)
      a * b
    end
  end
end

Calculator.add(1, 2)       # => 3
Calculator.subtract(5, 3)  # => 2
Calculator.multiply(4, 5)  # => 20
```

### クラス変数

```ruby
class Counter
  @@count = 0  # クラス変数

  def initialize
    @@count += 1
  end

  def self.count
    @@count
  end

  def count
    @@count
  end
end

Counter.count  # => 0
Counter.new
Counter.new
Counter.count  # => 2

# 注意: サブクラスと共有される
class SubCounter < Counter
end

SubCounter.new
Counter.count    # => 3（親クラスも影響を受ける）
SubCounter.count # => 3
```

### クラスインスタンス変数（推奨）

```ruby
class Counter
  @count = 0  # クラスインスタンス変数

  class << self
    attr_accessor :count
  end

  def initialize
    self.class.count += 1
  end
end

class SubCounter < Counter
  @count = 0  # サブクラス独自の変数
end

Counter.new
Counter.new
SubCounter.new

Counter.count     # => 2
SubCounter.count  # => 1（独立している）
```

### 定数

```ruby
class Config
  VERSION = "1.0.0"
  DEFAULT_OPTIONS = { timeout: 30, retries: 3 }.freeze

  # ネストした定数
  module Defaults
    HOST = "localhost"
    PORT = 3000
  end
end

Config::VERSION              # => "1.0.0"
Config::DEFAULT_OPTIONS      # => {:timeout=>30, :retries=>3}
Config::Defaults::HOST       # => "localhost"

# 定数の一覧
Config.constants             # => [:VERSION, :DEFAULT_OPTIONS, :Defaults]
```

## 5. 継承

### 基本的な継承

```ruby
class Animal
  def initialize(name)
    @name = name
  end

  def speak
    "..."
  end

  def introduce
    "私は#{@name}です"
  end
end

class Dog < Animal
  def speak
    "ワン！"
  end

  def fetch
    "#{@name}がボールを取ってきました"
  end
end

class Cat < Animal
  def speak
    "ニャー"
  end
end

dog = Dog.new("ポチ")
dog.speak      # => "ワン！"
dog.introduce  # => "私はポチです"
dog.fetch      # => "ポチがボールを取ってきました"

cat = Cat.new("タマ")
cat.speak      # => "ニャー"
```

### super

```ruby
class Animal
  def initialize(name)
    @name = name
  end

  def speak
    "動物の声"
  end
end

class Dog < Animal
  def initialize(name, breed)
    super(name)  # 親クラスのinitializeを呼ぶ
    @breed = breed
  end

  def speak
    original = super  # 親クラスのspeakを呼ぶ
    "#{original}...ワン！"
  end
end

dog = Dog.new("ポチ", "柴犬")
dog.speak  # => "動物の声...ワン！"

# superの引数
class Parent
  def greet(name, greeting = "Hello")
    "#{greeting}, #{name}!"
  end
end

class Child < Parent
  def greet(name)
    super         # すべての引数を渡す: super(name)
    super()       # 引数なし
    super(name, "Hi")  # 明示的に引数を指定
  end
end
```

### メソッドの検索順序

```ruby
module M1
  def foo
    "M1#foo"
  end
end

module M2
  def foo
    "M2#foo"
  end
end

class Parent
  def foo
    "Parent#foo"
  end
end

class Child < Parent
  include M1
  include M2  # 後からincludeしたものが優先

  def foo
    "Child#foo"
  end
end

Child.ancestors
# => [Child, M2, M1, Parent, Object, Kernel, BasicObject]

# メソッド検索: Child → M2 → M1 → Parent → Object → ...
```

### 継承の制限

```ruby
# Rubyは単一継承のみ
class A; end
class B < A; end
# class C < A, B  # SyntaxError!

# 多重継承の代わりにモジュールを使う
module Flyable
  def fly
    "飛んでいます"
  end
end

module Swimmable
  def swim
    "泳いでいます"
  end
end

class Duck < Animal
  include Flyable
  include Swimmable
end

duck = Duck.new("アヒル")
duck.fly   # => "飛んでいます"
duck.swim  # => "泳いでいます"
```

## 6. オブジェクトの初期化

### initialize メソッド

```ruby
class User
  def initialize(name, email)
    @name = name
    @email = email
    @created_at = Time.now
  end
end

# new が内部で initialize を呼ぶ
user = User.new("太郎", "taro@example.com")
```

### デフォルト値とオプション

```ruby
class User
  def initialize(name, email: nil, role: "user", **options)
    @name = name
    @email = email
    @role = role
    @options = options
  end
end

User.new("太郎")
User.new("太郎", email: "taro@example.com")
User.new("太郎", email: "taro@example.com", role: "admin")
User.new("太郎", age: 25, city: "Tokyo")  # 追加オプション
```

### ファクトリメソッド

```ruby
class User
  attr_reader :name, :email, :type

  def initialize(name, email, type)
    @name = name
    @email = email
    @type = type
  end

  # ファクトリメソッド
  def self.guest
    new("Guest", "guest@example.com", :guest)
  end

  def self.admin(name, email)
    new(name, email, :admin)
  end

  def self.from_hash(hash)
    new(hash[:name], hash[:email], hash[:type] || :user)
  end
end

guest = User.guest
admin = User.admin("管理者", "admin@example.com")
user = User.from_hash({ name: "太郎", email: "taro@example.com" })
```

### 遅延初期化

```ruby
class Config
  def database
    @database ||= load_database_config
  end

  def cache
    @cache ||= initialize_cache
  end

  private

  def load_database_config
    puts "Loading database config..."
    { host: "localhost", port: 5432 }
  end

  def initialize_cache
    puts "Initializing cache..."
    {}
  end
end

config = Config.new
# まだ何も読み込まれていない
config.database  # "Loading database config..." を出力
config.database  # 2回目はキャッシュを使用
```

## 7. オブジェクトの比較と複製

### 等価性

```ruby
class User
  attr_reader :id, :name

  def initialize(id, name)
    @id = id
    @name = name
  end

  # 値の等価性
  def ==(other)
    return false unless other.is_a?(User)
    id == other.id
  end

  # ハッシュキーとしての等価性
  def eql?(other)
    self == other
  end

  def hash
    id.hash
  end
end

u1 = User.new(1, "太郎")
u2 = User.new(1, "太郎")
u3 = User.new(2, "次郎")

u1 == u2        # => true（値が同じ）
u1 == u3        # => false
u1.equal?(u2)   # => false（別オブジェクト）

# ハッシュのキーとして使用
hash = { u1 => "value1" }
hash[u2]  # => "value1"（eql?とhashが定義されている）
```

### オブジェクトの複製

```ruby
class Container
  attr_accessor :items

  def initialize
    @items = []
  end
end

original = Container.new
original.items << "item1"

# 浅いコピー（dup）
copy = original.dup
copy.items << "item2"
original.items  # => ["item1", "item2"]（配列は共有）

# 浅いコピー（clone）
# dupとほぼ同じだが、特異メソッドとfrozen状態もコピー
clone = original.clone

# 深いコピー（自前実装）
class Container
  def deep_copy
    copy = dup
    copy.items = items.dup
    copy
  end
end

# Marshal を使った深いコピー
deep = Marshal.load(Marshal.dump(original))
```

### initialize_copy

```ruby
class Container
  attr_accessor :items, :metadata

  def initialize
    @items = []
    @metadata = {}
  end

  # dup/clone時に呼ばれる
  def initialize_copy(original)
    super
    @items = original.items.dup
    @metadata = original.metadata.dup
  end
end

original = Container.new
original.items << "item"
original.metadata[:key] = "value"

copy = original.dup
copy.items << "new item"
original.items  # => ["item"]（独立している）
```

### freeze

```ruby
# オブジェクトを凍結（変更不可に）
str = "hello"
str.freeze
# str << " world"  # FrozenError!

str.frozen?  # => true

# 配列の凍結
arr = [1, 2, 3]
arr.freeze
# arr << 4  # FrozenError!
# arr[0] = 10  # FrozenError!

# 注意: 中身は凍結されない
nested = [[1, 2], [3, 4]].freeze
# nested << [5]  # FrozenError!
nested[0] << 3   # OK!（内部の配列は凍結されていない）

# 完全に凍結するには
nested = [[1, 2].freeze, [3, 4].freeze].freeze
```

## 8. 特異メソッドと特異クラス

### 特異メソッド

```ruby
# 特定のオブジェクトだけに定義されるメソッド
str = "hello"

def str.shout
  upcase + "!"
end

str.shout  # => "HELLO!"
"hello".shout  # NoMethodError!（別のオブジェクト）

# クラスメソッドは特異メソッドの一種
class MyClass
  def self.foo  # MyClassオブジェクトの特異メソッド
    "class method"
  end
end
```

### 特異クラス（シングルトンクラス）

```ruby
str = "hello"

# 特異クラスにアクセス
singleton_class = str.singleton_class

# 特異クラスにメソッドを定義
singleton_class.define_method(:shout) do
  upcase + "!"
end

str.shout  # => "HELLO!"

# class << obj 構文
class << str
  def whisper
    downcase + "..."
  end
end

str.whisper  # => "hello..."
```

### クラスメソッドと特異クラス

```ruby
class MyClass
  # 方法1: self.メソッド名
  def self.method1
    "method1"
  end

  # 方法2: 特異クラスに定義
  class << self
    def method2
      "method2"
    end

    # アクセサも定義可能
    attr_accessor :config

    private

    def private_class_method
      "private"
    end
  end
end

MyClass.method1  # => "method1"
MyClass.method2  # => "method2"
MyClass.config = { key: "value" }
```

### extend

```ruby
module Greetable
  def greet
    "Hello!"
  end
end

# 特定のオブジェクトにモジュールをミックスイン
obj = Object.new
obj.extend(Greetable)
obj.greet  # => "Hello!"

# クラスにextendするとクラスメソッドになる
class MyClass
  extend Greetable
end

MyClass.greet  # => "Hello!"
```

## 9. オープンクラス

### 既存クラスの拡張

```ruby
# Rubyでは既存のクラスを再オープンできる
class String
  def shout
    upcase + "!"
  end
end

"hello".shout  # => "HELLO!"

# 組み込みクラスも拡張可能
class Integer
  def to_binary
    to_s(2)
  end
end

42.to_binary  # => "101010"
```

### モンキーパッチの注意点

```ruby
# 危険: 既存メソッドの上書き
class Array
  def first  # 元のメソッドが消える！
    "常にこれ"
  end
end

[1, 2, 3].first  # => "常にこれ"（壊れた）

# 安全: alias で元を保存
class String
  alias_method :original_reverse, :reverse

  def reverse
    puts "Reversing..."
    original_reverse
  end
end
```

### Refinements（安全な拡張）

```ruby
# モジュールで refinement を定義
module StringExtensions
  refine String do
    def shout
      upcase + "!"
    end
  end
end

# refinement を使用するスコープで有効化
class MyClass
  using StringExtensions

  def process(str)
    str.shout  # => 使える
  end
end

MyClass.new.process("hello")  # => "HELLO!"
# "hello".shout  # NoMethodError!（スコープ外では使えない）

# ファイルスコープで有効化
using StringExtensions
"hello".shout  # => "HELLO!"
```

## 10. Struct と Data

### Struct

```ruby
# 簡易的なクラスを作成
User = Struct.new(:name, :email, :age)

user = User.new("太郎", "taro@example.com", 25)
user.name   # => "太郎"
user.email  # => "taro@example.com"
user.age    # => 25

# 配列のようにアクセス
user[0]     # => "太郎"
user[:name] # => "太郎"

# イテレーション
user.each { |value| puts value }
user.to_a   # => ["太郎", "taro@example.com", 25]
user.to_h   # => {:name=>"太郎", :email=>"taro@example.com", :age=>25}

# メソッドを追加
User = Struct.new(:name, :email) do
  def greeting
    "Hello, #{name}!"
  end
end

# keyword_init オプション（Ruby 2.5+）
User = Struct.new(:name, :email, keyword_init: true)
user = User.new(name: "太郎", email: "taro@example.com")
```

### Data（Ruby 3.2+）

```ruby
# イミュータブルな値オブジェクト
Point = Data.define(:x, :y)

point = Point.new(x: 10, y: 20)
point.x  # => 10
point.y  # => 20

# イミュータブル
# point.x = 30  # NoMethodError!

# 値の変更は with で新しいオブジェクトを作成
new_point = point.with(x: 30)
new_point.x  # => 30
point.x      # => 10（元は変わらない）

# メソッドを追加
Point = Data.define(:x, :y) do
  def distance_from_origin
    Math.sqrt(x**2 + y**2)
  end
end

point = Point.new(x: 3, y: 4)
point.distance_from_origin  # => 5.0

# パターンマッチング
case point
in Point(x: 0, y: 0)
  "原点"
in Point(x:, y:) if x == y
  "対角線上"
in Point(x:, y:)
  "(#{x}, #{y})"
end
```

### Struct vs Data vs Class

```ruby
# Struct: ミュータブル、軽量
User = Struct.new(:name, :age)
user = User.new("太郎", 25)
user.name = "次郎"  # OK

# Data: イミュータブル、値オブジェクト向け
Point = Data.define(:x, :y)
point = Point.new(x: 10, y: 20)
# point.x = 30  # エラー

# Class: フル機能、複雑なロジック向け
class User
  attr_accessor :name, :age

  def initialize(name, age)
    @name = name
    @age = age
  end

  def adult?
    age >= 18
  end
end
```

### 使い分けの指針

| 用途 | 推奨 |
|-----|------|
| 単純なデータ保持 | Struct |
| 値オブジェクト（座標、金額など） | Data |
| ビジネスロジックを持つ | Class |
| 設定値、オプション | Struct / Hash |
| API レスポンス | Struct / Data |

## 実践例: 完全なクラス設計

```ruby
class BankAccount
  # 定数
  MINIMUM_BALANCE = 0

  # クラスインスタンス変数
  @total_accounts = 0

  class << self
    attr_accessor :total_accounts

    def create(holder_name, initial_deposit = 0)
      account = new(holder_name, initial_deposit)
      @total_accounts += 1
      account
    end
  end

  # インスタンス変数とアクセサ
  attr_reader :account_number, :holder_name, :balance

  def initialize(holder_name, initial_deposit = 0)
    @account_number = generate_account_number
    @holder_name = holder_name
    @balance = initial_deposit
    @transactions = []
  end

  # パブリックメソッド
  def deposit(amount)
    raise ArgumentError, "入金額は正の数である必要があります" if amount <= 0

    @balance += amount
    record_transaction(:deposit, amount)
    balance
  end

  def withdraw(amount)
    raise ArgumentError, "出金額は正の数である必要があります" if amount <= 0
    raise InsufficientFundsError, "残高不足です" if balance - amount < MINIMUM_BALANCE

    @balance -= amount
    record_transaction(:withdrawal, amount)
    balance
  end

  def transfer(amount, to_account)
    withdraw(amount)
    to_account.deposit(amount)
  end

  def statement
    @transactions.map do |t|
      "#{t[:date]} | #{t[:type]} | #{t[:amount]} | #{t[:balance]}"
    end.join("\n")
  end

  def to_s
    "口座番号: #{account_number}, 名義: #{holder_name}, 残高: #{balance}円"
  end

  # プライベートメソッド
  private

  def generate_account_number
    "#{Time.now.to_i}#{rand(1000..9999)}"
  end

  def record_transaction(type, amount)
    @transactions << {
      date: Time.now,
      type: type,
      amount: amount,
      balance: balance
    }
  end

  # カスタム例外
  class InsufficientFundsError < StandardError; end
end

# 使用例
account1 = BankAccount.create("山田太郎", 10000)
account2 = BankAccount.create("鈴木花子", 5000)

account1.deposit(5000)
account1.withdraw(3000)
account1.transfer(2000, account2)

puts account1  # => 口座番号: ..., 名義: 山田太郎, 残高: 10000円
puts account1.statement
```
