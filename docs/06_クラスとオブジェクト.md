## 1. オブジェクト指向の基礎

### Rubyのオブジェクトモデル

```ruby
# すべてがオブジェクト
42.class           # => Integer
"hello".class      # => String
[1, 2, 3].class    # => Array
nil.class          # => NilClass
true.class         # => TrueClass

# クラスもオブジェクト
String.class       # => Class
Class.class        # => Class

# クラスの継承階層
Integer.ancestors
# => [Integer, Numeric, Comparable, Object, Kernel, BasicObject]

String.ancestors
# => [String, Comparable, Object, Kernel, BasicObject]
```

### オブジェクトの基本操作

```ruby
obj = Object.new

# オブジェクトID（一意の識別子）
obj.object_id      # => 70123456789

# クラスの確認
obj.class          # => Object
obj.is_a?(Object)  # => true
obj.instance_of?(Object)  # => true

# メソッドの確認
obj.methods        # 利用可能なメソッド一覧
obj.respond_to?(:to_s)  # => true

# オブジェクトの文字列表現
obj.to_s           # => "#<Object:0x00007f...>"
obj.inspect        # => "#<Object:0x00007f...>"
```

## 2. クラスの定義

### 基本的なクラス

```ruby
class User
  def initialize(name)
    @name = name
  end

  def greet
    "Hello, #{@name}!"
  end
end

user = User.new("太郎")
user.greet  # => "Hello, 太郎!"
```

**🐍 Python:**

```python
class User:
    def __init__(self, name):
        self.name = name          # インスタンス変数は self.name

    def greet(self):              # 第一引数 self は明示的に必要
        return f"Hello, {self.name}!"

user = User("太郎")               # .new ではなく直接呼び出し
user.greet()                      # => "Hello, 太郎!"
```

Rubyの `initialize` はPythonの `__init__` に相当し、`@name` は `self.name` になる。Rubyは `self` が暗黙的だが、Pythonは各メソッドの第一引数として `self` を必ず明示する。インスタンス化も `User.new(...)` ではなく `User(...)` と書く。

### クラス名の規則

```ruby
# PascalCase（必須）
class MyClass; end
class HTTPClient; end
class XMLParser; end

# 名前空間を使う場合
class MyApp::User; end

# または
module MyApp
  class User
  end
end
```

### クラスの構造

```ruby
class Person
  # 定数
  DEFAULT_AGE = 0

  # クラス変数
  @@count = 0

  # クラスメソッド
  def self.count
    @@count
  end

  # 初期化メソッド
  def initialize(name, age = DEFAULT_AGE)
    @name = name
    @age = age
    @@count += 1
  end

  # インスタンスメソッド
  def introduce
    "私は#{@name}、#{@age}歳です"
  end

  # プライベートメソッド
  private

  def secret
    "秘密の情報"
  end
end
```

**🐍 Python:**

```python
class Person:
    DEFAULT_AGE = 0          # クラス属性（定数は慣習として大文字）
    count = 0                # クラス属性（Rubyの @@count に相当）

    @classmethod
    def get_count(cls):      # クラスメソッド（第一引数は cls）
        return cls.count

    def __init__(self, name, age=DEFAULT_AGE):
        self.name = name
        self.age = age
        Person.count += 1

    def introduce(self):
        return f"私は{self.name}、{self.age}歳です"

    def _secret(self):       # 慣習上の「内部用」。強制はされない
        return "秘密の情報"
```

重要な言語差: Rubyの `private/protected/public` は実行時に強制されるが、Pythonにはアクセス制御の強制がない。慣習として先頭に `_` を1つ付けたもの（`_secret`）を「内部用」とみなし、`__secret`（アンダースコア2つ）にすると名前マングリング（`_Person__secret` に変換）でサブクラスからの偶発的な衝突を防げるが、外部からのアクセス自体は禁止されない。

## 3. インスタンス変数とアクセサ

### インスタンス変数

```ruby
class User
  def initialize(name, email)
    @name = name      # インスタンス変数
    @email = email
  end

  # ゲッター（読み取り）
  def name
    @name
  end

  # セッター（書き込み）
  def name=(value)
    @name = value
  end

  # ゲッターとセッター
  def email
    @email
  end

  def email=(value)
    @email = value
  end
end

user = User.new("太郎", "taro@example.com")
user.name          # => "太郎"
user.name = "次郎"
user.name          # => "次郎"
```

**🐍 Python:**

```python
class User:
    def __init__(self, name, email):
        self.name = name       # Pythonの流儀では、まず公開属性にする
        self.email = email

user = User("太郎", "taro@example.com")
user.name          # => "太郎"
user.name = "次郎"  # そのまま代入・参照できる
```

Pythonでは属性はデフォルトで公開なので、Rubyのように単なる読み書きのためにゲッター/セッターを定義しない。バリデーションや計算が必要になったときだけ、後述の `@property` に「後付け」で切り替えられるのがPython流。

### アクセサメソッド

```ruby
class User
  # ゲッターのみ
  attr_reader :id

  # セッターのみ
  attr_writer :password

  # ゲッターとセッター
  attr_accessor :name, :email

  def initialize(id, name, email)
    @id = id
    @name = name
    @email = email
  end
end

user = User.new(1, "太郎", "taro@example.com")
user.id              # => 1
# user.id = 2        # NoMethodError!（読み取り専用）
user.name            # => "太郎"
user.name = "次郎"   # OK
user.password = "secret"  # OK
# user.password      # NoMethodError!（書き込み専用）
```

**🐍 Python:**

```python
from dataclasses import dataclass, field

# 方法1: 公開属性でよいなら dataclass が最も簡潔（attr_accessor 相当）
@dataclass
class User:
    id: int
    name: str
    email: str

user = User(1, "太郎", "taro@example.com")
user.name = "次郎"           # 自由に読み書きできる

# 方法2: 読み取り専用（attr_reader 相当）は @property で実現
class ReadOnlyUser:
    def __init__(self, id):
        self._id = id

    @property
    def id(self):
        return self._id      # ゲッターのみ → 代入するとエラー
```

Rubyの `attr_reader/attr_writer/attr_accessor` に一発で対応する構文はないが、単純なデータ保持なら `@dataclass` が定型コードを大幅に省ける。読み取り専用にしたい属性だけ `@property`（セッターを定義しない）を使う。

### カスタムアクセサ

```ruby
class Temperature
  def initialize(celsius)
    @celsius = celsius
  end

  # ゲッター
  attr_reader :celsius

  # カスタムセッター（バリデーション付き）
  def celsius=(value)
    raise ArgumentError, "温度が低すぎます" if value < -273.15
    @celsius = value
  end

  # 計算プロパティ
  def fahrenheit
    @celsius * 9.0 / 5.0 + 32
  end

  def fahrenheit=(value)
    @celsius = (value - 32) * 5.0 / 9.0
  end
end

temp = Temperature.new(25)
temp.fahrenheit      # => 77.0
temp.fahrenheit = 100
temp.celsius         # => 37.777...
```

**🐍 Python:**

```python
class Temperature:
    def __init__(self, celsius):
        self._celsius = celsius

    @property
    def celsius(self):                 # ゲッター（attr_reader 相当）
        return self._celsius

    @celsius.setter
    def celsius(self, value):          # バリデーション付きセッター
        if value < -273.15:
            raise ValueError("温度が低すぎます")
        self._celsius = value

    @property
    def fahrenheit(self):              # 計算プロパティ
        return self._celsius * 9.0 / 5.0 + 32

    @fahrenheit.setter
    def fahrenheit(self, value):
        self._celsius = (value - 32) * 5.0 / 9.0

temp = Temperature(25)
temp.fahrenheit                        # => 77.0
temp.fahrenheit = 100
temp.celsius                           # => 37.777...
```

Rubyのカスタムセッター `def celsius=(value)` は、Pythonでは `@property` でゲッターを定義してから `@celsius.setter` でセッターを追加する。呼び出し側は `temp.celsius = value` と通常の代入に見えるが、内部でメソッドが走る点はRubyと同じ。

### インスタンス変数の操作

```ruby
class Demo
  def initialize
    @a = 1
    @b = 2
  end
end

obj = Demo.new

# インスタンス変数の一覧
obj.instance_variables  # => [:@a, :@b]

# 直接アクセス
obj.instance_variable_get(:@a)      # => 1
obj.instance_variable_set(:@a, 10)  # @a = 10
obj.instance_variable_defined?(:@a) # => true
```

**🐍 Python:**

```python
class Demo:
    def __init__(self):
        self.a = 1
        self.b = 2

obj = Demo()

# インスタンス変数の一覧（属性名と値の辞書）
vars(obj)                    # => {'a': 1, 'b': 2}
obj.__dict__                 # 同上

# 名前を文字列で指定して直接アクセス
getattr(obj, "a")            # => 1
setattr(obj, "a", 10)        # obj.a = 10
hasattr(obj, "a")            # => True
```

Rubyの `instance_variable_get/set` は、Pythonの `getattr`/`setattr` に対応する。全インスタンス変数は `obj.__dict__`（または `vars(obj)`）に辞書として保持されている。

## 4. クラスメソッドとクラス変数

### クラスメソッド

```ruby
class Calculator
  # 方法1: self.メソッド名
  def self.add(a, b)
    a + b
  end

  # 方法2: class << self
  class << self
    def subtract(a, b)
      a - b
    end

    def multiply(a, b)
      a * b
    end
  end
end

Calculator.add(1, 2)       # => 3
Calculator.subtract(5, 3)  # => 2
Calculator.multiply(4, 5)  # => 20
```

**🐍 Python:**

```python
class Calculator:
    # self/cls を使わないなら @staticmethod
    @staticmethod
    def add(a, b):
        return a + b

    # クラス自身（cls）を受け取るなら @classmethod
    @classmethod
    def subtract(cls, a, b):
        return a - b

Calculator.add(1, 2)       # => 3
Calculator.subtract(5, 3)  # => 2
```

Rubyの `def self.add` に相当する書き方は2種類ある。クラスやインスタンスの状態を使わない純粋な関数なら `@staticmethod`、クラス自身（サブクラス含む）を `cls` として受け取りたいなら `@classmethod` を使う。

### クラス変数

```ruby
class Counter
  @@count = 0  # クラス変数

  def initialize
    @@count += 1
  end

  def self.count
    @@count
  end

  def count
    @@count
  end
end

Counter.count  # => 0
Counter.new
Counter.new
Counter.count  # => 2

# 注意: サブクラスと共有される
class SubCounter < Counter
end

SubCounter.new
Counter.count    # => 3（親クラスも影響を受ける）
SubCounter.count # => 3
```

**🐍 Python:**

```python
class Counter:
    count = 0                  # クラス属性（Rubyの @@count に相当）

    def __init__(self):
        Counter.count += 1     # クラス名で明示的に更新するのが安全

    @classmethod
    def get_count(cls):
        return cls.count

Counter()
Counter()
Counter.count                  # => 2
```

Rubyのクラス変数 `@@count` は、Pythonではクラス属性 `count = 0` として書く。参照は `Counter.count`。ただしインスタンス側から `self.count += 1` とすると、そのインスタンス独自の属性を新規作成してしまいクラス属性を更新できないため、更新は `Counter.count += 1` のようにクラス経由で行う。

### クラスインスタンス変数（推奨）

```ruby
class Counter
  @count = 0  # クラスインスタンス変数

  class << self
    attr_accessor :count
  end

  def initialize
    self.class.count += 1
  end
end

class SubCounter < Counter
  @count = 0  # サブクラス独自の変数
end

Counter.new
Counter.new
SubCounter.new

Counter.count     # => 2
SubCounter.count  # => 1（独立している）
```

### 定数

```ruby
class Config
  VERSION = "1.0.0"
  DEFAULT_OPTIONS = { timeout: 30, retries: 3 }.freeze

  # ネストした定数
  module Defaults
    HOST = "localhost"
    PORT = 3000
  end
end

Config::VERSION              # => "1.0.0"
Config::DEFAULT_OPTIONS      # => {:timeout=>30, :retries=>3}
Config::Defaults::HOST       # => "localhost"

# 定数の一覧
Config.constants             # => [:VERSION, :DEFAULT_OPTIONS, :Defaults]
```

> **🐍 Python:** `VERSION = "1.0.0"`（クラス属性を大文字にして「定数」を表現する慣習相当）

Pythonには真の定数がなく、大文字の名前で「変更しないでほしい」という意図を示すだけで、実際には再代入を防げない。参照は `Config.VERSION` のように書く（Rubyの `::` ではなく `.`）。

## 5. 継承

### 基本的な継承

```ruby
class Animal
  def initialize(name)
    @name = name
  end

  def speak
    "..."
  end

  def introduce
    "私は#{@name}です"
  end
end

class Dog < Animal
  def speak
    "ワン！"
  end

  def fetch
    "#{@name}がボールを取ってきました"
  end
end

class Cat < Animal
  def speak
    "ニャー"
  end
end

dog = Dog.new("ポチ")
dog.speak      # => "ワン！"
dog.introduce  # => "私はポチです"
dog.fetch      # => "ポチがボールを取ってきました"

cat = Cat.new("タマ")
cat.speak      # => "ニャー"
```

**🐍 Python:**

```python
class Animal:
    def __init__(self, name):
        self.name = name

    def speak(self):
        return "..."

    def introduce(self):
        return f"私は{self.name}です"

class Dog(Animal):              # class Dog < Animal に相当
    def speak(self):            # メソッドのオーバーライド
        return "ワン！"

    def fetch(self):
        return f"{self.name}がボールを取ってきました"

dog = Dog("ポチ")
dog.speak()                     # => "ワン！"
dog.introduce()                 # => "私はポチです"（親から継承）
```

Rubyの `class Dog < Animal` はPythonでは `class Dog(Animal):` と書く。親クラスは括弧内に指定する。

### super

```ruby
class Animal
  def initialize(name)
    @name = name
  end

  def speak
    "動物の声"
  end
end

class Dog < Animal
  def initialize(name, breed)
    super(name)  # 親クラスのinitializeを呼ぶ
    @breed = breed
  end

  def speak
    original = super  # 親クラスのspeakを呼ぶ
    "#{original}...ワン！"
  end
end

dog = Dog.new("ポチ", "柴犬")
dog.speak  # => "動物の声...ワン！"

# superの引数
class Parent
  def greet(name, greeting = "Hello")
    "#{greeting}, #{name}!"
  end
end

class Child < Parent
  def greet(name)
    super         # すべての引数を渡す: super(name)
    super()       # 引数なし
    super(name, "Hi")  # 明示的に引数を指定
  end
end
```

**🐍 Python:**

```python
class Animal:
    def __init__(self, name):
        self.name = name

    def speak(self):
        return "動物の声"

class Dog(Animal):
    def __init__(self, name, breed):
        super().__init__(name)     # 親の __init__ を呼ぶ（引数は明示）
        self.breed = breed

    def speak(self):
        original = super().speak() # 親の speak を呼ぶ
        return f"{original}...ワン！"

dog = Dog("ポチ", "柴犬")
dog.speak()                        # => "動物の声...ワン！"
```

重要な言語差: Rubyの `super`（引数なし）は現在のメソッドの引数を暗黙的にそのまま渡すが、Pythonの `super()` にはこの暗黙転送がない。`super().__init__(name)` のように呼ぶメソッド名と引数を毎回明示する必要がある。Rubyの `super()`（空括弧＝引数なし）に相当するのは、単に引数を渡さない `super().method()` の呼び出し。

### メソッドの検索順序

```ruby
module M1
  def foo
    "M1#foo"
  end
end

module M2
  def foo
    "M2#foo"
  end
end

class Parent
  def foo
    "Parent#foo"
  end
end

class Child < Parent
  include M1
  include M2  # 後からincludeしたものが優先

  def foo
    "Child#foo"
  end
end

Child.ancestors
# => [Child, M2, M1, Parent, Object, Kernel, BasicObject]

# メソッド検索: Child → M2 → M1 → Parent → Object → ...
```

**🐍 Python:**

```python
class M1:
    def foo(self):
        return "M1.foo"

class M2:
    def foo(self):
        return "M2.foo"

class Parent:
    def foo(self):
        return "Parent.foo"

# Pythonは多重継承。基底クラスを左から並べる
class Child(M2, M1, Parent):
    def foo(self):
        return "Child.foo"

[c.__name__ for c in Child.__mro__]
# => ['Child', 'M2', 'M1', 'Parent', 'object']
```

RubyのメソッドはモジュールをMixinして探索するが、Pythonは多重継承で基底クラスを直接並べる。探索順序は `Child.__mro__`（Method Resolution Order）で確認でき、Rubyの `ancestors` に相当する。左に書いたクラスが優先される点も同じ発想。

### 継承の制限

```ruby
# Rubyは単一継承のみ
class A; end
class B < A; end
# class C < A, B  # SyntaxError!

# 多重継承の代わりにモジュールを使う
module Flyable
  def fly
    "飛んでいます"
  end
end

module Swimmable
  def swim
    "泳いでいます"
  end
end

class Duck < Animal
  include Flyable
  include Swimmable
end

duck = Duck.new("アヒル")
duck.fly   # => "飛んでいます"
duck.swim  # => "泳いでいます"
```

**🐍 Python:**

```python
# Pythonは多重継承が可能なので、Mixin相当を基底クラスで表現する
class Flyable:
    def fly(self):
        return "飛んでいます"

class Swimmable:
    def swim(self):
        return "泳いでいます"

class Duck(Animal, Flyable, Swimmable):
    pass

duck = Duck("アヒル")
duck.fly()    # => "飛んでいます"
duck.swim()   # => "泳いでいます"
```

重要な言語差: Rubyは単一継承のみで、複数の振る舞いは `module` を `include` してMixinする。Pythonは多重継承そのものが許されているため、Rubyでモジュールにする部分を「Mixin用のクラス」として複数の基底クラスに並べる。Rubyのモジュールに近い純粋な抽象化が欲しい場合は、`abc.ABC` を継承した抽象基底クラスを使う。

## 6. オブジェクトの初期化

### initialize メソッド

```ruby
class User
  def initialize(name, email)
    @name = name
    @email = email
    @created_at = Time.now
  end
end

# new が内部で initialize を呼ぶ
user = User.new("太郎", "taro@example.com")
```

### デフォルト値とオプション

```ruby
class User
  def initialize(name, email: nil, role: "user", **options)
    @name = name
    @email = email
    @role = role
    @options = options
  end
end

User.new("太郎")
User.new("太郎", email: "taro@example.com")
User.new("太郎", email: "taro@example.com", role: "admin")
User.new("太郎", age: 25, city: "Tokyo")  # 追加オプション
```

**🐍 Python:**

```python
class User:
    def __init__(self, name, *, email=None, role="user", **options):
        self.name = name
        self.email = email
        self.role = role
        self.options = options       # 余ったキーワードを辞書で受け取る

User("太郎")
User("太郎", email="taro@example.com")
User("太郎", email="taro@example.com", role="admin")
User("太郎", age=25, city="Tokyo")   # options = {'age': 25, 'city': 'Tokyo'}
```

Rubyのキーワード引数 `email:` はPythonでは `email=None` に、残りをまとめる `**options` は同名の `**options`（辞書として受け取る）に対応する。`def __init__(self, name, *, ...)` の `*` は、それ以降の引数をキーワード専用にする指定（Rubyのキーワード引数の必須指定に近い）。

### ファクトリメソッド

```ruby
class User
  attr_reader :name, :email, :type

  def initialize(name, email, type)
    @name = name
    @email = email
    @type = type
  end

  # ファクトリメソッド
  def self.guest
    new("Guest", "guest@example.com", :guest)
  end

  def self.admin(name, email)
    new(name, email, :admin)
  end

  def self.from_hash(hash)
    new(hash[:name], hash[:email], hash[:type] || :user)
  end
end

guest = User.guest
admin = User.admin("管理者", "admin@example.com")
user = User.from_hash({ name: "太郎", email: "taro@example.com" })
```

**🐍 Python:**

```python
class User:
    def __init__(self, name, email, type):
        self.name = name
        self.email = email
        self.type = type

    @classmethod
    def guest(cls):
        return cls("Guest", "guest@example.com", "guest")

    @classmethod
    def admin(cls, name, email):
        return cls(name, email, "admin")

    @classmethod
    def from_dict(cls, data):
        return cls(data["name"], data["email"], data.get("type", "user"))

guest = User.guest()
admin = User.admin("管理者", "admin@example.com")
user = User.from_dict({"name": "太郎", "email": "taro@example.com"})
```

Rubyのファクトリメソッド `def self.guest` はPythonでは `@classmethod` で書く。`cls` はそのクラス自身を指すため、`cls(...)` で新しいインスタンスを生成できる（サブクラスから呼べば正しくサブクラスのインスタンスになる）。

### 遅延初期化

```ruby
class Config
  def database
    @database ||= load_database_config
  end

  def cache
    @cache ||= initialize_cache
  end

  private

  def load_database_config
    puts "Loading database config..."
    { host: "localhost", port: 5432 }
  end

  def initialize_cache
    puts "Initializing cache..."
    {}
  end
end

config = Config.new
# まだ何も読み込まれていない
config.database  # "Loading database config..." を出力
config.database  # 2回目はキャッシュを使用
```

**🐍 Python:**

```python
from functools import cached_property

class Config:
    @cached_property
    def database(self):
        print("Loading database config...")
        return {"host": "localhost", "port": 5432}

config = Config()
config.database   # "Loading database config..." を出力
config.database   # 2回目はキャッシュされた値を返す（再計算しない）
```

Rubyの `@database ||= ...` による遅延初期化（メモ化）イディオムは、Pythonでは `@cached_property` で表現できる。初回アクセス時にだけ計算し、結果をインスタンスに保存して以降は再利用する。

## 7. オブジェクトの比較と複製

### 等価性

```ruby
class User
  attr_reader :id, :name

  def initialize(id, name)
    @id = id
    @name = name
  end

  # 値の等価性
  def ==(other)
    return false unless other.is_a?(User)
    id == other.id
  end

  # ハッシュキーとしての等価性
  def eql?(other)
    self == other
  end

  def hash
    id.hash
  end
end

u1 = User.new(1, "太郎")
u2 = User.new(1, "太郎")
u3 = User.new(2, "次郎")

u1 == u2        # => true（値が同じ）
u1 == u3        # => false
u1.equal?(u2)   # => false（別オブジェクト）

# ハッシュのキーとして使用
hash = { u1 => "value1" }
hash[u2]  # => "value1"（eql?とhashが定義されている）
```

**🐍 Python:**

```python
class User:
    def __init__(self, id, name):
        self.id = id
        self.name = name

    def __eq__(self, other):           # == の定義
        if not isinstance(other, User):
            return NotImplemented
        return self.id == other.id

    def __hash__(self):                # 辞書/集合のキーにするため必須
        return hash(self.id)

u1 = User(1, "太郎")
u2 = User(1, "太郎")
u1 == u2          # => True（値が同じ）
u1 is u2          # => False（別オブジェクト。Rubyの equal? 相当）

{u1: "value1"}[u2]  # => "value1"
```

Rubyの `==` は Python の `__eq__`、`hash` は `__hash__` に対応する。RubyはハッシュキーのためにさらにRuby側の `eql?` を分けて定義するが、Pythonは `__eq__` と `__hash__` の2つで足りる。注意: Pythonで `__eq__` を定義すると `__hash__` が自動的に無効化される（`None` になる）ため、キーとして使うなら `__hash__` を明示的に定義し直す必要がある。オブジェクト同一性の比較（Rubyの `equal?`）は `is` 演算子。

### オブジェクトの複製

```ruby
class Container
  attr_accessor :items

  def initialize
    @items = []
  end
end

original = Container.new
original.items << "item1"

# 浅いコピー（dup）
copy = original.dup
copy.items << "item2"
original.items  # => ["item1", "item2"]（配列は共有）

# 浅いコピー（clone）
# dupとほぼ同じだが、特異メソッドとfrozen状態もコピー
clone = original.clone

# 深いコピー（自前実装）
class Container
  def deep_copy
    copy = dup
    copy.items = items.dup
    copy
  end
end

# Marshal を使った深いコピー
deep = Marshal.load(Marshal.dump(original))
```

**🐍 Python:**

```python
import copy

original = Container()
original.items.append("item1")

# 浅いコピー（Rubyの dup / clone 相当）
shallow = copy.copy(original)
shallow.items.append("item2")
original.items          # => ['item1', 'item2']（リストは共有される）

# 深いコピー（Rubyの Marshal.load(Marshal.dump(...)) 相当）
deep = copy.deepcopy(original)
deep.items.append("item3")
original.items          # => 変化しない（完全に独立）
```

Rubyの `dup`/`clone` はPythonでは `copy.copy()`、`Marshal` を使った深いコピーは `copy.deepcopy()` に対応する。浅いコピーで内部の配列が共有されてしまう挙動は両言語で共通。

### initialize_copy

```ruby
class Container
  attr_accessor :items, :metadata

  def initialize
    @items = []
    @metadata = {}
  end

  # dup/clone時に呼ばれる
  def initialize_copy(original)
    super
    @items = original.items.dup
    @metadata = original.metadata.dup
  end
end

original = Container.new
original.items << "item"
original.metadata[:key] = "value"

copy = original.dup
copy.items << "new item"
original.items  # => ["item"]（独立している）
```

### freeze

```ruby
# オブジェクトを凍結（変更不可に）
str = "hello"
str.freeze
# str << " world"  # FrozenError!

str.frozen?  # => true

# 配列の凍結
arr = [1, 2, 3]
arr.freeze
# arr << 4  # FrozenError!
# arr[0] = 10  # FrozenError!

# 注意: 中身は凍結されない
nested = [[1, 2], [3, 4]].freeze
# nested << [5]  # FrozenError!
nested[0] << 3   # OK!（内部の配列は凍結されていない）

# 完全に凍結するには
nested = [[1, 2].freeze, [3, 4].freeze].freeze
```

**🐍 Python:**

```python
from dataclasses import dataclass

# Pythonに汎用の freeze はない。用途ごとに不変な型を選ぶ
point = (10, 20)              # タプルは不変（変更しようとするとエラー）
labels = frozenset({"a", "b"})  # 不変な集合

# オブジェクトを不変にしたいなら frozen dataclass
@dataclass(frozen=True)
class Point:
    x: int
    y: int

p = Point(10, 20)
# p.x = 30                    # FrozenInstanceError!
```

重要な言語差: Rubyの `freeze` はあらゆるオブジェクトを凍結できる汎用メソッドだが、Pythonには相当する汎用機能がない。代わりに、もともと不変な型（`tuple`、`frozenset`、`str`）を使うか、`@dataclass(frozen=True)` で不変クラスを定義する。「中身までは凍結されない（浅い凍結）」という注意点はPythonのタプルでも同じ（タプル内のリストは変更できる）。

## 8. 特異メソッドと特異クラス

### 特異メソッド

```ruby
# 特定のオブジェクトだけに定義されるメソッド
str = "hello"

def str.shout
  upcase + "!"
end

str.shout  # => "HELLO!"
"hello".shout  # NoMethodError!（別のオブジェクト）

# クラスメソッドは特異メソッドの一種
class MyClass
  def self.foo  # MyClassオブジェクトの特異メソッド
    "class method"
  end
end
```

**🐍 Python:**

```python
import types

s = "hello"

# 通常の（組み込み）インスタンスに属性・メソッドは追加できない
# s.shout = ...  # AttributeError

# 自作クラスのインスタンスになら、束縛メソッドを個別に付けられる
class Box:
    pass

b = Box()
b.shout = types.MethodType(lambda self: "shout!", b)
b.shout()          # => "shout!"（この b だけが持つ）
```

重要な言語差: Rubyでは任意のオブジェクトに `def obj.method` で特異メソッドを自由に足せるが、Pythonにはこの言語機能の直接の対応物がない。`str` や `int` などの組み込み型インスタンスには属性すら追加できない（`__slots__` 相当の制約）。自作クラスのインスタンスになら `types.MethodType` で個別に関数を束縛できるが、Rubyほど日常的な手法ではない。

### 特異クラス（シングルトンクラス）

```ruby
str = "hello"

# 特異クラスにアクセス
singleton_class = str.singleton_class

# 特異クラスにメソッドを定義
singleton_class.define_method(:shout) do
  upcase + "!"
end

str.shout  # => "HELLO!"

# class << obj 構文
class << str
  def whisper
    downcase + "..."
  end
end

str.whisper  # => "hello..."
```

### クラスメソッドと特異クラス

```ruby
class MyClass
  # 方法1: self.メソッド名
  def self.method1
    "method1"
  end

  # 方法2: 特異クラスに定義
  class << self
    def method2
      "method2"
    end

    # アクセサも定義可能
    attr_accessor :config

    private

    def private_class_method
      "private"
    end
  end
end

MyClass.method1  # => "method1"
MyClass.method2  # => "method2"
MyClass.config = { key: "value" }
```

### extend

```ruby
module Greetable
  def greet
    "Hello!"
  end
end

# 特定のオブジェクトにモジュールをミックスイン
obj = Object.new
obj.extend(Greetable)
obj.greet  # => "Hello!"

# クラスにextendするとクラスメソッドになる
class MyClass
  extend Greetable
end

MyClass.greet  # => "Hello!"
```

## 9. オープンクラス

### 既存クラスの拡張

```ruby
# Rubyでは既存のクラスを再オープンできる
class String
  def shout
    upcase + "!"
  end
end

"hello".shout  # => "HELLO!"

# 組み込みクラスも拡張可能
class Integer
  def to_binary
    to_s(2)
  end
end

42.to_binary  # => "101010"
```

**🐍 Python:**

```python
# 自作クラスは後からメソッドを足せる（モンキーパッチ）
class User:
    def __init__(self, name):
        self.name = name

def shout(self):
    return self.name.upper() + "!"

User.shout = shout           # 既存クラスに追加
User("hello").shout()        # => "HELLO!"

# ただし組み込み型は拡張できない
# str.shout = shout          # TypeError!（Rubyとの大きな違い）
```

重要な言語差: Pythonでも自作クラスなら後からメソッドを差し込める（モンキーパッチ）が、Rubyと違って `str`・`int`・`list` などの組み込み型（Cで実装された型）は再オープンして拡張できない。Rubyの `Refinements`（スコープ限定の安全な拡張）に相当する仕組みもPythonにはない。

### モンキーパッチの注意点

```ruby
# 危険: 既存メソッドの上書き
class Array
  def first  # 元のメソッドが消える！
    "常にこれ"
  end
end

[1, 2, 3].first  # => "常にこれ"（壊れた）

# 安全: alias で元を保存
class String
  alias_method :original_reverse, :reverse

  def reverse
    puts "Reversing..."
    original_reverse
  end
end
```

### Refinements（安全な拡張）

```ruby
# モジュールで refinement を定義
module StringExtensions
  refine String do
    def shout
      upcase + "!"
    end
  end
end

# refinement を使用するスコープで有効化
class MyClass
  using StringExtensions

  def process(str)
    str.shout  # => 使える
  end
end

MyClass.new.process("hello")  # => "HELLO!"
# "hello".shout  # NoMethodError!（スコープ外では使えない）

# ファイルスコープで有効化
using StringExtensions
"hello".shout  # => "HELLO!"
```

## 10. Struct と Data

### Struct

```ruby
# 簡易的なクラスを作成
User = Struct.new(:name, :email, :age)

user = User.new("太郎", "taro@example.com", 25)
user.name   # => "太郎"
user.email  # => "taro@example.com"
user.age    # => 25

# 配列のようにアクセス
user[0]     # => "太郎"
user[:name] # => "太郎"

# イテレーション
user.each { |value| puts value }
user.to_a   # => ["太郎", "taro@example.com", 25]
user.to_h   # => {:name=>"太郎", :email=>"taro@example.com", :age=>25}

# メソッドを追加
User = Struct.new(:name, :email) do
  def greeting
    "Hello, #{name}!"
  end
end

# keyword_init オプション（Ruby 2.5+）
User = Struct.new(:name, :email, keyword_init: true)
user = User.new(name: "太郎", email: "taro@example.com")
```

**🐍 Python:**

```python
from dataclasses import dataclass, astuple, asdict

# ミュータブルな dataclass が Struct に最も近い
@dataclass
class User:
    name: str
    email: str
    age: int

user = User("太郎", "taro@example.com", 25)
user.name = "次郎"                     # ミュータブル（書き換え可能）
astuple(user)                          # => ('次郎', 'taro@example.com', 25)
asdict(user)                           # => {'name': '次郎', ...}

# メソッドも普通に追加できる
@dataclass
class User:
    name: str
    email: str

    def greeting(self):
        return f"Hello, {self.name}!"
```

Rubyの `Struct.new` はPythonの `@dataclass` が最も近い（ミュータブルで、`to_a`/`to_h` は `astuple`/`asdict` に対応）。`keyword_init: true` に相当する挙動は、dataclassでは初めからキーワード引数で生成できる点でカバーされる。

### Data（Ruby 3.2+）

```ruby
# イミュータブルな値オブジェクト
Point = Data.define(:x, :y)

point = Point.new(x: 10, y: 20)
point.x  # => 10
point.y  # => 20

# イミュータブル
# point.x = 30  # NoMethodError!

# 値の変更は with で新しいオブジェクトを作成
new_point = point.with(x: 30)
new_point.x  # => 30
point.x      # => 10（元は変わらない）

# メソッドを追加
Point = Data.define(:x, :y) do
  def distance_from_origin
    Math.sqrt(x**2 + y**2)
  end
end

point = Point.new(x: 3, y: 4)
point.distance_from_origin  # => 5.0

# パターンマッチング
case point
in Point(x: 0, y: 0)
  "原点"
in Point(x:, y:) if x == y
  "対角線上"
in Point(x:, y:)
  "(#{x}, #{y})"
end
```

**🐍 Python:**

```python
from dataclasses import dataclass, replace
import math

# イミュータブルな値オブジェクトは frozen dataclass
@dataclass(frozen=True)
class Point:
    x: int
    y: int

    def distance_from_origin(self):
        return math.sqrt(self.x**2 + self.y**2)

point = Point(x=10, y=20)
point.x                        # => 10
# point.x = 30                 # FrozenInstanceError!

# with 相当は replace（新しいインスタンスを返す）
new_point = replace(point, x=30)
new_point.x                    # => 30
point.x                        # => 10（元は変わらない）

# パターンマッチング（Python 3.10+）
match point:
    case Point(x=0, y=0):
        result = "原点"
    case Point(x=x, y=y) if x == y:
        result = "対角線上"
    case Point(x=x, y=y):
        result = f"({x}, {y})"
```

RubyのイミュータブルなDataは、Pythonでは `@dataclass(frozen=True)` に対応する。値を変えた新オブジェクトを作る `point.with(x: 30)` は `dataclasses.replace(point, x=30)` に、Rubyのパターンマッチング `case/in` はPython 3.10+の `match/case` に相当する。

### Struct vs Data vs Class

```ruby
# Struct: ミュータブル、軽量
User = Struct.new(:name, :age)
user = User.new("太郎", 25)
user.name = "次郎"  # OK

# Data: イミュータブル、値オブジェクト向け
Point = Data.define(:x, :y)
point = Point.new(x: 10, y: 20)
# point.x = 30  # エラー

# Class: フル機能、複雑なロジック向け
class User
  attr_accessor :name, :age

  def initialize(name, age)
    @name = name
    @age = age
  end

  def adult?
    age >= 18
  end
end
```

### 使い分けの指針

| 用途 | 推奨 |
|-----|------|
| 単純なデータ保持 | Struct |
| 値オブジェクト（座標、金額など） | Data |
| ビジネスロジックを持つ | Class |
| 設定値、オプション | Struct / Hash |
| API レスポンス | Struct / Data |

## 実践例: 完全なクラス設計

```ruby
class BankAccount
  # 定数
  MINIMUM_BALANCE = 0

  # クラスインスタンス変数
  @total_accounts = 0

  class << self
    attr_accessor :total_accounts

    def create(holder_name, initial_deposit = 0)
      account = new(holder_name, initial_deposit)
      @total_accounts += 1
      account
    end
  end

  # インスタンス変数とアクセサ
  attr_reader :account_number, :holder_name, :balance

  def initialize(holder_name, initial_deposit = 0)
    @account_number = generate_account_number
    @holder_name = holder_name
    @balance = initial_deposit
    @transactions = []
  end

  # パブリックメソッド
  def deposit(amount)
    raise ArgumentError, "入金額は正の数である必要があります" if amount <= 0

    @balance += amount
    record_transaction(:deposit, amount)
    balance
  end

  def withdraw(amount)
    raise ArgumentError, "出金額は正の数である必要があります" if amount <= 0
    raise InsufficientFundsError, "残高不足です" if balance - amount < MINIMUM_BALANCE

    @balance -= amount
    record_transaction(:withdrawal, amount)
    balance
  end

  def transfer(amount, to_account)
    withdraw(amount)
    to_account.deposit(amount)
  end

  def statement
    @transactions.map do |t|
      "#{t[:date]} | #{t[:type]} | #{t[:amount]} | #{t[:balance]}"
    end.join("\n")
  end

  def to_s
    "口座番号: #{account_number}, 名義: #{holder_name}, 残高: #{balance}円"
  end

  # プライベートメソッド
  private

  def generate_account_number
    "#{Time.now.to_i}#{rand(1000..9999)}"
  end

  def record_transaction(type, amount)
    @transactions << {
      date: Time.now,
      type: type,
      amount: amount,
      balance: balance
    }
  end

  # カスタム例外
  class InsufficientFundsError < StandardError; end
end

# 使用例
account1 = BankAccount.create("山田太郎", 10000)
account2 = BankAccount.create("鈴木花子", 5000)

account1.deposit(5000)
account1.withdraw(3000)
account1.transfer(2000, account2)

puts account1  # => 口座番号: ..., 名義: 山田太郎, 残高: 10000円
puts account1.statement
```
