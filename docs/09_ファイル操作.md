## 1. ファイルの基本操作

### File.open

```ruby
# ブロック形式（推奨）- 自動的にクローズ
File.open("example.txt", "r") do |file|
  content = file.read
  puts content
end

# 変数形式 - 手動でクローズが必要
file = File.open("example.txt", "r")
content = file.read
file.close

# ensure を使う場合
file = File.open("example.txt", "r")
begin
  content = file.read
ensure
  file.close
end
```

**🐍 Python:**

```python
# with 文がブロック形式に相当（抜けると自動的にクローズ）
with open("example.txt", "r") as file:
    content = file.read()
    print(content)

# 変数形式 - 手動でクローズが必要
file = open("example.txt", "r")
content = file.read()
file.close()

# try/finally を使う場合（Ruby の begin/ensure 相当）
file = open("example.txt", "r")
try:
    content = file.read()
finally:
    file.close()
```

Ruby の `File.open(...) do |f| ... end` は Python の `with open(...) as f:` に相当し、どちらもブロックを抜けると自動でクローズする。

### オープンモード

| モード | 説明 |
|-------|------|
| `"r"` | 読み取り専用（デフォルト） |
| `"w"` | 書き込み専用（既存は上書き） |
| `"a"` | 追記モード |
| `"r+"` | 読み書き両用 |
| `"w+"` | 読み書き両用（既存は上書き） |
| `"a+"` | 読み書き両用（追記） |
| `"b"` | バイナリモード（組み合わせ可能: "rb", "wb"） |

```ruby
# 読み取り
File.open("data.txt", "r") { |f| f.read }

# 書き込み（上書き）
File.open("data.txt", "w") { |f| f.write("Hello") }

# 追記
File.open("data.txt", "a") { |f| f.write("World") }

# バイナリ読み込み
File.open("image.png", "rb") { |f| f.read }

# バイナリ書き込み
File.open("output.bin", "wb") { |f| f.write(binary_data) }
```

**🐍 Python:**

```python
# 読み取り
with open("data.txt", "r") as f:
    f.read()

# 書き込み（上書き）
with open("data.txt", "w") as f:
    f.write("Hello")

# 追記
with open("data.txt", "a") as f:
    f.write("World")

# バイナリ読み込み / 書き込み（"b" を付ける）
with open("image.png", "rb") as f:
    f.read()
with open("output.bin", "wb") as f:
    f.write(binary_data)
```

モード文字列（`"r"`, `"w"`, `"a"`, `"r+"`, `"b"` の組み合わせ）は Ruby とほぼ共通。ただし Python はテキストモードで開くと `str`、バイナリモード（`"b"`）で開くと `bytes` を返す点が異なる。

### ファイルの存在確認と削除

```ruby
# 存在確認
File.exist?("example.txt")      # => true/false
File.file?("example.txt")       # ファイルか
File.directory?("dir")          # ディレクトリか

# 削除
File.delete("example.txt")      # ファイル削除
FileUtils.rm("example.txt")     # 同上
FileUtils.rm_f("example.txt")   # 存在しなくてもエラーにならない

# 複数ファイル削除
File.delete("file1.txt", "file2.txt")
FileUtils.rm(["file1.txt", "file2.txt"])

# ディレクトリごと削除
FileUtils.rm_rf("directory")
```

**🐍 Python:**

```python
from pathlib import Path
import shutil

# 存在確認
Path("example.txt").exists()      # File.exist? 相当
Path("example.txt").is_file()     # File.file? 相当
Path("dir").is_dir()              # File.directory? 相当

# 削除
Path("example.txt").unlink()              # ファイル削除（File.delete 相当）
Path("example.txt").unlink(missing_ok=True)  # 無ければ無視（rm_f 相当）

# ディレクトリごと削除
shutil.rmtree("directory")        # FileUtils.rm_rf 相当
```

Python には `File.exist?` のような `?` 付きメソッド名は無く、`exists()` / `is_file()` / `is_dir()` を使う。

### ファイルのコピーと移動

```ruby
require 'fileutils'

# コピー
FileUtils.cp("source.txt", "dest.txt")
FileUtils.cp("source.txt", "dir/")           # ディレクトリへ
FileUtils.cp_r("source_dir", "dest_dir")     # 再帰的コピー

# 移動（リネーム）
FileUtils.mv("old.txt", "new.txt")
FileUtils.mv("file.txt", "dir/")
File.rename("old.txt", "new.txt")            # 基本的なリネーム

# 複数ファイルをコピー/移動
FileUtils.cp(["file1.txt", "file2.txt"], "dir/")
FileUtils.mv(Dir.glob("*.txt"), "backup/")
```

**🐍 Python:**

```python
import shutil

# コピー
shutil.copy("source.txt", "dest.txt")       # FileUtils.cp 相当
shutil.copytree("source_dir", "dest_dir")   # FileUtils.cp_r 相当（再帰的）

# 移動（リネーム）
shutil.move("old.txt", "new.txt")           # FileUtils.mv 相当

from pathlib import Path
Path("old.txt").rename("new.txt")           # File.rename 相当
```

## 2. ファイルの読み込み

### 一括読み込み

```ruby
# 全体を文字列として読み込み
content = File.read("example.txt")

# エンコーディング指定
content = File.read("example.txt", encoding: "UTF-8")
content = File.read("shift_jis.txt", encoding: "Shift_JIS:UTF-8")

# バイナリ読み込み
binary = File.binread("image.png")
```

**🐍 Python:**

```python
from pathlib import Path

# 全体を文字列として読み込み
content = Path("example.txt").read_text()              # File.read 相当
content = open("example.txt").read()                   # 同上（明示的に開く）

# エンコーディング指定
content = Path("example.txt").read_text(encoding="utf-8")

# バイナリ読み込み
binary = Path("image.png").read_bytes()                # File.binread 相当
```

Ruby の `File.read("shift_jis.txt", encoding: "Shift_JIS:UTF-8")` のような「読み込み時のエンコーディング変換」は Python では単に `encoding="shift_jis"` で読み込む形になる（Python 内部表現は常に Unicode）。

### 行ごとの読み込み

```ruby
# 配列として取得
lines = File.readlines("example.txt")
lines = File.readlines("example.txt", chomp: true)  # 改行除去

# 1行ずつ処理（メモリ効率が良い）
File.foreach("example.txt") do |line|
  puts line
end

# chomp付き
File.foreach("example.txt", chomp: true) do |line|
  puts line
end
```

**🐍 Python:**

```python
# 配列（リスト）として取得
lines = open("example.txt").readlines()          # File.readlines 相当（改行は残る）
lines = Path("example.txt").read_text().splitlines()  # chomp: true 相当（改行除去）

# 1行ずつ処理（メモリ効率が良い）— File.foreach 相当
with open("example.txt") as f:
    for line in f:
        print(line, end="")      # line には改行が含まれる

# chomp 付き
with open("example.txt") as f:
    for line in f:
        print(line.rstrip("\n"))
```

Ruby の `File.foreach` は Python では `for line in open(...)`（またはファイルオブジェクトの反復）に相当する。Python の各行は既定で末尾に `\n` を含むため、Ruby の `chomp: true` に当たる処理は `rstrip("\n")` などで行う。

### IO メソッド

```ruby
File.open("example.txt", "r") do |file|
  # 1行読み込み
  line = file.gets
  line = file.readline  # EOFで例外

  # 全行読み込み
  lines = file.readlines

  # 1文字読み込み
  char = file.getc

  # 指定バイト読み込み
  bytes = file.read(1024)

  # イテレーション
  file.each_line { |line| puts line }
  file.each_char { |char| puts char }
  file.each_byte { |byte| puts byte }

  # 位置操作
  file.pos            # 現在位置
  file.pos = 0        # 先頭に移動
  file.rewind         # 先頭に移動
  file.seek(100)      # 100バイト目に移動
  file.seek(-10, IO::SEEK_END)  # 末尾から10バイト前

  # 状態確認
  file.eof?           # 終端か
  file.closed?        # クローズ済みか
end
```

**🐍 Python:**

```python
with open("example.txt", "r") as file:
    line = file.readline()      # 1行読み込み（Ruby の gets 相当、EOF では "" ）
    lines = file.readlines()    # 全行読み込み
    char = file.read(1)         # 1文字読み込み
    chunk = file.read(1024)     # 指定バイト（文字）読み込み

    # イテレーション
    for line in file:
        print(line)

    # 位置操作
    file.tell()                 # 現在位置（Ruby の pos 相当）
    file.seek(0)                # 先頭に移動（rewind 相当）
    file.seek(100)              # 100バイト目に移動
    file.seek(-10, 2)           # 末尾から10バイト前（2 = SEEK_END）

    # 状態確認
    file.closed                 # クローズ済みか（属性、() は付けない）
```

Python のファイルには Ruby の `eof?` に相当する直接のメソッドは無く、`read`/`readline` が空を返したことで終端を判定する。また `closed` はメソッドではなく属性。

### 大きなファイルの処理

```ruby
# 1行ずつ処理（メモリ効率が良い）
File.foreach("large_file.txt") do |line|
  process(line)
end

# バッファサイズ指定
File.open("large_file.bin", "rb") do |file|
  while (chunk = file.read(4096))
    process_chunk(chunk)
  end
end

# lazy enumeration（Ruby 2.0+）
File.open("large_file.txt") do |file|
  file.each_line
      .lazy
      .select { |line| line.include?("ERROR") }
      .first(10)
      .each { |line| puts line }
end
```

**🐍 Python:**

```python
# 1行ずつ処理（メモリ効率が良い）
with open("large_file.txt") as f:
    for line in f:
        process(line)

# バッファサイズ指定
with open("large_file.bin", "rb") as f:
    while chunk := f.read(4096):     # := は Python 3.8+ の代入式
        process_chunk(chunk)

# ジェネレータ式で遅延評価（Ruby の lazy 相当）
with open("large_file.txt") as f:
    errors = (line for line in f if "ERROR" in line)
    for line in list(errors)[:10]:
        print(line, end="")
```

Ruby の `Enumerator#lazy` に相当するのは Python のジェネレータ（`(...)` 式や `itertools`）。ファイルオブジェクトの反復自体が遅延評価なので、巨大ファイルでも一度に全体を読み込まない。

## 3. ファイルの書き込み

### 一括書き込み

```ruby
# 文字列を書き込み（上書き）
File.write("output.txt", "Hello, World!")

# 追記
File.write("output.txt", "追加テキスト", mode: "a")

# バイナリ書き込み
File.binwrite("output.bin", binary_data)
```

**🐍 Python:**

```python
from pathlib import Path

# 文字列を書き込み（上書き）
Path("output.txt").write_text("Hello, World!")     # File.write 相当

# 追記（write_text には追記モードが無いので open を使う）
with open("output.txt", "a") as f:                 # mode: "a" 相当
    f.write("追加テキスト")

# バイナリ書き込み
Path("output.bin").write_bytes(binary_data)        # File.binwrite 相当
```

Ruby の `File.write(path, str, mode: "a")` のような一括追記は、Python の `Path.write_text()` には無いため `open(path, "a")` を使う。

### ストリーム書き込み

```ruby
File.open("output.txt", "w") do |file|
  file.write("Hello")       # 改行なし
  file.puts("World")        # 改行あり
  file.print("Ruby")        # 改行なし
  file.printf("%s: %d\n", "Count", 42)  # フォーマット

  # 配列を書き込み
  file.puts(["Line 1", "Line 2", "Line 3"])

  # フラッシュ
  file.flush  # バッファを強制的に書き込み
end
```

**🐍 Python:**

```python
with open("output.txt", "w") as file:
    file.write("Hello")                  # 改行なし（Ruby の write 相当）
    print("World", file=file)            # 改行あり（Ruby の puts 相当）
    file.write("Ruby")                   # print(..., end="") でも可
    print(f"Count: {42}", file=file)     # printf 相当（f-string で整形）

    # リストの各要素を1行ずつ
    file.writelines(l + "\n" for l in ["Line 1", "Line 2", "Line 3"])

    file.flush()                         # バッファを強制書き込み
```

Ruby の `file.puts`（改行付き）に最も近いのは `print(..., file=file)`。Python の `file.write` は改行を付けず、`writelines` も改行を自動追加しない点に注意。

### 排他的書き込み

```ruby
# 既存ファイルがあればエラー
File.open("output.txt", File::WRONLY | File::CREAT | File::EXCL) do |file|
  file.write("New content")
end
```

### アトミック書き込み

```ruby
require 'tempfile'

def atomic_write(filename, content)
  dir = File.dirname(filename)
  basename = File.basename(filename)

  Tempfile.create(basename, dir) do |temp|
    temp.write(content)
    temp.close
    FileUtils.mv(temp.path, filename)
  end
end

# 書き込み中に読まれても古いファイルが見える
atomic_write("config.json", new_config.to_json)
```

## 4. ファイル情報の取得

### File クラスのメソッド

```ruby
# 基本情報
File.size("example.txt")           # バイト数
File.size?("example.txt")          # サイズ（0なら nil）
File.zero?("example.txt")          # サイズが0か

# 種別判定
File.file?("example.txt")          # ファイルか
File.directory?("dir")             # ディレクトリか
File.symlink?("link")              # シンボリックリンクか
File.readable?("example.txt")      # 読み取り可能か
File.writable?("example.txt")      # 書き込み可能か
File.executable?("script.rb")      # 実行可能か

# 日時
File.mtime("example.txt")          # 更新日時
File.atime("example.txt")          # アクセス日時
File.ctime("example.txt")          # 作成日時（Unix: inode変更日時）
File.birthtime("example.txt")      # 作成日時（対応OSのみ）
```

**🐍 Python:**

```python
from pathlib import Path
import os

p = Path("example.txt")

# 基本情報
p.stat().st_size          # バイト数（File.size 相当）

# 種別判定
p.is_file()               # File.file? 相当
p.is_dir()                # File.directory? 相当
p.is_symlink()            # File.symlink? 相当
os.access(p, os.R_OK)     # 読み取り可能か（File.readable? 相当）
os.access(p, os.W_OK)     # 書き込み可能か（File.writable? 相当）

# 日時（float の Unix タイムスタンプ）
p.stat().st_mtime         # 更新日時（File.mtime 相当）
p.stat().st_atime         # アクセス日時（File.atime 相当）
p.stat().st_ctime         # inode変更日時（File.ctime 相当）
```

Ruby では `File.mtime` などが直接 `Time` オブジェクトを返すが、Python の `st_mtime` は float の Unix タイムスタンプ。`datetime.fromtimestamp(p.stat().st_mtime)` で日時に変換する。

### File::Stat

```ruby
stat = File.stat("example.txt")

stat.size           # バイト数
stat.mtime          # 更新日時
stat.atime          # アクセス日時
stat.mode           # パーミッション
stat.uid            # 所有者ID
stat.gid            # グループID
stat.dev            # デバイスID
stat.ino            # inode番号

stat.file?          # ファイルか
stat.directory?     # ディレクトリか
stat.readable?      # 読み取り可能か

# lstat: シンボリックリンク自体の情報
lstat = File.lstat("symlink")
```

### パーミッション

```ruby
# パーミッション確認
mode = File.stat("example.txt").mode
puts sprintf("%o", mode)  # => 100644

# パーミッション変更
File.chmod(0644, "example.txt")
File.chmod(0755, "script.rb")

# 所有者変更（Unix系のみ、要権限）
File.chown(uid, gid, "example.txt")

# FileUtils での変更
FileUtils.chmod(0644, "example.txt")
FileUtils.chmod_R(0755, "directory")  # 再帰的
```

## 5. ディレクトリ操作

### ディレクトリの作成と削除

```ruby
require 'fileutils'

# 作成
Dir.mkdir("new_dir")                  # 単一ディレクトリ
FileUtils.mkdir_p("path/to/new_dir")  # 親ディレクトリも作成

# 削除
Dir.rmdir("empty_dir")                # 空のディレクトリのみ
FileUtils.rm_rf("directory")          # 再帰的に削除
```

**🐍 Python:**

```python
from pathlib import Path
import shutil

# 作成
Path("new_dir").mkdir()                      # Dir.mkdir 相当
Path("path/to/new_dir").mkdir(parents=True, exist_ok=True)  # mkdir_p 相当

# 削除
Path("empty_dir").rmdir()                    # 空のみ（Dir.rmdir 相当）
shutil.rmtree("directory")                   # 再帰的に削除（rm_rf 相当）
```

Ruby の `FileUtils.mkdir_p`（親ごと作成）は Python では `mkdir(parents=True)`。`exist_ok=True` を付けると既存でもエラーにならない。

### ディレクトリの内容取得

```ruby
# エントリ一覧
Dir.entries(".")        # => [".", "..", "file1.txt", ...]

# . と .. を除く
Dir.children(".")       # => ["file1.txt", "file2.txt", ...]

# イテレーション
Dir.foreach(".") do |entry|
  puts entry
end

Dir.each_child(".") do |entry|
  puts entry
end

# ブロック形式
Dir.open(".") do |dir|
  dir.each { |entry| puts entry }
end
```

**🐍 Python:**

```python
import os
from pathlib import Path

# エントリ一覧（. と .. は含まれない）
os.listdir(".")            # => ["file1.txt", "file2.txt", ...]（Dir.entries 相当だが . / .. は無い）
[p.name for p in Path(".").iterdir()]   # Dir.children 相当

# イテレーション
for entry in os.listdir("."):
    print(entry)

for entry in Path(".").iterdir():       # Path オブジェクトで反復
    print(entry.name)
```

Ruby の `Dir.entries` は `.` と `..` を含むが、Python の `os.listdir` / `Path.iterdir` はこれらを含まない（Ruby の `Dir.children` に近い）。

### glob（パターンマッチング）

```ruby
# ワイルドカード
Dir.glob("*.txt")                # カレントの.txtファイル
Dir.glob("**/*.rb")              # 再帰的に.rbファイル
Dir.glob("**/test_*.rb")         # test_で始まるrbファイル
Dir.glob("{*.rb,*.txt}")         # .rb または .txt
Dir.glob("[a-z]*.txt")           # a-zで始まる.txt
Dir.glob("**/*", File::FNM_DOTMATCH)  # 隠しファイルも含む

# Dir[] は glob のショートカット
Dir["*.txt"]
Dir["**/*.rb"]

# Pathname
require 'pathname'
Pathname.glob("**/*.rb")
```

**🐍 Python:**

```python
import glob
from pathlib import Path

# glob モジュール
glob.glob("*.txt")                    # カレントの .txt（Dir.glob 相当）
glob.glob("**/*.rb", recursive=True)  # 再帰的（** には recursive=True が必要）

# pathlib（推奨）
list(Path(".").glob("*.txt"))         # カレントの .txt
list(Path(".").rglob("*.rb"))         # 再帰的（Dir.glob("**/*.rb") 相当）
```

Ruby の `Dir.glob("**/*.rb")` の `**` は Python の `glob.glob(..., recursive=True)` か `Path.rglob(...)` に相当。`glob.glob` はデフォルトでは `**` を再帰展開しない点に注意。

### カレントディレクトリ

```ruby
# 取得
Dir.pwd                 # => "/home/user/project"
Dir.getwd               # 同上

# 変更
Dir.chdir("/tmp")
Dir.chdir("/tmp") { |path| puts Dir.pwd }  # ブロック後に戻る

# ホームディレクトリ
Dir.home                # => "/home/user"
Dir.home("otheruser")   # => "/home/otheruser"
```

**🐍 Python:**

```python
import os
from pathlib import Path

# 取得
os.getcwd()             # => "/home/user/project"（Dir.pwd 相当）

# 変更
os.chdir("/tmp")        # Dir.chdir 相当

# ホームディレクトリ
Path.home()             # => Path("/home/user")（Dir.home 相当）
os.path.expanduser("~") # => "/home/user"
```

Ruby の `Dir.chdir("/tmp") { ... }` のように「ブロック終了後に元へ戻す」機能は標準では無いので、`contextlib.chdir`（Python 3.11+）や手動で戻す処理を使う。

## 6. パス操作

### File クラスのパスメソッド

```ruby
path = "/home/user/documents/report.txt"

# 分解
File.dirname(path)      # => "/home/user/documents"
File.basename(path)     # => "report.txt"
File.basename(path, ".txt")  # => "report"
File.extname(path)      # => ".txt"

# 結合
File.join("home", "user", "file.txt")  # => "home/user/file.txt"

# 絶対パス
```

**🐍 Python:**

```python
import os
from pathlib import Path

# 分解（os.path）
os.path.dirname(path)       # => "/home/user/documents"（File.dirname 相当）
os.path.basename(path)      # => "report.txt"（File.basename 相当）
os.path.splitext(path)[1]   # => ".txt"（File.extname 相当）

# pathlib（推奨）
p = Path(path)
p.parent                    # => Path("/home/user/documents")
p.name                      # => "report.txt"
p.stem                      # => "report"（File.basename(path, ".txt") 相当）
p.suffix                    # => ".txt"（File.extname 相当）

# 結合

# 絶対パス / 正規化
```

Ruby の `File.extname` は `.txt` を返し、`File.basename(path, ".txt")` で拡張子を除いた名前を得るが、Python では `Path.suffix`（拡張子）と `Path.stem`（拡張子抜きの名前）が対応する。

## 8. ファイルのロック

### flock

```ruby
# 排他ロック
File.open("data.txt", "r+") do |file|
  file.flock(File::LOCK_EX)  # 排他ロック取得
  content = file.read
  file.rewind
  file.write(process(content))
  file.flock(File::LOCK_UN)  # ロック解放
end

# 共有ロック（読み取り用）
File.open("data.txt", "r") do |file|
  file.flock(File::LOCK_SH)  # 共有ロック
  content = file.read
  file.flock(File::LOCK_UN)
end

# ノンブロッキング
File.open("data.txt", "r+") do |file|
  if file.flock(File::LOCK_EX | File::LOCK_NB)
    # ロック取得成功
    process(file)
    file.flock(File::LOCK_UN)
  else
    # ロック取得失敗（他がロック中）
    puts "Could not acquire lock"
  end
end
```

**🐍 Python:**

```python
import fcntl   # Unix 系のみ（Windows は msvcrt や portalocker を使う）

# 排他ロック
with open("data.txt", "r+") as file:
    fcntl.flock(file, fcntl.LOCK_EX)   # File::LOCK_EX 相当
    content = file.read()
    file.seek(0)
    file.write(process(content))
    fcntl.flock(file, fcntl.LOCK_UN)   # ロック解放

# 共有ロック（読み取り用）
with open("data.txt", "r") as file:
    fcntl.flock(file, fcntl.LOCK_SH)   # File::LOCK_SH 相当
    content = file.read()

# ノンブロッキング（取得できないと BlockingIOError）
with open("data.txt", "r+") as file:
    try:
        fcntl.flock(file, fcntl.LOCK_EX | fcntl.LOCK_NB)
        process(file)
    except BlockingIOError:
        print("Could not acquire lock")
```

Ruby の `file.flock(File::LOCK_EX)` は Python では `fcntl.flock(file, fcntl.LOCK_EX)` に相当する（`fcntl` は Unix 専用）。Ruby は取得失敗時に `false` を返すが、Python の非ブロッキング版は `BlockingIOError` 例外を送出する点が異なる。

### ロックファイルパターン

```ruby
class FileLock
  def initialize(path)
    @path = path
    @lock_path = "#{path}.lock"
  end

  def with_lock
    File.open(@lock_path, File::RDWR | File::CREAT) do |lock_file|
      lock_file.flock(File::LOCK_EX)
      begin
        yield
      ensure
        lock_file.flock(File::LOCK_UN)
      end
    end
  end
end

# 使用
lock = FileLock.new("important_data.txt")
lock.with_lock do
  # 排他的な処理
  data = File.read("important_data.txt")
  File.write("important_data.txt", process(data))
end
```

### ログファイルの処理

```ruby
class LogAnalyzer
  def initialize(log_path)
    @log_path = log_path
  end

  def errors
    grep(/ERROR/)
  end

  def warnings
    grep(/WARN/)
  end

  def search(pattern)
    grep(pattern)
  end

  def tail(n = 10)
    lines = []
    File.foreach(@log_path) { |line| lines << line }
    lines.last(n)
  end

  private

  def grep(pattern)
    matches = []
    File.foreach(@log_path) do |line|

  end
```

```ruby
class FileWatcher
  def initialize(path, interval: 1)
    @path = path
    @interval = interval
    @last_mtime = File.mtime(path) if File.exist?(path)
  end

  def watch(&block)
    loop do
      current_mtime = File.exist?(@path) ? File.mtime(@path) : nil

      if current_mtime != @last_mtime
        block.call(@path, @last_mtime, current_mtime)
        @last_mtime = current_mtime
      end

      sleep @interval
    end
  end
end

# 使用
```

### バックアップ機能

```ruby
require 'fileutils'

class Backup
  def initialize(source_dir, backup_dir)
    @source_dir = source_dir
    @backup_dir = backup_dir
  end

  def create
    timestamp = Time.now.strftime("%Y%m%d_%H%M%S")
    backup_path = File.join(@backup_dir, timestamp)

    FileUtils.mkdir_p(backup_path)
    FileUtils.cp_r(Dir.glob("#{@source_dir}/*"), backup_path)

    puts "Backup created: #{backup_path}"
    backup_path
  end

  def list
    Dir.children(@backup_dir).sort.reverse
  end

  def restore(backup_name)
    backup_path = File.join(@backup_dir, backup_name)
    raise "Backup not found: #{backup_name}" unless File.directory?(backup_path)

    FileUtils.rm_rf(Dir.glob("#{@source_dir}/*"))
    FileUtils.cp_r(Dir.glob("#{backup_path}/*"), @source_dir)

    puts "Restored from: #{backup_path}"
  end

  def cleanup(keep: 5)
    backups = list
    return if backups.size <= keep

    backups[keep..].each do |backup_name|
      path = File.join(@backup_dir, backup_name)
      FileUtils.rm_rf(path)
      puts "Removed old backup: #{backup_name}"
    end
  end
end

# 使用
backup = Backup.new("./data", "./backups")
backup.create
backup.cleanup(keep: 7)
```
