## 1. ファイルの基本操作

### File.open

```ruby
# ブロック形式（推奨）- 自動的にクローズ
File.open("example.txt", "r") do |file|
  content = file.read
  puts content
end

# 変数形式 - 手動でクローズが必要
file = File.open("example.txt", "r")
content = file.read
file.close

# ensure を使う場合
file = File.open("example.txt", "r")
begin
  content = file.read
ensure
  file.close
end
```

### オープンモード

| モード | 説明 |
|-------|------|
| `"r"` | 読み取り専用（デフォルト） |
| `"w"` | 書き込み専用（既存は上書き） |
| `"a"` | 追記モード |
| `"r+"` | 読み書き両用 |
| `"w+"` | 読み書き両用（既存は上書き） |
| `"a+"` | 読み書き両用（追記） |
| `"b"` | バイナリモード（組み合わせ可能: "rb", "wb"） |

```ruby
# 読み取り
File.open("data.txt", "r") { |f| f.read }

# 書き込み（上書き）
File.open("data.txt", "w") { |f| f.write("Hello") }

# 追記
File.open("data.txt", "a") { |f| f.write("World") }

# バイナリ読み込み
File.open("image.png", "rb") { |f| f.read }

# バイナリ書き込み
File.open("output.bin", "wb") { |f| f.write(binary_data) }
```

### ファイルの存在確認と削除

```ruby
# 存在確認
File.exist?("example.txt")      # => true/false
File.file?("example.txt")       # ファイルか
File.directory?("dir")          # ディレクトリか

# 削除
File.delete("example.txt")      # ファイル削除
FileUtils.rm("example.txt")     # 同上
FileUtils.rm_f("example.txt")   # 存在しなくてもエラーにならない

# 複数ファイル削除
File.delete("file1.txt", "file2.txt")
FileUtils.rm(["file1.txt", "file2.txt"])

# ディレクトリごと削除
FileUtils.rm_rf("directory")
```

### ファイルのコピーと移動

```ruby
require 'fileutils'

# コピー
FileUtils.cp("source.txt", "dest.txt")
FileUtils.cp("source.txt", "dir/")           # ディレクトリへ
FileUtils.cp_r("source_dir", "dest_dir")     # 再帰的コピー

# 移動（リネーム）
FileUtils.mv("old.txt", "new.txt")
FileUtils.mv("file.txt", "dir/")
File.rename("old.txt", "new.txt")            # 基本的なリネーム

# 複数ファイルをコピー/移動
FileUtils.cp(["file1.txt", "file2.txt"], "dir/")
FileUtils.mv(Dir.glob("*.txt"), "backup/")
```

## 2. ファイルの読み込み

### 一括読み込み

```ruby
# 全体を文字列として読み込み
content = File.read("example.txt")

# エンコーディング指定
content = File.read("example.txt", encoding: "UTF-8")
content = File.read("shift_jis.txt", encoding: "Shift_JIS:UTF-8")

# バイナリ読み込み
binary = File.binread("image.png")
```

### 行ごとの読み込み

```ruby
# 配列として取得
lines = File.readlines("example.txt")
lines = File.readlines("example.txt", chomp: true)  # 改行除去

# 1行ずつ処理（メモリ効率が良い）
File.foreach("example.txt") do |line|
  puts line
end

# chomp付き
File.foreach("example.txt", chomp: true) do |line|
  puts line
end
```

### IO メソッド

```ruby
File.open("example.txt", "r") do |file|
  # 1行読み込み
  line = file.gets
  line = file.readline  # EOFで例外

  # 全行読み込み
  lines = file.readlines

  # 1文字読み込み
  char = file.getc

  # 指定バイト読み込み
  bytes = file.read(1024)

  # イテレーション
  file.each_line { |line| puts line }
  file.each_char { |char| puts char }
  file.each_byte { |byte| puts byte }

  # 位置操作
  file.pos            # 現在位置
  file.pos = 0        # 先頭に移動
  file.rewind         # 先頭に移動
  file.seek(100)      # 100バイト目に移動
  file.seek(-10, IO::SEEK_END)  # 末尾から10バイト前

  # 状態確認
  file.eof?           # 終端か
  file.closed?        # クローズ済みか
end
```

### 大きなファイルの処理

```ruby
# 1行ずつ処理（メモリ効率が良い）
File.foreach("large_file.txt") do |line|
  process(line)
end

# バッファサイズ指定
File.open("large_file.bin", "rb") do |file|
  while (chunk = file.read(4096))
    process_chunk(chunk)
  end
end

# lazy enumeration（Ruby 2.0+）
File.open("large_file.txt") do |file|
  file.each_line
      .lazy
      .select { |line| line.include?("ERROR") }
      .first(10)
      .each { |line| puts line }
end
```

## 3. ファイルの書き込み

### 一括書き込み

```ruby
# 文字列を書き込み（上書き）
File.write("output.txt", "Hello, World!")

# 追記
File.write("output.txt", "追加テキスト", mode: "a")

# バイナリ書き込み
File.binwrite("output.bin", binary_data)
```

### ストリーム書き込み

```ruby
File.open("output.txt", "w") do |file|
  file.write("Hello")       # 改行なし
  file.puts("World")        # 改行あり
  file.print("Ruby")        # 改行なし
  file.printf("%s: %d\n", "Count", 42)  # フォーマット

  # 配列を書き込み
  file.puts(["Line 1", "Line 2", "Line 3"])

  # フラッシュ
  file.flush  # バッファを強制的に書き込み
end
```

### 排他的書き込み

```ruby
# 既存ファイルがあればエラー
File.open("output.txt", File::WRONLY | File::CREAT | File::EXCL) do |file|
  file.write("New content")
end
```

### アトミック書き込み

```ruby
require 'tempfile'

def atomic_write(filename, content)
  dir = File.dirname(filename)
  basename = File.basename(filename)

  Tempfile.create(basename, dir) do |temp|
    temp.write(content)
    temp.close
    FileUtils.mv(temp.path, filename)
  end
end

# 書き込み中に読まれても古いファイルが見える
atomic_write("config.json", new_config.to_json)
```

## 4. ファイル情報の取得

### File クラスのメソッド

```ruby
# 基本情報
File.size("example.txt")           # バイト数
File.size?("example.txt")          # サイズ（0なら nil）
File.zero?("example.txt")          # サイズが0か

# 種別判定
File.file?("example.txt")          # ファイルか
File.directory?("dir")             # ディレクトリか
File.symlink?("link")              # シンボリックリンクか
File.readable?("example.txt")      # 読み取り可能か
File.writable?("example.txt")      # 書き込み可能か
File.executable?("script.rb")      # 実行可能か

# 日時
File.mtime("example.txt")          # 更新日時
File.atime("example.txt")          # アクセス日時
File.ctime("example.txt")          # 作成日時（Unix: inode変更日時）
File.birthtime("example.txt")      # 作成日時（対応OSのみ）
```

### File::Stat

```ruby
stat = File.stat("example.txt")

stat.size           # バイト数
stat.mtime          # 更新日時
stat.atime          # アクセス日時
stat.mode           # パーミッション
stat.uid            # 所有者ID
stat.gid            # グループID
stat.dev            # デバイスID
stat.ino            # inode番号

stat.file?          # ファイルか
stat.directory?     # ディレクトリか
stat.readable?      # 読み取り可能か

# lstat: シンボリックリンク自体の情報
lstat = File.lstat("symlink")
```

### パーミッション

```ruby
# パーミッション確認
mode = File.stat("example.txt").mode
puts sprintf("%o", mode)  # => 100644

# パーミッション変更
File.chmod(0644, "example.txt")
File.chmod(0755, "script.rb")

# 所有者変更（Unix系のみ、要権限）
File.chown(uid, gid, "example.txt")

# FileUtils での変更
FileUtils.chmod(0644, "example.txt")
FileUtils.chmod_R(0755, "directory")  # 再帰的
```

## 5. ディレクトリ操作

### ディレクトリの作成と削除

```ruby
require 'fileutils'

# 作成
Dir.mkdir("new_dir")                  # 単一ディレクトリ
FileUtils.mkdir_p("path/to/new_dir")  # 親ディレクトリも作成

# 削除
Dir.rmdir("empty_dir")                # 空のディレクトリのみ
FileUtils.rm_rf("directory")          # 再帰的に削除
```

### ディレクトリの内容取得

```ruby
# エントリ一覧
Dir.entries(".")        # => [".", "..", "file1.txt", ...]

# . と .. を除く
Dir.children(".")       # => ["file1.txt", "file2.txt", ...]

# イテレーション
Dir.foreach(".") do |entry|
  puts entry
end

Dir.each_child(".") do |entry|
  puts entry
end

# ブロック形式
Dir.open(".") do |dir|
  dir.each { |entry| puts entry }
end
```

### glob（パターンマッチング）

```ruby
# ワイルドカード
Dir.glob("*.txt")                # カレントの.txtファイル
Dir.glob("**/*.rb")              # 再帰的に.rbファイル
Dir.glob("**/test_*.rb")         # test_で始まるrbファイル
Dir.glob("{*.rb,*.txt}")         # .rb または .txt
Dir.glob("[a-z]*.txt")           # a-zで始まる.txt
Dir.glob("**/*", File::FNM_DOTMATCH)  # 隠しファイルも含む

# Dir[] は glob のショートカット
Dir["*.txt"]
Dir["**/*.rb"]

# Pathname
require 'pathname'
Pathname.glob("**/*.rb")
```

### カレントディレクトリ

```ruby
# 取得
Dir.pwd                 # => "/home/user/project"
Dir.getwd               # 同上

# 変更
Dir.chdir("/tmp")
Dir.chdir("/tmp") { |path| puts Dir.pwd }  # ブロック後に戻る

# ホームディレクトリ
Dir.home                # => "/home/user"
Dir.home("otheruser")   # => "/home/otheruser"
```

## 6. パス操作

### File クラスのパスメソッド

```ruby
path = "/home/user/documents/report.txt"

# 分解
File.dirname(path)      # => "/home/user/documents"
File.basename(path)     # => "report.txt"
File.basename(path, ".txt")  # => "report"
File.extname(path)      # => ".txt"

# 結合
File.join("home", "user", "file.txt")  # => "home/user/file.txt"

# 絶対パス
File.expand_path("~/file.txt")         # => "/home/user/file.txt"
File.expand_path("../file.txt", "/home/user/docs")
# => "/home/user/file.txt"

# 正規化
File.realpath("./symlink")  # シンボリックリンクを解決

# 分割
File.split("/path/to/file.txt")  # => ["/path/to", "file.txt"]
```

### Pathname クラス（推奨）

```ruby
require 'pathname'

path = Pathname.new("/home/user/documents/report.txt")

# 分解
path.dirname        # => Pathname:/home/user/documents
path.basename       # => Pathname:report.txt
path.basename(".txt")  # => Pathname:report
path.extname        # => ".txt"

# 判定
path.absolute?      # => true
path.relative?      # => false
path.exist?         # => true/false
path.file?          # => true/false
path.directory?     # => false

# 結合
path = Pathname.new("/home/user")
path / "documents" / "file.txt"  # => Pathname:/home/user/documents/file.txt
path.join("documents", "file.txt")

# 操作
path.read           # ファイル内容を読み込み
path.write("content")  # 書き込み
path.open("r") { |f| f.read }
path.children       # 子エントリ
path.each_child { |child| puts child }
path.mkdir          # ディレクトリ作成

# 相対パス
base = Pathname.new("/home/user")
target = Pathname.new("/home/user/docs/file.txt")
target.relative_path_from(base)  # => Pathname:docs/file.txt
```

### __FILE__ と __dir__

```ruby
# 現在のファイルのパス
puts __FILE__       # => "/path/to/current_file.rb"

# 現在のディレクトリ
puts __dir__        # => "/path/to"

# 相対パスでファイルを読み込み
config_path = File.join(__dir__, "config", "settings.yml")
File.read(config_path)

# Pathnameと組み合わせ
root = Pathname.new(__dir__).parent
lib_path = root / "lib"
```

## 8. ファイルのロック

### flock

```ruby
# 排他ロック
File.open("data.txt", "r+") do |file|
  file.flock(File::LOCK_EX)  # 排他ロック取得
  content = file.read
  file.rewind
  file.write(process(content))
  file.flock(File::LOCK_UN)  # ロック解放
end

# 共有ロック（読み取り用）
File.open("data.txt", "r") do |file|
  file.flock(File::LOCK_SH)  # 共有ロック
  content = file.read
  file.flock(File::LOCK_UN)
end

# ノンブロッキング
File.open("data.txt", "r+") do |file|
  if file.flock(File::LOCK_EX | File::LOCK_NB)
    # ロック取得成功
    process(file)
    file.flock(File::LOCK_UN)
  else
    # ロック取得失敗（他がロック中）
    puts "Could not acquire lock"
  end
end
```

### ロックファイルパターン

```ruby
class FileLock
  def initialize(path)
    @path = path
    @lock_path = "#{path}.lock"
  end

  def with_lock
    File.open(@lock_path, File::RDWR | File::CREAT) do |lock_file|
      lock_file.flock(File::LOCK_EX)
      begin
        yield
      ensure
        lock_file.flock(File::LOCK_UN)
      end
    end
  end
end

# 使用
lock = FileLock.new("important_data.txt")
lock.with_lock do
  # 排他的な処理
  data = File.read("important_data.txt")
  File.write("important_data.txt", process(data))
end
```

## 10. 実践的なファイル操作

### 設定ファイルの管理

```ruby
require 'yaml'

class Config
  def initialize(path)
    @path = path
    @data = load_config
  end

  def [](key)
    @data[key.to_s]
  end

  def []=(key, value)
    @data[key.to_s] = value
  end

  def save
    File.write(@path, @data.to_yaml)
  end

  private

  def load_config
    return {} unless File.exist?(@path)
    YAML.safe_load_file(@path) || {}
  end
end

# 使用
config = Config.new("config.yml")
config["api_key"] = "secret"
config.save
```

### ログファイルの処理

```ruby
class LogAnalyzer
  def initialize(log_path)
    @log_path = log_path
  end

  def errors
    grep(/ERROR/)
  end

  def warnings
    grep(/WARN/)
  end

  def search(pattern)
    grep(pattern)
  end

  def tail(n = 10)
    lines = []
    File.foreach(@log_path) { |line| lines << line }
    lines.last(n)
  end

  private

  def grep(pattern)
    matches = []
    File.foreach(@log_path) do |line|
      matches << line if line.match?(pattern)
    end
    matches
  end
end

# 使用
analyzer = LogAnalyzer.new("app.log")
analyzer.errors.each { |line| puts line }
```

### ファイル監視

```ruby
require 'rb-inotify'  # Linux
# または
require 'listen'      # クロスプラットフォーム

# Listen gem を使用
listener = Listen.to('watched_directory') do |modified, added, removed|
  puts "Modified: #{modified}"
  puts "Added: #{added}"
  puts "Removed: #{removed}"
end

listener.start
sleep  # メインスレッドを維持

# 簡易的なポーリング実装
class FileWatcher
  def initialize(path, interval: 1)
    @path = path
    @interval = interval
    @last_mtime = File.mtime(path) if File.exist?(path)
  end

  def watch(&block)
    loop do
      current_mtime = File.exist?(@path) ? File.mtime(@path) : nil

      if current_mtime != @last_mtime
        block.call(@path, @last_mtime, current_mtime)
        @last_mtime = current_mtime
      end

      sleep @interval
    end
  end
end

# 使用
watcher = FileWatcher.new("config.yml")
watcher.watch do |path, old_time, new_time|
  puts "File changed: #{path}"
  reload_config
end
```

### バックアップ機能

```ruby
require 'fileutils'

class Backup
  def initialize(source_dir, backup_dir)
    @source_dir = source_dir
    @backup_dir = backup_dir
  end

  def create
    timestamp = Time.now.strftime("%Y%m%d_%H%M%S")
    backup_path = File.join(@backup_dir, timestamp)

    FileUtils.mkdir_p(backup_path)
    FileUtils.cp_r(Dir.glob("#{@source_dir}/*"), backup_path)

    puts "Backup created: #{backup_path}"
    backup_path
  end

  def list
    Dir.children(@backup_dir).sort.reverse
  end

  def restore(backup_name)
    backup_path = File.join(@backup_dir, backup_name)
    raise "Backup not found: #{backup_name}" unless File.directory?(backup_path)

    FileUtils.rm_rf(Dir.glob("#{@source_dir}/*"))
    FileUtils.cp_r(Dir.glob("#{backup_path}/*"), @source_dir)

    puts "Restored from: #{backup_path}"
  end

  def cleanup(keep: 5)
    backups = list
    return if backups.size <= keep

    backups[keep..].each do |backup_name|
      path = File.join(@backup_dir, backup_name)
      FileUtils.rm_rf(path)
      puts "Removed old backup: #{backup_name}"
    end
  end
end

# 使用
backup = Backup.new("./data", "./backups")
backup.create
backup.cleanup(keep: 7)
```
