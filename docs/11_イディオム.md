## 1. Rubyらしいコード

### 明示的より暗黙的

```ruby
# 悪い例: 冗長
if array.length > 0
  # ...
end

# 良い例: 簡潔
if array.any?
  # ...
end

# 悪い例: 不要なreturn
def add(a, b)
  return a + b
end

# 良い例: 最後の式が戻り値
def add(a, b)
  a + b
end

# 悪い例: 不要なself
def full_name
  self.first_name + " " + self.last_name
end

# 良い例: self省略
def full_name
  "#{first_name} #{last_name}"
end
```

### 述語メソッド

```ruby
# ?で終わるメソッドは真偽値を返す
def admin?
  role == "admin"
end

def valid?
  errors.empty?
end

def empty?
  @items.empty?
end

# 使用
if user.admin?
  show_admin_panel
end
```

### 破壊的メソッド

```ruby
# !で終わるメソッドは元のオブジェクトを変更
array = [3, 1, 2]
sorted = array.sort    # 新しい配列を返す
array.sort!            # array自体をソート

# または例外を発生させる
user.save   # 失敗時はfalseを返す
user.save!  # 失敗時は例外を発生
```

## 2. 条件分岐のイディオム

### 後置if/unless

```ruby
# 1行で書ける場合
return if user.nil?
raise ArgumentError unless valid?
puts "Hello" if verbose?

# 複雑な場合は通常のif
if user.admin? && user.active?
  grant_access
end
```

### 三項演算子

```ruby
# シンプルな条件
status = active? ? "有効" : "無効"

# 複雑な場合はifを使う
result = if complex_condition?
           complex_true_value
         else
           complex_false_value
         end
```

### 論理演算子でのデフォルト値

```ruby
# || でデフォルト値
name = params[:name] || "Guest"

# ||= で遅延初期化
@users ||= User.all

# 注意: false や nil を区別する必要がある場合
# ||= は false を上書きしてしまう
@enabled ||= true  # @enabled が false でも true になる

# nilのみチェックする場合
@enabled = default_enabled if @enabled.nil?
# または
@enabled //= default_enabled  # Ruby 2.3+（実験的）
```

### 安全ナビゲーション演算子

```ruby
# &. でnilチェック
user&.name           # userがnilならnil
user&.address&.city  # チェーン可能

# 配列やハッシュでも
array&.first
hash&.dig(:user, :name)

# メソッド呼び出しと組み合わせ
user&.admin? || false
```

### case/when の活用

```ruby
# 型によるマッチング
case value
when String
  process_string(value)
when Array
  process_array(value)
when Hash
  process_hash(value)
end

# 範囲でマッチング
case score
when 90..100 then "A"
when 80...90 then "B"
when 70...80 then "C"
else "F"
end

# 正規表現でマッチング
case email
when /@gmail\.com$/
  "Gmail"
when /@yahoo\.com$/
  "Yahoo"
else
  "Other"
end

# Procでマッチング
even = ->(n) { n.even? }
case number
when even
  "偶数"
else
  "奇数"
end
```

## 3. コレクション操作

### map/collect

```ruby
# 変換
numbers = [1, 2, 3, 4, 5]
squares = numbers.map { |n| n ** 2 }

# メソッド参照
names = users.map(&:name)

# 複数の値を返す
users.map { |u| [u.id, u.name] }

# ネストした配列を平坦化
arrays.flat_map { |arr| arr.map(&:upcase) }
```

### select/reject

```ruby
# フィルタリング
evens = numbers.select(&:even?)
adults = users.select { |u| u.age >= 18 }

# 除外
odds = numbers.reject(&:even?)
inactive = users.reject(&:active?)

# 組み合わせ
active_adults = users.select(&:active?).select { |u| u.age >= 18 }
```

### reduce/inject

```ruby
# 合計
sum = numbers.reduce(0) { |acc, n| acc + n }
sum = numbers.reduce(:+)
sum = numbers.sum  # Ruby 2.4+

# オブジェクトの構築
hash = array.reduce({}) { |h, item| h.merge(item.id => item) }

# each_with_object（reduceの代替）
hash = array.each_with_object({}) { |item, h| h[item.id] = item }
```

### group_by/partition

```ruby
# グループ化
by_status = users.group_by(&:status)
# => { "active" => [...], "inactive" => [...] }

# 2分割
passed, failed = students.partition { |s| s.score >= 60 }
```

### any?/all?/none?/one?

```ruby
# いずれか
users.any?(&:admin?)

# すべて
users.all?(&:active?)

# どれも〜ない
users.none?(&:banned?)

# ちょうど1つ
users.one?(&:superadmin?)
```

### find/detect

```ruby
# 最初の一致
admin = users.find(&:admin?)
admin = users.detect(&:admin?)

# 見つからない場合のデフォルト
user = users.find { |u| u.id == id } || User.new
```

### sort/sort_by

```ruby
# ソート
sorted = users.sort { |a, b| a.name <=> b.name }

# sort_by（より高速で読みやすい）
sorted = users.sort_by(&:name)
sorted = users.sort_by { |u| [u.last_name, u.first_name] }

# 降順
sorted = users.sort_by(&:created_at).reverse
sorted = users.sort_by { |u| -u.created_at.to_i }
```

### take/drop/first/last

```ruby
# 先頭から
top_5 = users.take(5)
top_5 = users.first(5)

# 末尾から
last_5 = users.last(5)

# スキップ
remaining = users.drop(5)

# 条件付き
users.take_while(&:active?)
users.drop_while(&:pending?)
```

## 4. メソッドチェーン

### 基本的なチェーン

```ruby
result = users
  .select(&:active?)
  .reject(&:admin?)
  .sort_by(&:name)
  .map(&:email)
  .first(10)
```

### tap でデバッグ

```ruby
result = users
  .select(&:active?)
  .tap { |arr| puts "Active users: #{arr.size}" }
  .sort_by(&:name)
  .tap { |arr| puts "First: #{arr.first.name}" }
  .map(&:email)
```

### then/yield_self

```ruby
# 値を変換してチェーン継続
result = fetch_data
  .then { |data| parse(data) }
  .then { |parsed| transform(parsed) }
  .then { |transformed| save(transformed) }

# 条件付き変換
value = input
  .then { |v| v.nil? ? default : v }
  .then { |v| process(v) }
```

### チェーン可能なメソッド設計

```ruby
class QueryBuilder
  def initialize
    @conditions = []
    @order = nil
    @limit = nil
  end

  def where(condition)
    @conditions << condition
    self  # selfを返してチェーン可能に
  end

  def order(field)
    @order = field
    self
  end

  def limit(n)
    @limit = n
    self
  end

  def to_sql
    # SQLを生成
  end
end

query = QueryBuilder.new
  .where("status = 'active'")
  .where("age >= 18")
  .order("created_at DESC")
  .limit(10)
```

## 5. ブロックのイディオム

### ブロック vs do...end

```ruby
# {} は1行向け
items.each { |item| process(item) }
names = users.map { |u| u.name }

# do...end は複数行向け
items.each do |item|
  validate(item)
  process(item)
  save(item)
end

# 慣例: 戻り値を使う場合は {}
result = items.map { |item| transform(item) }

# 副作用が目的の場合は do...end
items.each do |item|
  puts item
end
```

### Symbol#to_proc

```ruby
# メソッド名をシンボルで
names = users.map(&:name)
# 同等: users.map { |u| u.name }

active = users.select(&:active?)
# 同等: users.select { |u| u.active? }

total = numbers.reduce(&:+)
# 同等: numbers.reduce { |a, b| a + b }
```

### ブロックをメソッドに渡す

```ruby
def with_retry(max_attempts = 3)
  attempts = 0
  begin
    attempts += 1
    yield
  rescue => e
    retry if attempts < max_attempts
    raise
  end
end

with_retry(5) { api_call }
```

### Proc/Lambda の活用

```ruby
# フィルタの再利用
active_filter = ->(u) { u.active? }
admin_filter = ->(u) { u.admin? }

active_users = users.select(&active_filter)
active_admins = users.select(&active_filter).select(&admin_filter)

# 設定として
VALIDATORS = {
  email: ->(v) { v.match?(/\A[\w+\-.]+@[a-z\d\-.]+\.[a-z]+\z/i) },
  phone: ->(v) { v.match?(/\A\d{10,11}\z/) }
}

def validate(field, value)
  VALIDATORS[field]&.call(value)
end
```

## 6. 文字列操作

### 式展開

```ruby
# 基本
"Hello, #{name}!"

# 式を含む
"Total: #{items.sum}"
"Status: #{active? ? 'Active' : 'Inactive'}"

# メソッド呼び出し
"Name: #{user.name.upcase}"
```

### ヒアドキュメント

```ruby
# 基本
sql = <<~SQL
  SELECT *
  FROM users
  WHERE active = true
SQL

# 式展開
message = <<~MSG
  Dear #{name},

  Your order ##{order_id} has been shipped.
MSG

# メソッドチェーン
query = <<~SQL.strip.gsub(/\s+/, ' ')
  SELECT *
  FROM users
  WHERE active = true
SQL
```

### フォーマット

```ruby
# % 演算子
"Name: %s, Age: %d" % [name, age]

# sprintf / format
sprintf("%.2f", price)
format("%<name>s is %<age>d", name: "Alice", age: 25)
```

### 正規表現

```ruby
# マッチング
email =~ /\A[\w+\-.]+@[a-z\d\-.]+\.[a-z]+\z/i
email.match?(/pattern/)  # 高速（Ruby 2.4+）

# 置換
text.gsub(/\s+/, ' ')
text.gsub(/\b\w/) { |match| match.upcase }

# キャプチャ
if match = text.match(/(\d+)-(\d+)-(\d+)/)
  year, month, day = match.captures
end
```

## 7. オブジェクト生成

### ファクトリメソッド

```ruby
class User
  def self.guest
    new(name: "Guest", role: :guest)
  end

  def self.admin(name)
    new(name: name, role: :admin)
  end

  def self.from_hash(hash)
    new(**hash)
  end
end

guest = User.guest
admin = User.admin("Alice")
```

### ビルダーパターン

```ruby
class QueryBuilder
  def self.build
    builder = new
    yield(builder)
    builder.to_query
  end

  # ...
end

query = QueryBuilder.build do |b|
  b.select(:name, :email)
  b.from(:users)
  b.where(active: true)
end
```

### Struct の活用

```ruby
# 簡易クラス
Point = Struct.new(:x, :y) do
  def distance_from_origin
    Math.sqrt(x**2 + y**2)
  end
end

point = Point.new(3, 4)
point.distance_from_origin  # => 5.0

# 値オブジェクト
Money = Struct.new(:amount, :currency, keyword_init: true)
price = Money.new(amount: 1000, currency: "JPY")
```

### freeze でイミュータブル

```ruby
# 定数の保護
DEFAULTS = {
  timeout: 30,
  retries: 3
}.freeze

# オブジェクトの凍結
class Config
  def initialize(options)
    @options = options.freeze
  end
end
```

## 8. メタプログラミング

### define_method

```ruby
class Model
  ATTRIBUTES = [:name, :email, :age]

  ATTRIBUTES.each do |attr|
    define_method(attr) { @attributes[attr] }
    define_method("#{attr}=") { |value| @attributes[attr] = value }
  end
end
```

### method_missing

```ruby
class OpenStruct
  def initialize(hash = {})
    @data = hash
  end

  def method_missing(name, *args)
    if name.to_s.end_with?("=")
      @data[name.to_s.chop.to_sym] = args.first
    else
      @data[name]
    end
  end

  def respond_to_missing?(name, include_private = false)
    true
  end
end
```

### クラスマクロ

```ruby
module Validatable
  def self.included(base)
    base.extend(ClassMethods)
  end

  module ClassMethods
    def validates(attribute, **options)
      validations[attribute] = options
    end

    def validations
      @validations ||= {}
    end
  end

  def valid?
    self.class.validations.all? do |attr, opts|
      value = send(attr)
      opts.all? { |type, arg| validate(type, value, arg) }
    end
  end
end

class User
  include Validatable
  validates :name, presence: true
  validates :email, format: /\A.+@.+\z/
end
```

## 9. パフォーマンス最適化

### 文字列の凍結

```ruby
# 文字列リテラルを凍結（メモリ節約）
# frozen_string_literal: true

# または個別に
CONSTANT = "frozen".freeze
```

### 遅延評価

```ruby
# lazy で遅延評価
huge_array
  .lazy
  .select(&:valid?)
  .map(&:transform)
  .first(10)
  .to_a

# 無限シーケンス
(1..Float::INFINITY)
  .lazy
  .select { |n| n % 3 == 0 }
  .take(10)
  .to_a
```

### キャッシュ

```ruby
# インスタンスレベル
def expensive_calculation
  @result ||= perform_calculation
end

# クラスレベル
class Config
  def self.instance
    @instance ||= new
  end
end

# 条件付きキャッシュ
def data(force_reload: false)
  @data = nil if force_reload
  @data ||= load_data
end
```

### 効率的なコレクション操作

```ruby
# 悪い例: 複数回のイテレーション
result = array.map { |x| x * 2 }.select { |x| x > 10 }.first(5)

# 良い例: 1回のイテレーション
result = []
array.each do |x|
  transformed = x * 2
  if transformed > 10
    result << transformed
    break if result.size >= 5
  end
end

# または lazy
result = array.lazy.map { |x| x * 2 }.select { |x| x > 10 }.first(5).to_a
```

### Set の活用

```ruby
require 'set'

# 悪い例: 配列での検索（O(n)）
valid_statuses = ["active", "pending", "approved"]
valid_statuses.include?(status)

# 良い例: Setでの検索（O(1)）
VALID_STATUSES = Set["active", "pending", "approved"]
VALID_STATUSES.include?(status)
```

## 10. アンチパターン

### 避けるべきパターン

```ruby
# 悪い例: 例外を握りつぶす
begin
  risky_operation
rescue
  # 何もしない
end

# 良い例: 最低限ログを残す
begin
  risky_operation
rescue => e
  logger.error("Error: #{e.message}")
  raise  # または適切に処理
end
```

```ruby
# 悪い例: 過度なメタプログラミング
class User
  def method_missing(name, *args)
    # 何でも受け入れる
  end
end

# 良い例: 明示的なメソッド定義
class User
  attr_accessor :name, :email
end
```

```ruby
# 悪い例: グローバル変数
$config = load_config

# 良い例: シングルトンまたは依存性注入
class Config
  def self.instance
    @instance ||= new
  end
end
```

```ruby
# 悪い例: 深いネスト
def process(data)
  if data
    if data[:users]
      if data[:users].any?
        data[:users].each do |user|
          if user[:active]
            # ...
          end
        end
      end
    end
  end
end

# 良い例: 早期リターン
def process(data)
  return unless data
  return unless data[:users]&.any?

  data[:users].each do |user|
    next unless user[:active]
    # ...
  end
end
```

```ruby
# 悪い例: 神クラス
class User
  def save; end
  def send_email; end
  def generate_report; end
  def calculate_salary; end
  def update_inventory; end
end

# 良い例: 単一責任
class User
  def save; end
end

class UserMailer
  def send_email(user); end
end

class ReportGenerator
  def generate(user); end
end
```

```ruby
# 悪い例: マジックナンバー
if user.age >= 18
  # ...
end

# 良い例: 定数
LEGAL_AGE = 18

if user.age >= LEGAL_AGE
  # ...
end
```

```ruby
# 悪い例: N+1クエリ
users = User.all
users.each do |user|
  puts user.posts.count  # 各ユーザーでクエリ発行
end

# 良い例: Eager loading
users = User.includes(:posts).all
users.each do |user|
  puts user.posts.size
end
```

## Ruby Style Guide のポイント

| カテゴリ | 推奨 |
|---------|------|
| インデント | スペース2つ |
| 行の長さ | 最大120文字 |
| メソッド | 10行以下を目安 |
| ブロック | 1行なら{}、複数行ならdo...end |
| 命名 | snake_case（変数/メソッド）、CamelCase（クラス） |
| 文字列 | 式展開はダブルクォート、それ以外はシングル |
| 配列 | %w[], %i[] を活用 |
| 比較 | == を使用、eql? は特殊な場合のみ |
| 真偽値 | and/or より && / \|\| |
